        진입경과분 = 0;
        저수익처리 = False;
        저수익대기 = False;
        저수익수량 = 0;
        저수익전수량 = 0;
        장부수량 = 0;
        장부실현손익 = 0;
        장부평가손익 = 0;
        장부전체손익 = 0;
        장부전체수익률 = 0;
        장부원금 = 0;
        청산진단봉 = -1;
        청산진단사유 = -1;
        청산진단수량 = -1;
        당일사용 = False;
        일차처리 = False;
        일차대기 = False;
        일차수량 = 0;
        일차전수량 = 0;
        일차사유 = 0;
        일차완료선물봉 = -1;
        일차완료일 = 0;
        잔량경과분 = 0;
        가격비율하락확인 = False;
        일차기준수익률 = 0;
        보유확인 = False;
        청산요청 = False;
        모델유효 = False;
        자기수량 = 0;
        상대수량 = 0;
        모델투입금 = 0;
        자기첫만남가격 = 0;
        상대첫만남가격 = 0;
    }
    Else
    {
        당일사용 = True;
        청산요청 = True;
        청산사유 = 4;
    }
}

// V16: 거래량 및 누적 무거래 봉수 필터 제거. 동일 시각의 유효 가격은 필요.

// 자기 전략 포지션을 확인한 뒤에만 보유 상태로 전환합니다.
// MarketPosition은 전략 포지션이며 실제 계좌 체결확인 함수가 아닙니다.
If MarketPosition == 1 && 보유확인 == False Then
{
    보유확인 = True;
    진입대기 = False;
    If 신호일 == Bdate && Index == 신호봉 + 1 && 동시봉 Then
    {
        자기진입 = 자기첫만남가격;
        상대진입 = 상대첫만남가격;
        모델투입금 = (자기첫만남가격 * 자기수량 + 상대첫만남가격 * 상대수량) * 250000;
        모델유효 = 모델투입금 > 0;
    }
    Else
    {
        // 진입 모델을 확정하지 못해도 강제청산하지 않음.
        // 모델이 없으면 합산 익절/손절 계산은 불가하며 시간/날짜 청산만 유지.
        모델유효 = False;
    }
}
// 해당 옵션 장부: 실제 체결가가 아니라 기존 AtMarket의 수량감소 봉 시가 모델.
// 실현+미실현은 최초 해당 옵션 원금으로 나눔. 상대 옵션 청산은 추정하지 않음.
If 자기첫만남가격 > 0 && 자기수량 > 0 && (보유확인 || 장부수량 > 0) Then
{
    장부원금 = 자기첫만남가격 * 자기수량 * 250000;
    장부감소수량 = Max(0, 장부수량 - IntPortion(CurrentContracts));
    If 장부감소수량 > 0 Then
    {
        장부실현손익 = 장부실현손익 + (O - 자기첫만남가격) * 장부감소수량 * 250000;
        If 진단출력사용 == 1 Then
            MessageLog("WSF V34 MODEL_FILL date=%.0f time=%.0f qty=%.0f modelOpen=%.4f realizedWon=%.0f", sDate, sTime, 장부감소수량, O, 장부실현손익);
    }
    장부수량 = IntPortion(CurrentContracts);
    장부평가손익 = (C - 자기첫만남가격) * 장부수량 * 250000;
    장부전체손익 = 장부실현손익 + 장부평가손익;
    장부전체수익률 = 장부전체손익 / 장부원금 * 100;
    If 진단출력사용 == 1 && (Index != 장부진단봉 || 장부감소수량 > 0) Then
    {
        장부진단봉 = Index;
        MessageLog("WSF V34 OWN_LEDGER realizedWon=%.0f unrealizedWon=%.0f totalWon=%.0f totalPct=%.4f remaining=%.0f", 장부실현손익, 장부평가손익, 장부전체손익, 장부전체수익률, 장부수량);
    }
}

// 포지션이 없어져도 당일사용은 유지하여 재진입 차단.
If 보유확인 && MarketPosition == 0 Then
{
    보유확인 = False;
    저수익대기 = False;
    일차대기 = False;
    청산요청 = False;
    모델유효 = False;
}
If 진입대기 && MarketPosition == 0 && Index > 신호봉 + 1 Then
    진입대기 = False;

// 전략 진입 확인 후 첫 동시각 Data2를 시간 기준으로 저장. 같은 날에만 계산.
If MarketPosition == 1 && 보유확인 && Data2(sDate) == sDate &&
   Data2(sTime) == sTime && Data2(DataCompress) == 2 && Data2(BarInterval) == 1 Then
{
    If 진입선물봉 < 0 Then
    {
        진입선물봉 = Data2(DayIndex);
        진입선물일 = sDate;
    }
    If 진입선물일 == sDate Then
        진입경과분 = Max(0, Data2(DayIndex) - 진입선물봉);
}
If MarketPosition == 1 && 저수익대기 && CurrentContracts <= 저수익전수량 - 저수익수량 Then
    저수익대기 = False;

// 잔량 익절 판단 전에 1차 전략 수량 감소를 확인합니다.
// 실제 계좌/상대 차트 체결 확인은 아닙니다.
If MarketPosition == 1 && 일차대기 && CurrentContracts <= 일차전수량 - 일차수량 Then
    일차대기 = False;

// 1차 전략청산 확인 후 동일시각 Data2가 확보된 때부터 대기 시간을 계산.
// Data1/Data3 빈 봉 수 대신 Data2 1분봉 DayIndex 차이를 사용.
If MarketPosition == 1 && 일차처리 && 일차대기 == False &&
   Data2(sDate) == sDate && Data2(sTime) == sTime &&
   Data2(DataCompress) == 2 && Data2(BarInterval) == 1 Then
{
    If 일차완료선물봉 < 0 Then
    {
        일차완료선물봉 = Data2(DayIndex);
        일차완료일 = sDate;
    }
    If 일차완료일 == sDate Then
        잔량경과분 = Max(0, Data2(DayIndex) - 일차완료선물봉);
    // 1차 이전의 하락은 기억하지 않음. 옵션 동시봉 누락과 독립적으로 관찰.
    If 함수계산가능 == 1 && 이탈_삼선준비 == 1 &&
       이탈_두구간최대가격범위 > 0 && 이탈_두구간가격비율 < 100 Then
        가격비율하락확인 = True;
}

// 시간 청산은 상대 데이터가 없거나 손익 계산 불가해도 실행.
If MarketPosition == 1 Then
{
    If sTime >= 전량청산 ||
       (NextBarSdate == sDate && NextBarStime >= 전량청산) ||
       NextBarSdate != sDate Then
    {
        청산요청 = True;
        청산사유 = 3;
    }
    If 모델유효 && 동시봉 && 청산요청 == False Then
    {
        자기평가 = C;
        상대평가 = Data3(C);
        // 보조지표와 동일: 첫 만남 봉 양쪽 종가 기준 동일금액50:50. 비용/분할 실현손익 미반영.
        합산손익 = ((자기평가 - 자기첫만남가격) * 자기수량 +
                    (상대평가 - 상대첫만남가격) * 상대수량) * 250000;
        합산수익률 = ((자기평가 / 자기첫만남가격 - 1) +
                        (상대평가 / 상대첫만남가격 - 1)) * 50;
        // 잔량 개별 수익률: 최초 매수가 기준. 먼저 판 물량과 상대 옵션 손익 제외.
        자기잔량수익률 = (자기평가 - 자기진입) / 자기진입 * 100;
        잔량비율유효 = Data2(sDate) == sDate && Data2(sTime) == sTime &&
                         Data2(DataCompress) == 2 && Data2(BarInterval) == 1 &&
                         함수계산가능 == 1 && 이탈_삼선준비 == 1 &&
                         이탈_두구간최대가격범위 > 0;
        // 만기일 기본보호 익절만 삼선70 이상. 목표익절/개별0%미만 청산은 예외.
        // 손절/시간/날짜 청산에는 적용하지 않음. 진입 삼선 기준 Input과 별개.
        만기잔량삼선허용 = Data2(sDate) == sDate && Data2(sTime) == sTime &&
                             Data2(DataCompress) == 2 && Data2(BarInterval) == 1 &&
                             함수계산가능 == 1 && 삼선_목표준비 == 1 &&
                             삼선_지속목표차_최고 > 0 && 삼선_지속목표차_비율 >= 70;
        If 합산수익률 <= -손절률 Then
        {
            청산요청 = True;
            청산사유 = 2;
        }
        // 만기 잔량 개별 수익률0% 미만은 삼선/가격비율 보류보다 우선.
        Else If 분할청산 == 1 && 일차처리 && 일차대기 == False && 만기요일 &&
                자기잔량수익률 < 0 Then
        {
            청산요청 = True;
            청산사유 = 12;
        }
        // 만기 잔량: 1차 후 기본15분 + 가격비율100미만 관찰 후100복귀 + 개별60% 이상.

        Else If 분할청산 == 1 && 일차처리 && 일차대기 == False && 만기요일 &&
                일차완료선물봉 >= 0 && 일차완료일 == sDate &&
                잔량경과분 >= Max(0, 잔량익절대기분) && 가격비율하락확인 &&
                잔량비율유효 && 이탈_두구간가격비율 == 100 &&
                만기잔량익절률 > 0 && 자기잔량수익률 >= 만기잔량익절률 Then
        {
            청산요청 = True;
            청산사유 = 11;
        }
        // 만기일 가격비율100 미만이면 잔량 보호청산 보류. 100 이상이면 허용. 자료 불명은 보류.
        // 다른 요일은 기존 보호청산. 손절/시간/날짜 청산에는 영향 없음.
        Else If 분할청산 == 1 && 일차처리 && 일차대기 == False &&
                ((만기요일 == False && 합산수익률 < 익절률) ||
                 (만기요일 && 자기잔량수익률 < 익절률 && 만기잔량삼선허용 &&
                  잔량비율유효 && 이탈_두구간가격비율 >= 100)) Then
        {
            청산요청 = True;
            청산사유 = 10;
        }
        // 만기 당일은 일반 익절 대신 수익률 1.5배 초과 + Data2 가격비율 100.
        Else If 저수익대기 == False && 일차처리 == False && 만기요일 && 합산수익률 > 익절률 * 1.5 &&
                Data2(sDate) == sDate && Data2(sTime) == sTime &&
                Data2(DataCompress) == 2 && Data2(BarInterval) == 1 &&
                함수계산가능 == 1 && 이탈_삼선준비 == 1 &&
                이탈_두구간최대가격범위 > 0 && 이탈_두구간가격비율 == 100 Then
        {
            청산요청 = True;
            청산사유 = 9;
        }
        Else If 저수익대기 == False && 일차처리 == False && 만기요일 == False && 합산수익률 >= 익절률 Then
        {
            청산요청 = True;
            청산사유 = 1;
        }

    }
    // 기존 청산 및 기존 1차 주문이 우선. 60분 이후 <=5%가 된 최초 한 번만 처리.
    // 기존 익절1차 상태와 독립: 이 청산으로 일차처리를 켜지 않음.
    If 청산요청 == False && 일차대기 == False && 저수익처리 == False &&
       모델유효 && 동시봉 && 진입선물봉 >= 0 && 진입선물일 == sDate &&
       Data2(sDate) == sDate && Data2(sTime) == sTime &&
       Data2(DataCompress) == 2 && Data2(BarInterval) == 1 &&
       진입경과분 >= Max(0, 저수익대기분) && 합산수익률 <= 저수익기준률 Then
    {
        저수익처리 = True;
        저수익전수량 = IntPortion(CurrentContracts);
        저수익수량 = IntPortion(저수익전수량 * Min(100, Max(0, 저수익청산비율)) / 100);
        // 2계약 이상이고 비율>0이면 최소1계약, 최소1계약 잔량 유지. 1계약은 보유.
        If 저수익전수량 >= 2 && 저수익청산비율 > 0 Then
            저수익수량 = Min(저수익전수량 - 1, Max(1, 저수익수량));
        Else 저수익수량 = 0;
        저수익대기 = 저수익수량 > 0;
    }
    If 청산요청 Then 저수익대기 = False;
    // 청산 신호 직전 값. 같은 봉/사유/수량은 중복 출력하지 않음.
    If 진단출력사용 == 1 && 저수익대기 && 청산요청 == False &&
       (청산진단봉 != Index || 청산진단사유 != 13 || 청산진단수량 != 저수익수량) Then
    {
        청산진단봉 = Index;
        청산진단사유 = 13;
        청산진단수량 = 저수익수량;
        MessageLog("WSF V34 EXIT_SIGNAL date=%.0f time=%.0f reason=%.0f qty=%.0f pairPct=%.4f ownPct=%.4f", sDate, sTime, 13, 저수익수량, 합산수익률, 자기잔량수익률);
        MessageLog("WSF V34 EXIT_CONTEXT priceRatio=%.4f triRatio=%.4f entryBars=%.0f firstBars=%.0f sync=%s model=%s", 이탈_두구간가격비율, 삼선_지속목표차_비율, 진입경과분, 잔량경과분, 동시봉, 모델유효);
    }
    If 저수익대기 && 청산요청 == False Then
        ExitLong("시간저수익부분청산", AtMarket, Def, "페어진입", 저수익수량, 1);

    // 익절만 분할 전환. 손절/시간/날짜 청산은 잔량 전부 우선 청산.
    If 청산요청 && 분할청산 == 1 && (청산사유 == 1 || 청산사유 == 9) Then
    {
        일차처리 = True;
        일차기준수익률 = 합산수익률;
        일차전수량 = IntPortion(CurrentContracts);
        If 일차전수량 == IntPortion(일차전수량 / 2) * 2 Then
            일차수량 = IntPortion(일차전수량 * 0.5);
        Else
            일차수량 = IntPortion(일차전수량 * 0.75);
        일차사유 = 청산사유;
        일차대기 = 일차수량 > 0;
        청산요청 = False;
        청산사유 = 0;
    }
    If 청산요청 Then 일차대기 = False;
    // 청산 신호 직전 값. 같은 봉/사유/수량은 중복 출력하지 않음.
    If 진단출력사용 == 1 && 일차대기 && 청산요청 == False &&
       (청산진단봉 != Index || 청산진단사유 != 일차사유 || 청산진단수량 != 일차수량) Then
    {
        청산진단봉 = Index;
        청산진단사유 = 일차사유;
        청산진단수량 = 일차수량;
        MessageLog("WSF V34 EXIT_SIGNAL date=%.0f time=%.0f reason=%.0f qty=%.0f pairPct=%.4f ownPct=%.4f", sDate, sTime, 일차사유, 일차수량, 합산수익률, 자기잔량수익률);
        MessageLog("WSF V34 EXIT_CONTEXT priceRatio=%.4f triRatio=%.4f entryBars=%.0f firstBars=%.0f sync=%s model=%s", 이탈_두구간가격비율, 삼선_지속목표차_비율, 진입경과분, 잔량경과분, 동시봉, 모델유효);
    }
    If 일차대기 && 청산요청 == False Then
    {
        // 수량옵션 1: 전체 보유 중 지정 수량, 반복청산 옵션 2는 사용하지 않음.
        If 일차사유 == 9 Then
            ExitLong("만기익절1차", AtMarket, Def, "페어진입", 일차수량, 1);
        Else
            ExitLong("합산익절1차", AtMarket, Def, "페어진입", 일차수량, 1);
    }
    // 청산 신호 직전 값. 같은 봉/사유/수량은 중복 출력하지 않음.
    If 진단출력사용 == 1 && 청산요청 &&
       (청산진단봉 != Index || 청산진단사유 != 청산사유 || 청산진단수량 != CurrentContracts) Then
    {
        청산진단봉 = Index;
        청산진단사유 = 청산사유;
        청산진단수량 = CurrentContracts;
        MessageLog("WSF V34 EXIT_SIGNAL date=%.0f time=%.0f reason=%.0f qty=%.0f pairPct=%.4f ownPct=%.4f", sDate, sTime, 청산사유, CurrentContracts, 합산수익률, 자기잔량수익률);
        MessageLog("WSF V34 EXIT_CONTEXT priceRatio=%.4f triRatio=%.4f entryBars=%.0f firstBars=%.0f sync=%s model=%s", 이탈_두구간가격비율, 삼선_지속목표차_비율, 진입경과분, 잔량경과분, 동시봉, 모델유효);
    }
    // 전량 청산. 요청 후에도 전략 포지션이 남아 있으면 재평가.
    If 청산요청 Then
    {
        If 청산사유 == 1 Then ExitLong("합산익절", AtMarket);
        Else If 청산사유 == 2 Then ExitLong("합산손절", AtMarket);
        Else If 청산사유 == 3 Then ExitLong("시간청산", AtMarket);
        Else If 청산사유 == 4 Then ExitLong("날짜변경청산", AtMarket);
        Else If 청산사유 == 9 Then ExitLong("만기익절", AtMarket);
        Else If 청산사유 == 10 Then ExitLong("잔량익절보호", AtMarket);
        Else If 청산사유 == 11 Then ExitLong("만기잔량익절", AtMarket);
        Else If 청산사유 == 12 Then ExitLong("만기잔량손실청산", AtMarket);
        Else ExitLong("기타청산", AtMarket);
    }
}

// 갭 지연 제거. Data2 삼선 계산이 준비된 경우만 첫 만남 허용.
// 초기값 0을 유효 비율로 오인하지 않도록 목표와 비율 분모를 확인.
// Data2 주간장 첫 봉=0. 08:45 시작 기준 15=09:00, 195=12:00.
// 개장 지연 시 당일 봉 수를 따라 이동. 오전70/오후50 미만, 기존 첫 만남 유지.
If Data2(DayIndex) < Max(0, 오후전환봉수) Then 적용삼선만남비율 = 오전삼선만남최대비율;
Else 적용삼선만남비율 = 오후삼선만남최대비율;
삼선만남허용 = Data2(sDate) == sDate &&
               Data2(DataCompress) == 2 && Data2(BarInterval) == 1 &&
               삼선_목표준비 == 1 && 삼선_지속목표차_최고 > 0 &&
               삼선_지속목표차_비율 < 적용삼선만남비율;
// 금요일 오전 첫 만남: 유효한 동시각 Data2 가격비율이 100 미만일 때만 허용.
If DayOfWeek(Bdate) == 5 && Data2(DayIndex) < Max(0, 오후전환봉수) Then
    삼선만남허용 = 삼선만남허용 && Data2(sTime) == sTime &&
                   함수계산가능 == 1 && 이탈_삼선준비 == 1 &&
                   이탈_두구간최대가격범위 > 0 && 이탈_두구간가격비율 < 100;

// 완료봉 조건으로 다음봉 시가 신호. AtMarket은 신호타입이며
// 실주문 시장가 여부는 HTS 자동주문 설정과 별개입니다.
만남 = False;
If 주기정상 && 동시봉 && 삼선만남허용 && Data2(DayIndex) >= Max(0, 진입시작봉수) &&
   sTime < 전량청산 && NextBarSdate == sDate &&
   NextBarStime < 전량청산 && 총투자금 > 0 && 익절률 > 0 && 손절률 > 0 Then
{
    If 만남방식 == 1 Then
        만남 = Max(L, Data3(L)) <= Min(H, Data3(H)) + Max(0,허용오차);
    If 만남방식 == 2 Then
        만남 = C - Max(0,허용오차) <= Data3(C) &&
               C + Max(0,허용오차) >= Data3(C);
}
// 매매종료=1은 신규 진입만 금지. 기존 보유 청산은 위에서 계속 처리.
If 매매종료 == 0 && MarketPosition == 0 && 당일사용 == False &&
   진입대기 == False && 청산요청 == False && 만남 Then
{
    // 첫 만남에서 예산 부족이어도 그날 재시도하지 않음.
    당일사용 = True;
    // 첫 유효 만남 봉의 양쪽 종가를 별도로 저장. 재만남으로 덮어쓰지 않음.
    자기첫만남가격 = C;
    상대첫만남가격 = Data3(C);
    자기예상 = C;
    상대예상 = Data3(C);
    자기수량 = IntPortion((총투자금 / 2) / (자기예상 * 250000));
    상대수량 = IntPortion((총투자금 / 2) / (상대예상 * 250000));
    If 자기수량 >= 1 && 상대수량 >= 1 Then
    {
        신호봉 = Index;
        신호일 = Bdate;
        진입대기 = True;
        Buy("페어진입", AtMarket, Def, 자기수량);
    }
}

If 진단출력사용 == 1 && Index != 진단봉 Then

    If MarketPosition == 1 || 청산요청 Then

        MessageLog("WSF V34 ownPct=%.4f target=%.2f priceRatio=%.4f triRatio=%.4f", 자기잔량수익률, 만기잔량익절률, 이탈_두구간가격비율, 삼선_지속목표차_비율);
        MessageLog("WSF V34 elapsed=%.0f wait=%.0f below100=%s startD2=%.0f", 잔량경과분, 잔량익절대기분, 가격비율하락확인, 일차완료선물봉);
        MessageLog("WSF V34 first=%s pending=%s qty=%.0f remaining=%.0f", 일차처리, 일차대기, 일차수량, CurrentContracts);
        MessageLog("WSF V34 date=%.0f time=%.0f model=%s sync=%s exit=%s reason=%.0f pnlPct=%.4f",
                   sDate, sTime, 모델유효, 동시봉, 청산요청, 청산사유, 합산수익률);
    }
