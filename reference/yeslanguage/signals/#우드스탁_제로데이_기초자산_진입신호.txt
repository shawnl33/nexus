            Else If 조정신호종류 == 1 && 조정신호등급 == 2 Then {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+진입계약수; Else 주문코드 = 22*10+진입계약수;}
            Else If 조정신호종류 == 1 && 조정신호등급 == 1 Then {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+진입계약수; Else 주문코드 = 22*10+진입계약수;}
            Else If 조정신호종류 == 2 && 조정신호등급 == 3 Then {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+진입계약수; Else 주문코드 = 22*10+진입계약수;}
            Else If 조정신호종류 == 2 && 조정신호등급 == 2 Then {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+진입계약수; Else 주문코드 = 22*10+진입계약수;}
            Else If 조정신호종류 == 2 && 조정신호등급 == 1 Then {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+조정약계약수; Else 주문코드 = 22*10+진입계약수;}
            Else {If 장약세진입 == 1 Then 주문코드 = 24*10+장약세계약수; Else If 반대방향진입 == 1 Then 주문코드 = 23*10+진입계약수; Else 주문코드 = 22*10+진입계약수;}
        };
        가상매도진입 = 1;
    };

    // (짝V68) 세션 첫 진입(세션진입수 == 0)이 01:45(자정+첫진입늦은금지분) ~ 06:59 봉이면 돌파매수(주문번호 1)·압축매도_반대(주문번호 11) 금지. 0 이면 끔
    If 첫진입늦은금지분 > 0 && 주문코드 > 0 && 세션진입수 == 0 &&
       본장기준봉분 >= 1440+첫진입늦은금지분 && 본장기준봉분 < 1860 &&
       (IntPortion(주문코드/10) == 1 || IntPortion(주문코드/10) == 11) Then
       {주문코드 = 0; 가상매수진입 = 0; 가상매도진입 = 0;}

    // (짝V54) 진입 주문 실행 - 예스랭귀지는 같은 주문 이름을 두 번 선언할 수 없어서 진입명마다 한 줄씩만 둠
    If 주문코드 > 0 Then
    {
        주문번호 = IntPortion(주문코드/10);
        주문수량 = 주문코드-주문번호*10;
        If 주문번호 == 1 Then Buy("돌파매수",OnClose,Def,주문수량);
        If 주문번호 == 2 Then Buy("돌파매수_반대",OnClose,Def,주문수량);
        If 주문번호 == 3 Then Buy("돌파매수_약세",OnClose,Def,주문수량);
        If 주문번호 == 4 Then Sell("돌파매도",OnClose,Def,주문수량);
        If 주문번호 == 5 Then Sell("돌파매도_반대",OnClose,Def,주문수량);
        If 주문번호 == 6 Then Sell("돌파매도_약세",OnClose,Def,주문수량);
        If 주문번호 == 7 Then Buy("압축매수",OnClose,Def,주문수량);
        If 주문번호 == 8 Then Buy("압축매수_반대",OnClose,Def,주문수량);
        If 주문번호 == 9 Then Buy("압축매수_약세",OnClose,Def,주문수량);
        If 주문번호 == 10 Then Sell("압축매도",OnClose,Def,주문수량);
        If 주문번호 == 11 Then Sell("압축매도_반대",OnClose,Def,주문수량);
        If 주문번호 == 12 Then Sell("압축매도_약세",OnClose,Def,주문수량);
        If 주문번호 == 13 Then Buy("반전매수",OnClose,Def,주문수량);
        If 주문번호 == 14 Then Buy("반전매수_반대",OnClose,Def,주문수량);
        If 주문번호 == 15 Then Buy("반전매수_약세",OnClose,Def,주문수량);
        If 주문번호 == 16 Then Sell("반전매도",OnClose,Def,주문수량);
        If 주문번호 == 17 Then Sell("반전매도_반대",OnClose,Def,주문수량);
        If 주문번호 == 18 Then Sell("반전매도_약세",OnClose,Def,주문수량);
        If 주문번호 == 19 Then Buy("조정매수",OnClose,Def,주문수량);
        If 주문번호 == 20 Then Buy("조정매수_반대",OnClose,Def,주문수량);
        If 주문번호 == 21 Then Buy("조정매수_약세",OnClose,Def,주문수량);
        If 주문번호 == 22 Then Sell("조정매도",OnClose,Def,주문수량);
        If 주문번호 == 23 Then Sell("조정매도_반대",OnClose,Def,주문수량);
        If 주문번호 == 24 Then Sell("조정매도_약세",OnClose,Def,주문수량);
    };

    // (짝V2) 가상 진입 기록
    If 가상매수진입 == 1 || 가상매도진입 == 1 Then
    {
        세션진입수 = 세션진입수+1;
        마지막진입방향 = IFF(가상매수진입 == 1,1,-1);
        가상포지션 = IFF(가상매수진입 == 1,1,-1);
        가상진입가 = C; 가상최고가 = C; 가상최저가 = C;
        // (짝V29) 1차 목표틱 결정 (진입 순간 고정)
        현재1차틱 = 일차청산틱;
        If 유동일차 == 1 Then
        {
            통합진행 = 통합상태*가상포지션;
            If 통합진행 >= 100 Then 현재1차틱 = 일차강틱;
            Else If 통합진행 >= 50 Then 현재1차틱 = 일차중틱;
            기울기진행 = 선행기울기Cnt*가상포지션;
            If 기울기진행 > 0 && 기울기진행 <= 기울기초기봉 Then 현재1차틱 = 현재1차틱+기울기초기보너스틱;
            If 기울기진행 <= 0 || (기울기후기봉 > 0 && 기울기진행 > 기울기후기봉) Then 현재1차틱 = 일차청산틱;
        };
        // (짝V42) 본장 시작 후 늦은1차분(150)분 이후 진입은 1차 목표 늦은1차틱(10)틱 (진입 순간 고정). 늦은1차분 = 0 이면 끔
        If 늦은1차분 > 0 && 늦은1차틱 > 0 && 본장기준봉분 >= 본장시작분+늦은1차분 Then 현재1차틱 = 늦은1차틱;
        // (짝V55) 직전 거래가 수익이고 그 청산 후 직전수익1차봉수(60)봉 이내 진입이면 1차 목표 직전수익1차틱(10)틱. 직전수익1차틱 = 0 이면 끔
        //   (짝V56) + 진입 봉 삼선비율(S삼선비율, 스나이퍼스코프 삼선_지속목표차_비율) >= 직전수익1차삼선비율(90)% 일 때만
        //   (짝V57) 직전 거래 손익과 상관없이 포지션 정리(최종 청산)가 직전수익1차봉수(60)봉 안에 있었으면 적용. 직전수익1차수익필요 = 1 이면 수익일 때만(V56과 같음)
        //   (짝V59) 진입 순간에는 '정리 후 60봉 이내 진입' 여부만 기억 (삼선비율은 보유 중 매 봉 확인)
        직전정리진입 = 0;
        가상진입봉 = CurrentBar;
        수익전환수 = 0; 직전봉평가손익 = 0;
        If 직전수익1차틱 > 0 && (직전수익1차수익필요 == 0 || 직전거래손익 > 0) && CurrentBar-직전청산봉 <= 직전수익1차봉수 Then 직전정리진입 = 1;
    };

    // (짝V59) 정리 후 60봉 이내 진입한 포지션은 보유 중 어느 봉에서든 삼선비율 >= 직전수익1차삼선비율(80) 이 되면 그때부터 1차 목표 직전수익1차틱(10)틱
    //   (짝V62) + 진입 봉 이후 직전수익1차경과봉(4)봉이 지난 뒤부터 (진입 후 4분 경과)
    If 가상포지션 != 0 && 직전정리진입 == 1 && 직전수익1차틱 > 0 && S삼선비율 >= 직전수익1차삼선비율 &&
       CurrentBar-가상진입봉 >= 직전수익1차경과봉 &&
       (현재1차틱 <= 0 || 현재1차틱 > 직전수익1차틱) Then 현재1차틱 = 직전수익1차틱;

    // (짝V63) 진입 후 종가 기준 평가손익이 0 이하에서 + 로 바뀐 횟수(수익전환수)가 수익전환기준(3)회 이상이면 1차 목표 수익전환1차틱(10)틱. 수익전환기준 = 0 이면 끔
    If 가상포지션 != 0 Then
    {
        보유평가손익 = (C-가상진입가)*가상포지션;
        If CurrentBar > 가상진입봉 && 보유평가손익 > 0 && 직전봉평가손익 <= 0 Then 수익전환수 = 수익전환수+1;
        직전봉평가손익 = 보유평가손익;
        If 수익전환기준 > 0 && 수익전환1차틱 > 0 && 수익전환수 >= 수익전환기준 &&
           (현재1차틱 <= 0 || 현재1차틱 > 수익전환1차틱) Then 현재1차틱 = 수익전환1차틱;
    };

    // (짝V3) 1차: 진입가 ± 일차청산틱(10) 지정가 1계약. 진입한 봉에서도 같이 걸어 둠
    // (짝V5) 2차 트레일링 스탑 삭제 -> 남은 계약은 색 전환·시간·반대 신호 청산
    If 가상포지션 == 1 && (MarketPosition == 1 || 가상매수진입 == 1) Then
    {
        If 현재1차틱 > 0 Then ExitLong("1차청산매수",AtLimit,가상진입가+현재1차틱*PriceScale,"",1,1);
    };
    If 가상포지션 == -1 && (MarketPosition == -1 || 가상매도진입 == 1) Then
    {
        If 현재1차틱 > 0 Then ExitShort("1차청산매도",AtLimit,가상진입가-현재1차틱*PriceScale,"",1,1);
    };
};
