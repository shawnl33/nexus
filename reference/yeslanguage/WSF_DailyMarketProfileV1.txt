
Input : 계산기간입력(Numeric),
        밴드배수입력(Numeric),
        중심가격(NumericRef),
        상단가격(NumericRef),
        하단가격(NumericRef),
        계산유효(NumericRef);

Var : 계산기간(0),
      밴드배수(0),
      기준가격(0),
      누적거래량(0),
      누적가격거래량(0),
      가중분산합(0),
      가중표준편차(0),
      j(0);

계산기간 = Abs(계산기간입력);
If 계산기간 < 2 Then
    계산기간 = 2;

밴드배수 = Abs(밴드배수입력);
If 밴드배수 <= 0 Then
    밴드배수 = 1;

중심가격 = 0;
상단가격 = 0;
하단가격 = 0;
계산유효 = 0;

기준가격 = (H + L + C) / 3;
누적거래량 = 0;
누적가격거래량 = 0;

For j = 0 To 계산기간 - 1
{
    If V[j] > 0 Then
    {
        누적거래량 = 누적거래량 + V[j];
        누적가격거래량 = 누적가격거래량 + 기준가격[j] * V[j];
    };
};

If 누적거래량 > 0 Then
{
    중심가격 = 누적가격거래량 / 누적거래량;
    가중분산합 = 0;

    For j = 0 To 계산기간 - 1
    {
        If V[j] > 0 Then
            가중분산합 = 가중분산합
                         + (기준가격[j] - 중심가격)^2 * V[j];
    };

    가중표준편차 = SquareRoot(가중분산합 / 누적거래량);
    상단가격 = 중심가격 + 가중표준편차 * 밴드배수;
    하단가격 = 중심가격 - 가중표준편차 * 밴드배수;
    계산유효 = 1;
};

Value1 = 계산유효;
WSF_DailyMarketProfileV1 = Value1;
