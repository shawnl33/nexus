// 표시실험3. 쉬는 봉의 앞뒤 값을 다르게 둔다.
// 맨 앞 40봉만 보이게 하고, 범례가 나오게 전체 창을 캡처한다.

Var : n(0);
n = CurrentBar;

// 1~10봉은 80 파랑, 11~20봉은 값 없음, 21~30봉은 40 빨강.
// 같은 Plot 번호는 이름을 한 번만 쓸 수 있다.
If n <= 10 Then
{
    Plot1(80,"공백실험",RGB(0,0,255),Def,2);
}
Else If n > 20 && n <= 30 Then
{
    Plot1(40,"공백실험",RGB(255,0,0),Def,2);
};
