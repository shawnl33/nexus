// 최근 40봉이 보이게 축소하고, 범례 숫자가 나오게 전체 창을 캡처해 주세요.
// 가격 지표는 끄셔도 됩니다. 이 지표만 있으면 됩니다.

Var : n(0);
n = CurrentBar;

// 질문 1. 값이 100이다가 130으로 바뀔 때 선이 사선인지, 가로 다음 수직인지.
// 질문 2. 색과 굵기가 바뀌는 구간은 어느 색으로 칠해지는지.
If n <= 15 Then
{
    Plot1(100,"유지후점프",RGB(255,128,0),Def,2);
}
Else If n <= 30 Then
{
    Plot1(130,"유지후점프",RGB(255,0,0),Def,3);
};

// 질문 3. 값을 안 주는 봉에서 이전 선이 끊기는지, 오른쪽까지 남는지.
If n <= 10 || (n > 20 && n <= 30) Then
{
    Plot2(Close,"열봉그리고열봉쉼",RGB(0,0,255),Def,2);
};

// 질문 4. 103, 50, -15가 눈금 어디에 놓이는지. 0과 100으로 잘리는지.
Plot3(103,"위103",RGB(0,0,255),Def,1);
Plot4(50,"가운데50",RGB(128,128,128),Def,1);
Plot5(-15,"아래-15",RGB(255,0,0),Def,1);
