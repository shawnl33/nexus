// 스나이퍼 Plot29 압축신호점과 같은 색이다.
// 위 신호 파랑 RGB(0,0,255), 아래 신호 빨강 RGB(255,0,0), 둘 다 초록 RGB(0,128,0).
// 굵기 3. 원본 주석대로 이 플롯은 점 그래프다.
// 비율은 0~100. 30 미만인 봉만 점을 찍는다.

Var : n(0), 칸(0), 비율(0);
n = CurrentBar;
칸 = 칸 + 1;
If 칸 > 3 Then 칸 = 1;
비율 = 12 + 칸 * 5;

If n <= 40 && 비율 < 30 Then
{
    If 칸 == 1 Then
        Plot1(비율,"삼선_압축신호점",RGB(0,0,255),Def,3);
    Else If 칸 == 2 Then
        Plot1(비율,"삼선_압축신호점",RGB(255,0,0),Def,3);
    Else
        Plot1(비율,"삼선_압축신호점",RGB(0,128,0),Def,3);
};
