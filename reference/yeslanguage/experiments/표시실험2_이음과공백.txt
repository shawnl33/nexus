// 표시실험2. 표시실험1의 Close를 뺐다. 눈금이 100과 130만 보게 한다.
// 아무 1분봉의 맨 앞 40봉만 보이게 하고, 범례가 나오게 전체 창을 캡처한다.

Var : n(0);
n = CurrentBar;

// 15봉은 100 주황 굵기 2, 다음 15봉은 130 빨강 굵기 3.
If n <= 15 Then
{
    Plot1(100,"유지후점프",RGB(255,128,0),Def,2);
}
Else If n <= 30 Then
{
    Plot1(130,"유지후점프",RGB(255,0,0),Def,3);
};

// 1~10봉, 21~30봉만 80. 11~20봉은 값을 주지 않는다.
If n <= 10 || (n > 20 && n <= 30) Then
{
    Plot2(80,"열봉그리고열봉쉼",RGB(0,0,255),Def,2);
};
