// 표시실험6. 십자는 "+" , 원은 "o" 다.
// 눈금이 그 높이를 포함해야 글자가 보인다. 선으로 눈금을 만든다.
// 글자 번호는 봉마다 다른 칸에 둔다. 한 변수에 다시 넣으면 이전 글자가 지워질 수 있다.
// 맨 앞 40봉만 보이게 하고 전체 창을 캡처한다.

Array : 글자번호[60](0);
Var : n(0);
n = CurrentBar;

Plot1(80,"십자선",RGB(255,128,0),Def,1);
Plot2(60,"원선",RGB(0,128,0),Def,1);

If n >= 1 && n <= 30 Then
{
    글자번호[n] = Text_New(sDate, sTime, 80, "+");
    글자번호[n + 30] = Text_New(sDate, sTime, 60, "o");
};
