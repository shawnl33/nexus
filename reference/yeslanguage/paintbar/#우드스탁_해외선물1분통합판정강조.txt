            If 선행직전하락기울기[전환j] > 전환직전상승제한 Then 전환직전하락초과 = 1;
        };
        If 통합상태 > 0 || 압축지속봉수 >= 전환압축봉수제한 || 전환직전하락초과 == 1 Then 전환매도 = 0;
    };
    // (강조V34) 삼선점수색상이 회색이면 전환매수·전환매도 금지 (전환삼선회색금지 = 0 이면 끔)
    If 전환삼선회색금지 == 1 && S삼선회색 == 1 Then
    {
        전환매도 = 0;  // (강조V35) 전환매수는 해제, 전환매도만
    };
    // (강조V33) ① 압축 색 필터: 최근 10봉 압축 막대 중 반대색이 과반이면 금지
    //   전환매수: 파랑(박스 하단) 과반이면 금지 / 전환매도: 빨강(박스 상단) 과반이면 금지 (전환압축색봉수 = 0 이면 끔)
    // (강조V35) 전환매수 압축 색 필터 해제 (전환매도만 적용)
    If 전환압축색봉수 > 0 && 전환매도 == 1 && 전환압축빨강수*2 > 전환압축색봉수 Then 전환매도 = 0;
    // (강조V33) ② 반대 전환 직후 금지: 반대 방향 전환 신호 뒤 전환반대금지봉수(10)봉 안이면 금지 (0 이면 끔)
    // (강조V35) 전환매수 반대 전환 직후 금지 해제 (전환매도만 적용, 전환매수경과는 전환매도 판정용으로 계속 셈)
    If 전환반대금지봉수 > 0 && 전환매도 == 1 && 전환매수경과 <= 전환반대금지봉수 Then 전환매도 = 0;
    // (강조V36) 전환매도는 통합상태 == -50 일 때만 허용 (-100은 늦은 매도, 0 이상은 역행. 전환매도통합기준 = 0 이면 끔)
    If 전환매도통합기준 == 1 && 전환매도 == 1 && 통합상태 != -50 Then 전환매도 = 0;
    // (강조V40) 전환매도: 최근 전환매도직전100봉수(10)봉 안에 통합상태 -100이 있었으면 삭제 (하락이 끝나고 약해지는 -50 = 늦은 매도). 0 이면 끔
    If 전환매도직전100봉수 > 0 && 전환매도 == 1 Then
    {
        For 전환k = 1 To 전환매도직전100봉수
        {
            If 통합상태[전환k] == -100 Then 전환매도 = 0;
        };
    };
    // (강조V45) 전환매수: 최근 전환매수직전100봉수(10)봉 안에 통합상태 +100 또는 -100 이 있었으면 삭제. 0 이면 끔
    //   +100 = 상승이 이미 나온 뒤 늦은 매수(03/03 23:37), -100 = 하락이 아직 살아 있는 이른 매수(06/06 00:06)
    If 전환매수직전100봉수 > 0 && 전환매수 == 1 Then
    {
        For 전환k = 1 To 전환매수직전100봉수
        {
            If 통합상태[전환k] == 100 || 통합상태[전환k] == -100 Then 전환매수 = 0;
        };
    };
    If 전환매수 == 1 Then 전환매수경과 = 0;
    If 전환매도 == 1 Then 전환매도경과 = 0;
    신호있음 = 0;
    신호색 = 통합색상;
    If 조정표시 == 1 && 조정신호방향 == 1 Then {신호색 = RGB(255,140,0); 신호있음 = 1;}
    Else If 조정표시 == 1 && 조정신호방향 == -1 Then {신호색 = RGB(150,0,220); 신호있음 = 1;}
    If 전환매수 == 1 Then {신호색 = RGB(0,100,0); 신호있음 = 1;}  // (강조V29) 기울기전환매수 진녹색
    Else If 전환매도 == 1 Then {신호색 = RGB(150,0,80); 신호있음 = 1;}  // (강조V32) 기울기전환매도 진자주
    If 선행방향 == 1 Then {신호색 = RGB(150,75,0); 신호있음 = 1;}
    Else If 선행방향 == -1 Then {신호색 = RGB(0,0,0); 신호있음 = 1;}
    If 구조돌파방향 == 1 Then {신호색 = RGB(0,200,0); 신호있음 = 1;}
    Else If 구조돌파방향 == -1 Then {신호색 = RGB(255,0,255); 신호있음 = 1;}
    If 후보방향 == 1 Then {신호색 = RGB(0,255,0); 신호있음 = 1;}
    Else If 후보방향 == -1 Then {신호색 = Cyan; 신호있음 = 1;}

    // 평소 봉 = 통합상태 색·평소굵기, 신호 봉 = 신호색·신호굵기
    PlotPaintBar(O,C,"당일통합상태",IFF(신호있음 == 1,신호색,통합색상),Def,IFF(신호있음 == 1,신호굵기,평소굵기));
};
