
Input : ADX기간입력(Numeric),
        ADX출력(NumericRef),
        상승DI출력(NumericRef),
        하락DI출력(NumericRef),
        ADX유효출력(NumericRef);

Var : 일분봉여부(0),
      새계산시작(0),
      기간값(0),
      계산봉수(0),
      상승이동(0),
      하락이동(0),
      상승DM(0),
      하락DM(0),
      실제범위(0),
      평활TR(0),
      평활상승DM(0),
      평활하락DM(0),
      계산상승DI(0),
      계산하락DI(0),
      DX값(0),
      DX누적(0),
      계산ADX(0),
      계산유효(0);

/* 1분봉은 새 장마다 초기화하고, 일봉은 연속 계산한다. */
일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

기간값 = Round(Abs(ADX기간입력), 0);
If 기간값 < 2 Then
    기간값 = 2;

ADX출력 = 0;
상승DI출력 = 0;
하락DI출력 = 0;
ADX유효출력 = 0;

새계산시작 = 0;

If CurrentBar == 1 Then
    새계산시작 = 1;

If CurrentBar > 1 && 일분봉여부 == 1 && BDate <> BDate[1] Then
    새계산시작 = 1;

If 새계산시작 == 1 Then
{
    계산봉수 = 1;
    상승DM = 0;
    하락DM = 0;
    실제범위 = H - L;
    평활TR = 실제범위;
    평활상승DM = 0;
    평활하락DM = 0;
    계산상승DI = 0;
    계산하락DI = 0;
    DX값 = 0;
    DX누적 = 0;
    계산ADX = 0;
    계산유효 = 0;
}
Else
{
    계산봉수 = 계산봉수[1] + 1;

    상승이동 = H - H[1];
    하락이동 = L[1] - L;

    상승DM = IFF(상승이동 > 하락이동 && 상승이동 > 0,
                 상승이동, 0);

    하락DM = IFF(하락이동 > 상승이동 && 하락이동 > 0,
                 하락이동, 0);

    실제범위 = Max(H - L,
                   Max(Abs(H - C[1]), Abs(L - C[1])));

    If 계산봉수 <= 기간값 Then
    {
        평활TR = 평활TR[1] + 실제범위;
        평활상승DM = 평활상승DM[1] + 상승DM;
        평활하락DM = 평활하락DM[1] + 하락DM;
    }
    Else
    {
        평활TR = 평활TR[1] - 평활TR[1] / 기간값 + 실제범위;
        평활상승DM = 평활상승DM[1] - 평활상승DM[1] / 기간값 + 상승DM;
        평활하락DM = 평활하락DM[1] - 평활하락DM[1] / 기간값 + 하락DM;
    };

    If 평활TR > 0 Then
    {
        계산상승DI = 평활상승DM / 평활TR * 100;
        계산하락DI = 평활하락DM / 평활TR * 100;
    }
    Else
    {
        계산상승DI = 0;
        계산하락DI = 0;
    };

    If 계산상승DI + 계산하락DI > 0 Then
        DX값 = Abs(계산상승DI - 계산하락DI)
               / (계산상승DI + 계산하락DI) * 100;
    Else
        DX값 = 0;

    If 계산봉수 > 기간값 && 계산봉수 < 기간값 * 2 Then
    {
        DX누적 = DX누적[1] + DX값;
        계산ADX = 0;
        계산유효 = 0;
    }
    Else If 계산봉수 == 기간값 * 2 Then
    {
        DX누적 = DX누적[1] + DX값;
        계산ADX = DX누적 / 기간값;
        계산유효 = 1;
    }
    Else If 계산봉수 > 기간값 * 2 Then
    {
        DX누적 = DX누적[1];
        계산ADX = (계산ADX[1] * (기간값 - 1) + DX값) / 기간값;
        계산유효 = 1;
    };
};

If 계산유효 == 1 Then
{
    ADX출력 = 계산ADX;
    상승DI출력 = 계산상승DI;
    하락DI출력 = 계산하락DI;
    ADX유효출력 = 1;
};

Value1 = ADX유효출력;
WSF_AutoSessionADXV1 = Value1;
