// #우드스탁_해외선물진입진단V1  (하단 지표, 1분봉 전용)  - 진입후보 V4가 특정 봉에서 왜 후보를 안 냈는지 확인용
// V4와 같은 박스 함수(WSF_FXSqueezeV3_CO)와 통합판정을 계산해 이벤트를 높이별로 표시. 모두 상승=+, 하락=-.
//   Plot1 통합상태(±100/±50)   Plot2 박스돌파 ±80   Plot3 즉시해제 ±60   Plot4 지연확정 ±40
//   Plot5 미래방향 ±20          Plot6 DI방향 ±10     Plot7 압축지속봉수(참고)
// 제목줄 값으로 해당 봉의 각 이벤트 유무를 읽고, 직전 봉 값은 커서를 한 봉 앞에 두고 확인.
Input : 예측변수(10),
        가치영역배수(1.5),
        최소봉수(5),
        ADX기간(14),
        ADX추세기준(20),
        ADX강세기준(35),
        매물대기준모드(1),
        폭계산봉수(30),
        비율기준봉수(120),
        좁음기준(40),
        단계1봉수(15),
        방향확정봉수(5),
        확정봉판정(0),
        돌파여유틱(2);
Var : FX세션키(0), FX초기화(0), FX세션봉수(0),
      곡선하(0), 곡선상(0), 곡선예측가격(0), 곡선예측변화(0), 곡선예측방향(0), 미래방향(0),
      매물대중심(0), 매물대상단1(0), 매물대하단1(0), 매물대상단2(0), 매물대하단2(0),
      매물대5상태(0), 매물대유효(0),
      세션ADX(0), 상승DI(0), 하락DI(0), ADX유효(0),
      통합상태(0), 매물대상태(0), ADX상태(0), DI방향(0),
      압축해제(0), 압축지속봉수(0), 폭비율(0), 즉시해제방향(0), 지연확정방향(0), 해제길이(0),
      박스돌파방향(0), 박스돌파길이(0), 돌파박스상단(0), 돌파박스하단(0);

FX세션키 = WSF_FXSessionKeyV1(sDate,sTime);
FX초기화 = 0;
If CurrentBar == 1 Then FX초기화 = 1;
Else If FX세션키 != FX세션키[1] Then FX초기화 = 1;
If FX초기화 == 1 Then FX세션봉수 = 1;
Else FX세션봉수 = FX세션봉수[1]+1;

If DataCompress == 2 && BarInterval == 1 Then
{
    WSF_FXCurveV1(FX초기화,FX세션봉수,예측변수,
                  곡선하,곡선상,곡선예측가격,곡선예측변화,곡선예측방향);
    미래방향 = IFF(곡선예측방향 > 0, 1, IFF(곡선예측방향 < 0, -1, 0));
    If FX초기화 == 1 Then 미래방향 = 0;
    WSF_FXSessionProfileV1_CO(FX초기화,가치영역배수,최소봉수,
        매물대중심,매물대상단1,매물대하단1,매물대상단2,매물대하단2,
        매물대5상태,매물대유효);
    WSF_FXADXV1(FX초기화,FX세션봉수,ADX기간,세션ADX,상승DI,하락DI,ADX유효);
    WSF_FXDecisionV2_CO(미래방향,매물대5상태,매물대유효,매물대중심,C,매물대기준모드,
        세션ADX,상승DI,하락DI,ADX유효,ADX추세기준,ADX강세기준,
        통합상태,매물대상태,ADX상태,DI방향);
    압축해제 = WSF_FXSqueezeV3_CO(FX초기화,FX세션봉수,가치영역배수,최소봉수,
        폭계산봉수,비율기준봉수,좁음기준,단계1봉수,방향확정봉수,확정봉판정,돌파여유틱,
        압축지속봉수,폭비율,즉시해제방향,지연확정방향,해제길이,
        박스돌파방향,박스돌파길이,돌파박스상단,돌파박스하단);

    Plot1(통합상태,"통합상태",IFF(통합상태 > 0,RGB(255,150,150),IFF(통합상태 < 0,RGB(150,170,255),RGB(170,170,170))),Def,1);
    If 박스돌파방향 != 0 Then Plot2(박스돌파방향*80,"박스돌파",IFF(박스돌파방향 > 0,RGB(220,0,0),RGB(0,0,200)),Def,5);
    Else NoPlot(2);
    If 즉시해제방향 != 0 Then Plot3(즉시해제방향*60,"즉시해제",IFF(즉시해제방향 > 0,RGB(255,90,0),RGB(0,120,255)),Def,5);
    Else NoPlot(3);
    If 지연확정방향 != 0 Then Plot4(지연확정방향*40,"지연확정",IFF(지연확정방향 > 0,RGB(255,160,60),RGB(80,170,255)),Def,5);
    Else NoPlot(4);
    Plot5(미래방향*20,"미래방향",RGB(200,0,120),Def,1);
    Plot6(IFF(ADX유효 == 1,DI방향,0)*10,"DI방향",RGB(0,150,150),Def,1);
    Plot7(압축지속봉수,"압축지속봉수",RGB(255,170,80),Def,0);
}
Else
{
    NoPlot(1); NoPlot(2); NoPlot(3); NoPlot(4); NoPlot(5); NoPlot(6); NoPlot(7);
};
