// 지표명: WS_CrossSizeRV_Research_09_DataActivity
// 목적: NQ/MNQ의 봉 거래량과 각 종목의 직전 봉 대비 종가 변화 여부를 관찰합니다.
// Data1: E-mini Nasdaq-100 선물 NQ(주종목)
// Data2: Micro E-mini Nasdaq-100 선물 MNQ(참조종목)
// 설정: 동일 월물, 동일 시간축의 1분봉을 사용하며 NQ가 Data1인 왼쪽 화면에 적용합니다.
// DisplayMode=1: 거래량 두 개를 표시합니다. 단위는 각 종목의 계약 수입니다.
// DisplayMode=2: 종가 변화 상태 두 개를 표시합니다. 1=변화, 0=동일, N/A=비교 불가.
// 거래량과 0/1 상태의 축이 섞이지 않도록 표시 모드를 분리합니다.
// 두 모드를 동시에 보려면 같은 지표를 별도 영역 두 곳에 적용하고 모드를 각각 설정합니다.
// 같은 종가에서도 거래는 발생할 수 있으므로 상태 0은 갱신 중단 판정이 아닙니다.
// 거래량 0도 통신 장애의 증거가 아닙니다. 결측이나 보간 여부는 데이터 제공 방식을 확인해야 합니다.
// 두 종목의 거래량은 승수가 다른 계약 수이며 명목 거래금액으로 환산한 값이 아닙니다.
// 진행 중인 봉의 종가와 거래량은 변합니다. 마지막 체결시각 또는 수신 지연을 측정하지 않습니다.
// 날짜/시간 일치는 05_DataSync로 함께 확인합니다. 주문 코드는 없습니다.

// 1은 거래량 보기, 2는 종가 변화 상태 보기입니다.
input : DisplayMode(1);

// 각 종목의 현재 봉 거래량, 현재 종가, 해당 종목의 직전 봉 종가를 저장합니다.
var : NQVolume(Nan), MNQVolume(Nan), NQClose(Nan), MNQClose(Nan);
var : NQPreviousClose(Nan), MNQPreviousClose(Nan);
// 각 종목의 종가 변화 상태와 화면에 내보낼 네 개의 값을 저장합니다.
var : NQChanged(Nan), MNQChanged(Nan);
var : NQVolumePlot(Nan), MNQVolumePlot(Nan), NQChangedPlot(Nan), MNQChangedPlot(Nan);

// V는 현재 봉 거래량입니다. 참조종목은 data2 내부에서 읽습니다.
NQVolume = V;
MNQVolume = data2(V);
// 각 종목의 현재 종가를 읽습니다.
NQClose = C;
MNQClose = data2(C);

// 이전 값이 남지 않도록 비교 값과 상태를 매 계산 시점에 초기화합니다.
NQPreviousClose = Nan;
MNQPreviousClose = Nan;
NQChanged = Nan;
MNQChanged = Nan;
// 이전 봉을 비교할 수 있는 구간에서 각 종목 자체의 직전 봉 종가를 참조합니다.
if CurrentBar > 1 Then {
    NQPreviousClose = C[1];
    // 대괄호를 data2 안에 두어 MNQ 자체의 직전 봉을 참조합니다.
    // MNQ 봉이 누락되었다면 NQ 직전 봉과 시각이 다를 수 있습니다.
    MNQPreviousClose = data2(C[1]);
    // NQ 현재/직전 종가가 유효하면 동일 여부를 비교합니다.
    if IsNan(NQClose) == False and IsNan(NQPreviousClose) == False Then {
        NQChanged = 1;
        if NQClose == NQPreviousClose Then NQChanged = 0;
    }
    // MNQ 현재/직전 종가가 유효하면 동일 여부를 비교합니다.
    if IsNan(MNQClose) == False and IsNan(MNQPreviousClose) == False Then {
        MNQChanged = 1;
        if MNQClose == MNQPreviousClose Then MNQChanged = 0;
    }
}

// 선택하지 않은 출력은 N/A로 두어 서로 다른 단위가 같은 축에 표시되지 않게 합니다.
NQVolumePlot = Nan;
MNQVolumePlot = Nan;
NQChangedPlot = Nan;
MNQChangedPlot = Nan;
// 거래량 모드에서는 두 종목의 유효한 0 이상 거래량을 그대로 표시합니다.
if DisplayMode == 1 Then {
    if IsNan(NQVolume) == False Then {
        if NQVolume >= 0 Then NQVolumePlot = NQVolume;
    }
    if IsNan(MNQVolume) == False Then {
        if MNQVolume >= 0 Then MNQVolumePlot = MNQVolume;
    }
}
// 상태 모드에서는 종가가 직전 봉과 다른지 각각 0 또는 1로 표시합니다.
if DisplayMode == 2 Then {
    NQChangedPlot = NQChanged;
    MNQChangedPlot = MNQChanged;
}

// 범례 이름과 색상을 종목별로 고정합니다. NQ는 파랑, MNQ는 빨강입니다.
plot1(NQVolumePlot, "NQVolume", Blue);
plot2(MNQVolumePlot, "MNQVolume", Red);
plot3(NQChangedPlot, "NQCloseChanged", Blue);
plot4(MNQChangedPlot, "MNQCloseChanged", Red);
// 상태가 같으면 두 선이 겹칩니다. 해당 봉의 범례 수치로 각각 확인합니다.
