    Else If 삼선_지속목표차 != 삼선_지속목표차[1] &&

// ⑩ 동일한 삼선 폭 유지 구간의 봉당 평균 거래량 비교
// ============================================================
Var : 압축현재거래합(0), 압축직전거래합(0), 압축현재평균(0), 압축직전평균(0),
      압축거래량비율(0), 압축거래량준비(0), 압축가격준비(0), 가격거래동시압축(0), 압축최소봉(0);
압축현재평균 = 0; 압축직전평균 = 0; 압축거래량비율 = 0;
압축거래량준비 = 0; 압축가격준비 = 0; 가격거래동시압축 = 0;
압축최소봉 = Max(1,Round(거래량최소봉수,0));
If 세션초기화 == 1 || 삼선_목표준비 == 0 Then
{
    압축현재거래합 = 0; 압축직전거래합 = 0;
}
Else
{
    압축현재거래합 = 압축현재거래합[1]; 압축직전거래합 = 압축직전거래합[1];
};
If 삼선_목표준비 == 1 Then
{
    If 세션초기화 == 1 || 삼선_목표준비[1] == 0 Then
    {
        압축현재거래합 = Max(0,V); 압축직전거래합 = 0;
    }
    Else If 이탈_구간변경 == 1 Then
    {
        압축직전거래합 = 압축현재거래합[1]; 압축현재거래합 = Max(0,V);
    }
    Else 압축현재거래합 = 압축현재거래합[1]+Max(0,V);
    If 이탈_삼선_지속목표_Cnt > 0 Then 압축현재평균 = 압축현재거래합/이탈_삼선_지속목표_Cnt;
    If 이탈_삼선_지속목표_Cnt_직전 > 0 Then 압축직전평균 = 압축직전거래합/이탈_삼선_지속목표_Cnt_직전;
    If 압축직전평균 > 0 && 이탈_삼선_지속목표_Cnt >= 압축최소봉 && 이탈_삼선_지속목표_Cnt_직전 >= 압축최소봉 Then
    {
        압축거래량준비 = 1;
        압축거래량비율 = 압축현재평균/압축직전평균*100;
    };
    If 이탈_두구간최대가격범위 > 0 Then 압축가격준비 = 1;
};
If 압축가격준비 == 1 && 압축거래량준비 == 1 && 이탈_두구간가격비율 < 가격압축기준 && 압축거래량비율 < 거래량압축기준 Then
    가격거래동시압축 = 1;

// ============================================================
// 표시부: 모든 Plot / NoPlot은 이 아래에만 배치 (번호·위치·색상은 V4와 동일)
// 번호 | 표시명                 | 위치       | 색상       | 굵기
// 1~9  | 미사용
// 10   | 삼선_구간최고대비비율  | 비율 0~100 | 스코프제외 ±1~3 단계색/0압축녹색/기본연회색 | <=30:8+삼선비율점수 / >30:5+삼선비율점수
// 11   | 회귀_구간최고대비비율  | 비율 0~100 | 기본색     | <30:2 / >=30:0
// 12   | 마켓_구간최고대비비율  | 비율 0~100 | 기본색     | <30:2 / >=30:0
// 13   | 이탈_두구간가격비율    | 비율 0~100 | 회색       | <=50:1 / >50:0
// 14~20| 미사용
// 21   | 직전범위위치 | -3 / 103 | 빨강/파랑 | 1   (확정봉판정 적용)
// 22   | 첫상향_직전봉 | -6 | 진빨강 | 1
// 23   | 첫하향_직전봉 | 106 | 진파랑 | 1
// 24   | 복합압축 | -9 | 주황 | 1
// 25   | 가격압축_상향이탈 | -12 | 자홍 | 1
// 26   | 가격압축_하향이탈 | 109 | 청록 | 1
// 27   | 스코프1_상단유지 | -15 | 빨강 | 1   (확정봉판정 적용)
// 28   | 스코프2_하단유지 | 112 | 파랑 | 1   (확정봉판정 적용)
// 29   | 삼선_압축신호점 | 삼선비율(<30) | 위:파랑/아래:빨강/동시:초록 | 3
// 30   | 삼선_비율점수강조 | 삼선비율 | 빨강계열:노랑/파랑계열:Cyan/녹색:흰색 | 3
// 31   | 구간평균거래량비율_확정 | 거래량비율(100초과가능) | 파랑 | 1
// 32   | 가격거래량_동시압축 | -18 | 주황 | 1
// 33   | 스코프종합상태 | -2~+2 (상승=+) | 회색 | 0  (연동 확인용, 차트에서 꺼도 됨)
// 향후 추가 출력은 Plot34부터 순차 배정.
// ============================================================

// Plot10: 삼선 비율. 정확히30% 포함 굵기8~11, 초과 굵기5~8.
NoPlot(10);
If 삼선_목표준비 == 1 Then
    Plot10(삼선_지속목표차_비율,"삼선_구간최고대비비율",삼선점수색상,Def,IFF(삼선_지속목표차_비율 <= 30,8+삼선비율점수,5+삼선비율점수));

// Plot11: 회귀 비율.
NoPlot(11);
If 기준선준비 == 1 Then
    Plot11(평탄회귀차_비율,"회귀_구간최고대비비율",Def,Def,IFF(평탄회귀차_비율 < 30,2,0));

// Plot12: 마켓 비율.
NoPlot(12);
If 마켓_기준선준비 == 1 Then
    Plot12(마켓중심차_비율,"마켓_구간최고대비비율",Def,Def,IFF(마켓중심차_비율 < 30,2,0));

// Plot13: 현재+직전 구간 가격범위 비율. 정확히50%도 굵기1.
NoPlot(13);
If 삼선_목표준비 == 1 && 이탈_두구간최대가격범위 > 0 Then
    Plot13(이탈_두구간가격비율,"이탈_두구간가격비율",RGB(128,128,128),Def,IFF(이탈_두구간가격비율 <= 50,1,0));

// Plot21: 범위 안(위치0)은 숨김.
NoPlot(21);
If      위치사용 == -1 Then Plot21(103,"직전범위위치",RGB(0,0,255),Def,1);
Else If 위치사용 ==  1 Then Plot21(-3,"직전범위위치",RGB(255,0,0),Def,1);

// Plot22~23: 첫이탈 확인, 세션 경계에서는 이전 신호 숨김.
NoPlot(22); NoPlot(23);
If 세션초기화 == 0 && 삼선_목표준비 == 1 Then
{
    If 이탈_첫이탈[1] == 1 Then Plot22(-6,"첫상향_직전봉",RGB(180,0,0),Def,1);
    If 이탈_첫이탈[1] == -1 Then Plot23(106,"첫하향_직전봉",RGB(0,0,150),Def,1);
};

// Plot24~26: 가설 조건이 이어지는 동안 계속 표시.
NoPlot(24); NoPlot(25); NoPlot(26);
If 세션초기화 == 0 && 함수계산가능 == 1 Then
{
    If 복합압축조건[1] == 1 Then Plot24(-9,"복합압축",RGB(255,140,0),Def,1);
    If 가격압축이탈조건[1] == 1 Then Plot25(-12,"가격압축_상향이탈",RGB(255,0,255),Def,1);
    If 가격압축이탈조건[1] == -1 Then Plot26(109,"가격압축_하향이탈",RGB(0,160,160),Def,1);
};

// Plot27~28: 상단유지 우선. 30% 미만/중립/미준비는 숨김.
NoPlot(27); NoPlot(28);
If 스코프사용 == 1 Then Plot27(-15,"스코프1_상단유지",RGB(255,0,0),Def,1);
Else If 스코프사용 == -1 Then Plot28(112,"스코프2_하단유지",RGB(0,0,255),Def,1);

// Plot29: 차트 표시에서 종류를 점 그래프로 설정. 굵기(크기)3.
NoPlot(29);
If 삼선_목표준비 == 1 && 삼선_지속목표차_비율 < 30 && (비율점위신호 == 1 || 비율점아래신호 == 1) Then
    Plot29(삼선_지속목표차_비율,"삼선_압축신호점",비율점색상,Def,3);

// Plot30: 삼선비율점수 양수일 때 비율선 위 강조. 기본회색은 숨김.
NoPlot(30);
If 삼선_목표준비 == 1 && 삼선비율점수 > 0 Then
{
    If 삼선점수색상 == RGB(255,160,160) || 삼선점수색상 == RGB(255,0,0) || 삼선점수색상 == RGB(180,0,0) Then
        Plot30(삼선_지속목표차_비율,"삼선_비율점수강조",RGB(255,255,0),Def,3);
    Else If 삼선점수색상 == RGB(128,160,255) || 삼선점수색상 == RGB(0,0,255) || 삼선점수색상 == RGB(0,0,150) Then
        Plot30(삼선_지속목표차_비율,"삼선_비율점수강조",RGB(0,255,255),Def,3);
    Else If 삼선점수색상 == RGB(0,128,0) Then
        Plot30(삼선_지속목표차_비율,"삼선_비율점수강조",RGB(255,255,255),Def,3);
};

// Plot31~32: 직전 완료봉 조건. 세션 시작에는 이전 구간 값 숨김.
NoPlot(31); NoPlot(32);
If 함수계산가능 == 1 && 세션초기화 == 0 Then
{
    If 압축거래량준비[1] == 1 Then Plot31(압축거래량비율[1],"구간평균거래량비율_확정",RGB(0,0,255),Def,1);
    If 가격거래동시압축[1] == 1 Then Plot32(-18,"가격거래량_동시압축",RGB(255,140,0),Def,1);
};

// Plot33: 스코프종합상태 -2~+2 (상승=+). 다른 지표 연동 확인용.
Plot33(스코프종합상태,"스코프종합상태",RGB(100,100,100),Def,0);
