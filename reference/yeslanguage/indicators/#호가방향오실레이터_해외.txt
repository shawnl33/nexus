//======================================================================
// #호가방향오실레이터V5_해외 (나스닥용)
// 1분봉 호가방향 점수 색상(상·하 방향) + 평활 기울기 진하기 3단계 오실레이터
//======================================================================
Input : 관심기준(15),
        강세기준(35),
        부호반전(0),
        기울기표시(1),
        기울기평활기간(3),
        약기울기기준(2),
        강기울기기준(5);
Var : 일분봉여부(0), 호가점수(0), 호가기울기(0), 호가상태(0), 호가유효(0),
      평활기울기(0), 점수색상(0), 기울기색상(0);
일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1,1,0);
호가점수 = 0; 호가기울기 = 0; 호가상태 = 0; 호가유효 = 0;
If 일분봉여부 == 1 Then
{
    OSF_OrderBookDirectionV2(
        관심기준,강세기준,부호반전,
        호가점수,호가기울기,호가상태,호가유효
    );
};
평활기울기 = Average(호가기울기,Max(1,기울기평활기간));
점수색상 = RGB(170,170,170);
기울기색상 = RGB(150,150,150);
// 방향은 호가점수의 0선 위·아래로 결정한다.
// 평활 기울기 절대값은 색의 진하기만 3단계로 조절한다.
If 호가점수 > 0 Then
{
    If Abs(평활기울기) >= 강기울기기준 Then
        점수색상 = RGB(220,0,0);
    Else If Abs(평활기울기) >= 약기울기기준 Then
        점수색상 = RGB(255,100,70);
    Else
        점수색상 = RGB(255,185,185);
}
Else If 호가점수 < 0 Then
{
    If Abs(평활기울기) >= 강기울기기준 Then
        점수색상 = RGB(0,0,180);
    Else If Abs(평활기울기) >= 약기울기기준 Then
        점수색상 = RGB(60,130,255);
    Else
        점수색상 = RGB(180,210,255);
}
Else
{
    점수색상 = RGB(150,150,150);
};
기울기색상 = IFF(평활기울기 > 0,RGB(255,100,100),
             IFF(평활기울기 < 0,RGB(100,135,255),
			 RGB(150,150,150)));
If 일분봉여부 == 1 && 호가유효 == 1 Then
{
    Plot1(호가점수,"호가방향",점수색상,Def,4);
    If 기울기표시 == 1 Then
        Plot2(평활기울기,"평활호가기울기",기울기색상,Def,2);
    Else
        NoPlot(2);
}
Else
{
    NoPlot(1); NoPlot(2);
};
If 일분봉여부 == 1 Then
{
    Plot3(0,"영선",RGB(90,90,90),Def,0);
    Plot4(관심기준,"매수관심",RGB(255,170,170),Def,0);
    Plot5(-관심기준,"매도관심",RGB(170,190,255),Def,0);
    Plot6(강세기준,"매수강세",RGB(255,0,0),Def,0);
    Plot7(-강세기준,"매도강세",RGB(0,0,255),Def,0);
}
Else
{
    NoPlot(3); NoPlot(4); NoPlot(5); NoPlot(6); NoPlot(7);
};
