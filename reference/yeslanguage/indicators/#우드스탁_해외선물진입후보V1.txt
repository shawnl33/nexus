// #우드스탁_해외선물진입후보V1  (가격 차트 겹치기, 1분봉 전용)  - 작업 플랜 5단계: 진입 후보 신호
// 재료: 1분 통합판정(미래곡선→세션매물대→세션ADX/DI, WSF_FXDecisionV2_CO) + 매물대 압축 해제(WSF_FXSqueezeV1_CO)
// 후보 발생 (둘 중 하나):
//   ① 통합상태가 +100(-100)에 처음 들어선 봉
//   ② 통합상태가 이미 +100(-100)인 동안 같은 방향 압축 해제(즉시/지연확정)가 나온 봉
// 등급: 최근 해제유효봉수 이내 가장 최근 해제가
//   같은 방향 즉시해제 = A, 같은 방향 지연확정 = B, 해제 없음 = C, 반대 방향 = 후보 무효
//   해제필수 = 1이면 C 등급은 내지 않음
// 중복 금지: 같은 방향 후보 뒤에는 통합상태가 0 이하(매수)/0 이상(매도)로 돌아갈 때까지 같은 방향 후보 없음
// 저장: 후보방향, 후보등급, 진입가격(후보 봉 종가), 진입날짜, 진입시각. 세션(07:00/15:30) 경계에서 초기화.
// 사용 함수: WSF_FXSessionKeyV1, WSF_FXCurveV1, WSF_FXSessionProfileV1_CO, WSF_FXADXV1, WSF_FXDecisionV2_CO,
//            WSF_FXSessionProfileV2_CO, WSF_FXSqueezeV1_CO
// Plot1 진입후보: 점 그래프 권장. 매수=저가 아래, 매도=고가 위. 진하기/크기 A>B>C
// Plot2 진입가격선: 마지막 후보의 진입가격을 다음 후보 또는 세션 끝까지 유지
Input : 예측변수(10),
        가치영역배수(1.5),
        최소봉수(5),
        ADX기간(14),
        ADX추세기준(20),
        ADX강세기준(35),
        매물대기준모드(1),
        폭계산봉수(30),
        비율기준봉수(120),
        좁음기준(40),
        단계1봉수(15),
        방향확정봉수(5),
        확정봉판정(0),
        해제유효봉수(10),
        해제필수(0),
        표시여유틱(4);
Var : FX세션키(0), FX초기화(0), FX세션봉수(0),
      곡선하(0), 곡선상(0), 곡선예측가격(0), 곡선예측변화(0), 곡선예측방향(0), 미래방향(0),
      매물대중심(0), 매물대상단1(0), 매물대하단1(0), 매물대상단2(0), 매물대하단2(0),
      매물대5상태(0), 매물대유효(0),
      세션ADX(0), 상승DI(0), 하락DI(0), ADX유효(0),
      통합상태(0), 매물대상태(0), ADX상태(0), DI방향(0),
      압축해제(0), 압축지속봉수(0), 폭비율(0), 즉시해제방향(0), 지연확정방향(0), 해제길이(0),
      최근해제방향(0), 최근해제종류(0), 최근해제경과(999),
      트리거방향(0), 후보방향(0), 후보등급(0), 차단방향(0),
      진입방향(0), 진입등급(0), 진입가격(0), 진입날짜(0), 진입시각(0),
      후보가격(0), 후보색상(0), 후보굵기(0), 선색상(0);

// 세션 (07:00 / 15:30)
FX세션키 = WSF_FXSessionKeyV1(sDate,sTime);
FX초기화 = 0;
If CurrentBar == 1 Then FX초기화 = 1;
Else If FX세션키 != FX세션키[1] Then FX초기화 = 1;
If FX초기화 == 1 Then FX세션봉수 = 1;
Else FX세션봉수 = FX세션봉수[1]+1;

If DataCompress == 2 && BarInterval == 1 Then
{
    // ① 1분 통합판정 (#우드스탁_해외선물1분통합판정V1과 같은 계산)
    WSF_FXCurveV1(FX초기화,FX세션봉수,예측변수,
                  곡선하,곡선상,곡선예측가격,곡선예측변화,곡선예측방향);
    미래방향 = IFF(곡선예측방향 > 0, 1, IFF(곡선예측방향 < 0, -1, 0));
    If FX초기화 == 1 Then 미래방향 = 0;
    WSF_FXSessionProfileV1_CO(FX초기화,가치영역배수,최소봉수,
        매물대중심,매물대상단1,매물대하단1,매물대상단2,매물대하단2,
        매물대5상태,매물대유효);
    WSF_FXADXV1(FX초기화,FX세션봉수,ADX기간,세션ADX,상승DI,하락DI,ADX유효);
    WSF_FXDecisionV2_CO(미래방향,매물대5상태,매물대유효,매물대중심,C,매물대기준모드,
        세션ADX,상승DI,하락DI,ADX유효,ADX추세기준,ADX강세기준,
        통합상태,매물대상태,ADX상태,DI방향);

    // ② 매물대 압축 해제 (#우드스탁_해외선물매물대압축지속V3와 같은 계산)
    압축해제 = WSF_FXSqueezeV1_CO(FX초기화,FX세션봉수,가치영역배수,최소봉수,
        폭계산봉수,비율기준봉수,좁음기준,단계1봉수,방향확정봉수,확정봉판정,
        압축지속봉수,폭비율,즉시해제방향,지연확정방향,해제길이);

    // 상태 이어받기 (같은 봉 재평가는 직전 봉 상태에서 시작)
    If CurrentBar > 1 Then
    {
        최근해제방향 = 최근해제방향[1]; 최근해제종류 = 최근해제종류[1]; 최근해제경과 = 최근해제경과[1];
        차단방향 = 차단방향[1];
        진입방향 = 진입방향[1]; 진입등급 = 진입등급[1]; 진입가격 = 진입가격[1];
        진입날짜 = 진입날짜[1]; 진입시각 = 진입시각[1];
    };
    If FX초기화 == 1 Then
    {
        최근해제방향 = 0; 최근해제종류 = 0; 최근해제경과 = 999;
        차단방향 = 0;
        진입방향 = 0; 진입등급 = 0; 진입가격 = 0; 진입날짜 = 0; 진입시각 = 0;
    };

    // 가장 최근 해제 기억 (종류 1=즉시, 2=지연확정)
    최근해제경과 = 최근해제경과 + 1;
    If 즉시해제방향 != 0 Then
    {
        최근해제방향 = 즉시해제방향; 최근해제종류 = 1; 최근해제경과 = 0;
    }
    Else If 지연확정방향 != 0 Then
    {
        최근해제방향 = 지연확정방향; 최근해제종류 = 2; 최근해제경과 = 0;
    };

    // 중복 금지 해제: 통합상태가 반대쪽 또는 0으로 돌아가면 같은 방향 차단을 푼다
    If 차단방향 == 1 && 통합상태 <= 0 Then 차단방향 = 0;
    If 차단방향 == -1 && 통합상태 >= 0 Then 차단방향 = 0;

    // 트리거
    트리거방향 = 0;
    If FX초기화 == 0 Then
    {
        If 통합상태 == 100 && 통합상태[1] != 100 Then 트리거방향 = 1;
        Else If 통합상태 == -100 && 통합상태[1] != -100 Then 트리거방향 = -1;
        Else If 통합상태 == 100 && 최근해제경과 == 0 && 최근해제방향 == 1 Then 트리거방향 = 1;
        Else If 통합상태 == -100 && 최근해제경과 == 0 && 최근해제방향 == -1 Then 트리거방향 = -1;
    };

    // 등급과 무효 판정
    후보방향 = 0; 후보등급 = 0;
    If 트리거방향 != 0 && 차단방향 != 트리거방향 Then
    {
        If 최근해제방향 != 0 && 최근해제경과 <= 해제유효봉수 Then
        {
            If 최근해제방향 == 트리거방향 Then
            {
                후보방향 = 트리거방향;
                후보등급 = IFF(최근해제종류 == 1,3,2);
            };
            // 반대 방향 해제가 최근이면 후보 무효
        }
        Else If 해제필수 == 0 Then
        {
            후보방향 = 트리거방향;
            후보등급 = 1;
        };
    };

    // 후보 저장
    If 후보방향 != 0 Then
    {
        차단방향 = 후보방향;
        진입방향 = 후보방향; 진입등급 = 후보등급; 진입가격 = C;
        진입날짜 = sDate; 진입시각 = sTime;
    };

    // 표시
    If 후보방향 != 0 Then
    {
        If 후보방향 == 1 Then
        {
            후보가격 = L - 표시여유틱*PriceScale;
            후보색상 = IFF(후보등급 == 3,RGB(220,0,0),IFF(후보등급 == 2,RGB(255,100,100),RGB(255,180,180)));
        }
        Else
        {
            후보가격 = H + 표시여유틱*PriceScale;
            후보색상 = IFF(후보등급 == 3,RGB(0,0,200),IFF(후보등급 == 2,RGB(80,120,255),RGB(170,190,255)));
        };
        후보굵기 = 2 + 후보등급;
        Plot1(후보가격,"진입후보",후보색상,Def,후보굵기);
    }
    Else NoPlot(1);

    If 진입방향 != 0 && 진입가격 > 0 Then
    {
        선색상 = IFF(진입방향 == 1,RGB(255,120,120),RGB(120,150,255));
        Plot2(진입가격,"진입가격선",선색상,Def,1);
    }
    Else NoPlot(2);
}
Else
{
    NoPlot(1); NoPlot(2);
};
