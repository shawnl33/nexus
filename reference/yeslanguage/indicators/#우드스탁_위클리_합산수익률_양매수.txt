For k = 3 To 3
{
    만남[k] = False;
    If 추적허용 && 참조주기정상[k] && 참조가격[k] > 0 && 참조날짜[k] == sDate && 참조시간[k] == sTime Then
    {
        If 만남방식 == 1 Then
            만남[k] = Max(L, 참조저가[k]) <= Min(H, 참조고가[k]) + 오차;
        If 만남방식 == 2 Then
            만남[k] = C - 오차 <= 참조가격[k] && C + 오차 >= 참조가격[k];
    }

    // 첫 유효 만남만 저장하고 다음 영업일까지 유지.
    If 추적허용 && 저장유효[k] == False && 만남[k] && C > 0 Then
    {
        // 두 차트에서 동일한 공통 표시가격. 겹치는 가격 범위 중간값을 표시.
        If 만남방식 == 1 Then
            저장가격[k] = (Max(L, 참조저가[k]) + Min(H, 참조고가[k])) / 2;
        Else
            저장가격[k] = (C + 참조가격[k]) / 2;
        자기만남가격 = C;
        상대만남가격 = 참조가격[k];
        저장날짜[k] = sDate;
        저장시간[k] = sTime;
        저장유효[k] = True;
    }
}

// 하단 별도 영역. 첫 유효 만남 봉의 각 종목 종가, 동일 투자금50:50.
// 청산/분할과 무관하게 당일 가상 보유 수익률을 표시.
If 주기정상 && 참조주기정상[3] && 저장유효[3] &&
   자기만남가격 > 0 && 상대만남가격 > 0 && C > 0 && 참조가격[3] > 0 &&
   참조날짜[3] == sDate && 참조시간[3] == sTime Then
{
    합산수익률 = ((C / 자기만남가격 - 1) + (참조가격[3] / 상대만남가격 - 1)) * 50;
    Plot1(합산수익률, "첫만남합산수익률");
}
Else NoPlot(1);
Plot2(0, "영점기준");
