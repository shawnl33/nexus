// #우드스탁_해외선물스윙전체폭V2  (하단 지표, 1분봉 전용)
// 지난상승·지난하락 두 스윙 구간을 합친 전체 폭
//   전체최고 = Max(핵심지난상승최고가, 핵심지난하락최고가)
//   전체최저 = Min(핵심지난상승최저가, 핵심지난하락최저가)
//   스윙전체폭 = 전체최고 - 전체최저   (네 값이 모두 있을 때만 계산, 세션 07:00/15:30마다 초기화)
// 값은 #우드스탁_해외선물미래곡선과 같은 함수(WSF_FXFutureValuesV3_CO)·같은 기본 입력으로 계산.
// 표시 = % (세션 초기화 이후 최대 스윙전체폭 대비, 0~100)
// Plot1 스윙전체폭 (막대/선)
Input : 예측변수(10),
        예측봉수1(5),
        예측봉수2(10),
        예측봉수3(15),
        예측봉수4(30),
        예측봉수5(60),
        최소신뢰도(0.4),
        지속봉수(5),
        마켓계산기간(20),
        최소전환유지봉수(3),
        스윙연결(0),
        기세무시틱(1);
Var : 단계화_1분_통합(0),
      곡선회귀선_평탄(0),
      마켓중심가격(0),
      지속저장목표1(0),
      지속저장목표2(0),
      지속저장목표3(0),
      지속저장목표4(0),
      지속저장목표5(0),
      핵심지난상승최고가(0),
      핵심지난상승최저가(0),
      핵심지난상승_382(0),
      핵심지난상승_500(0),
      핵심지난상승_618(0),
      핵심지난하락최고가(0),
      핵심지난하락최저가(0),
      핵심지난하락_382(0),
      핵심지난하락_500(0),
      핵심지난하락_618(0),
      함수계산가능(0),
      전체최고(0), 전체최저(0), 스윙전체폭(0), 표시값(0),
      세션키(0), 세션최대폭(0);

함수계산가능 = WSF_FXFutureValuesV3_CO(
    예측변수,
    예측봉수1,
    예측봉수2,
    예측봉수3,
    예측봉수4,
    예측봉수5,
    최소신뢰도,
    지속봉수,
    마켓계산기간,
    최소전환유지봉수,
    단계화_1분_통합,
    곡선회귀선_평탄,
    마켓중심가격,
    지속저장목표1,
    지속저장목표2,
    지속저장목표3,
    지속저장목표4,
    지속저장목표5,
    핵심지난상승최고가,
    핵심지난상승최저가,
    핵심지난상승_382,
    핵심지난상승_500,
    핵심지난상승_618,
    핵심지난하락최고가,
    핵심지난하락최저가,
    핵심지난하락_382,
    핵심지난하락_500,
    핵심지난하락_618,
    스윙연결,
    기세무시틱
);

세션키 = WSF_FXSessionKeyV1(sDate,sTime);
세션최대폭 = 세션최대폭[1];
If CurrentBar <= 1 || 세션키 != 세션키[1] Then 세션최대폭 = 0;

If 함수계산가능 == 1 &&
   핵심지난상승최고가 > 0 && 핵심지난하락최고가 > 0 &&
   핵심지난상승최저가 > 0 && 핵심지난하락최저가 > 0 Then
{
    전체최고 = Max(핵심지난상승최고가,핵심지난하락최고가);
    전체최저 = Min(핵심지난상승최저가,핵심지난하락최저가);
    스윙전체폭 = 전체최고-전체최저;
    If 스윙전체폭 > 세션최대폭 Then 세션최대폭 = 스윙전체폭;
    표시값 = 0;
    If 세션최대폭 > 0 Then 표시값 = 스윙전체폭/세션최대폭*100;
    Plot1(표시값,"스윙전체폭",RGB(120,60,160),Def,2);
}
Else
    NoPlot(1);
