// 지표명: WS_CrossSizeRV_Research_04_ZScore
// 목적: NQ와 MNQ의 Raw Spread가 최근 평균에서 얼마나 떨어졌는지 Z-score로 표시합니다.
// Data1: E-mini Nasdaq-100 선물 NQ(주종목, 동일 월물)
// Data2: Micro E-mini Nasdaq-100 선물 MNQ(참조종목, 동일 월물)
// 주의: 주문, 진입, 청산 등 자동매매 코드는 포함하지 않습니다.

// Spread 평균과 표준편차를 계산할 봉 수이며 기본값은 120개 1분봉입니다.
input : RollingPeriod(120);

// RawSpread는 두 종가의 차이, SpreadMean은 이동평균, SpreadStdDev는 표준편차를 저장합니다.
// SpreadZScore는 계산 가능한 시점 전에는 Nan(표시하지 않음)으로 시작합니다.
var : RawSpread(0), SpreadMean(0), SpreadStdDev(0), SpreadZScore(Nan);

// Data2(MNQ) 종가에서 Data1(NQ) 종가를 빼서 Raw Spread를 계산합니다.
RawSpread = data2(C) - C;
// 지정한 기간 동안 Raw Spread의 단순이동평균을 계산합니다.
SpreadMean = ma(RawSpread, RollingPeriod);
// 지정한 기간 동안 Raw Spread의 표준편차를 계산합니다.
SpreadStdDev = std(RawSpread, RollingPeriod);

// 충분한 봉이 쌓였고 표준편차가 0보다 큰 경우에만 Z-score를 계산합니다.
if CurrentBar >= RollingPeriod and SpreadStdDev > 0 Then {
    // 현재 Spread와 평균의 차이를 표준편차로 나누어 Z-score를 계산합니다.
    SpreadZScore = (RawSpread - SpreadMean) / SpreadStdDev;
} else {
    // 계산 조건을 만족하지 않는 초기 구간은 값을 표시하지 않습니다.
    SpreadZScore = Nan;
}

// 첫 번째 Plot으로 계산된 Z-score를 출력합니다.
plot1(SpreadZScore, "SpreadZScore");
// 두 번째 Plot으로 +2 기준선을 출력합니다.
plot2(2, "ZUpper2");
// 세 번째 Plot으로 0 기준선을 출력합니다.
plot3(0, "ZZero");
// 네 번째 Plot으로 -2 기준선을 출력합니다.
plot4(-2, "ZLower2");
