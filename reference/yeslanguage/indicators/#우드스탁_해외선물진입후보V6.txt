// #우드스탁_해외선물진입후보V6  (1분봉 전용)  - V5 + 즉시해제 A 경로·거짓 돌파 뒤 재돌파·B 보완 (WSF_FXEntryCandV3_CO)
// 해제A사용=0, 재돌파허용봉수=0, B보완사용=0 이면 V5와 같음. B사용=0 이면 A만.
// (아래는 V5 설명) V4 + 진입 시간대(기본 20:00~02:00)·세션당 최대진입횟수(기본 2)
// 시각은 HHMMSS. 시간대 밖이거나 횟수를 넘은 후보는 표시하지 않음. 최대진입횟수 0이면 횟수 제한 없음.
// (아래는 V4 설명)
// V3 + 변동성 확장 확인: 확인된 박스 돌파 뒤 확장확인봉수(5) 이내에 압축이 실제로 풀려야 A 후보.
//   돌파 시점에 이미 압축이 풀려 있으면 바로 A. 그 사이 종가가 박스 안으로 돌아오면 취소.
// 계산은 WSF_FXEntryCandV3_CO(판정방식 4)에 있음. 판정방식을 1~3으로 바꾸면 V1~V3 규칙으로 표시.
// 사용 함수: WSF_FXSessionKeyV1, WSF_FXEntryCandV3_CO (내부: WSF_FXCurveV1, WSF_FXSessionProfileV1_CO,
//            WSF_FXADXV1, WSF_FXDecisionV2_CO, WSF_FXSqueezeV4_CO, WSF_FXSessionProfileV2_CO)
// Plot1 진입후보: 점 그래프 권장. 매수=저가 아래, 매도=고가 위. 진하기/크기 A>B>C
// Plot2 진입가격선: 마지막 후보의 진입가격을 다음 후보 또는 세션 끝까지 유지
Input : 판정방식(4),
        예측변수(10),
        가치영역배수(1.5),
        최소봉수(5),
        ADX기간(14),
        ADX추세기준(20),
        ADX강세기준(35),
        매물대기준모드(1),
        폭계산봉수(30),
        비율기준봉수(120),
        좁음기준(40),
        단계1봉수(15),
        방향확정봉수(5),
        확정봉판정(0),
        해제유효봉수(10),
        등급C사용(-1),
        돌파여유틱(-1),
        돌파통합최소(50),
        돌파확인봉수(1),
        DI일치필수(1),
        확장확인봉수(5),
        진입시작시각(200000),
        진입종료시각(20000),
        최대진입횟수(2),
        해제A사용(1),
        재돌파허용봉수(3),
        B보완사용(1),
        B사용(1),
        표시여유틱(4);
Var : FX세션키(0), FX초기화(0), FX세션봉수(0),
      후보방향(0), 후보등급(0), 통합상태(0),
      진입방향(0), 진입등급(0), 진입가격(0), 진입날짜(0), 진입시각(0),
      후보가격(0), 후보색상(0), 선색상(0);

FX세션키 = WSF_FXSessionKeyV1(sDate,sTime);
FX초기화 = 0;
If CurrentBar == 1 Then FX초기화 = 1;
Else If FX세션키 != FX세션키[1] Then FX초기화 = 1;
If FX초기화 == 1 Then FX세션봉수 = 1;
Else FX세션봉수 = FX세션봉수[1]+1;

If DataCompress == 2 && BarInterval == 1 Then
{
    WSF_FXEntryCandV3_CO(FX초기화,FX세션봉수,판정방식,
        예측변수,가치영역배수,최소봉수,ADX기간,ADX추세기준,ADX강세기준,매물대기준모드,
        폭계산봉수,비율기준봉수,좁음기준,단계1봉수,방향확정봉수,확정봉판정,해제유효봉수,
        등급C사용,돌파여유틱,돌파통합최소,돌파확인봉수,DI일치필수,확장확인봉수,
        진입시작시각,진입종료시각,최대진입횟수,
        해제A사용,재돌파허용봉수,B보완사용,B사용,
        후보방향,후보등급,통합상태);

    If CurrentBar > 1 Then
    {
        진입방향 = 진입방향[1]; 진입등급 = 진입등급[1]; 진입가격 = 진입가격[1];
        진입날짜 = 진입날짜[1]; 진입시각 = 진입시각[1];
    };
    If FX초기화 == 1 Then
    {
        진입방향 = 0; 진입등급 = 0; 진입가격 = 0; 진입날짜 = 0; 진입시각 = 0;
    };
    If 후보방향 != 0 Then
    {
        진입방향 = 후보방향; 진입등급 = 후보등급; 진입가격 = C;
        진입날짜 = sDate; 진입시각 = sTime;
    };

    If 후보방향 == 1 Then
    {
        후보가격 = L - 표시여유틱*PriceScale;
        후보색상 = IFF(후보등급 == 3,RGB(220,0,0),IFF(후보등급 == 2,RGB(255,100,100),RGB(255,180,180)));
        Plot1(후보가격,"진입후보",후보색상,Def,2+후보등급);
    }
    Else If 후보방향 == -1 Then
    {
        후보가격 = H + 표시여유틱*PriceScale;
        후보색상 = IFF(후보등급 == 3,RGB(0,0,200),IFF(후보등급 == 2,RGB(80,120,255),RGB(170,190,255)));
        Plot1(후보가격,"진입후보",후보색상,Def,2+후보등급);
    }
    Else NoPlot(1);

    If 진입방향 != 0 && 진입가격 > 0 Then
    {
        선색상 = IFF(진입방향 == 1,RGB(255,120,120),RGB(120,150,255));
        Plot2(진입가격,"진입가격선",선색상,Def,1);
    }
    Else NoPlot(2);
}
Else
{
    NoPlot(1); NoPlot(2);
};
