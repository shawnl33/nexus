//======================================================================
// #우드스탁_미래곡선_1분봉통합신호V19_검증지표_해외
// V17 미래방향 계산 유지
// 새 세션 첫 봉은 미래방향을 0으로 표시하고 전일 방향을 이월하지 않음
//======================================================================

Input : 예측변수(10),
마켓계산기간(20),
회귀최소신뢰도(0.40),
호가관심기준(15),
호가강세기준(35),
호가부호반전(0),호가기울기평활기간(3),
호가약기울기기준(2),
호가강기울기기준(5);

Var : 일분봉여부(0),
세션첫봉(0),

곡선하(0),
곡선상(0),
예측가격(0),
예측변화(0),
원시미래방향(0),
세션미래방향(0),
미래방향부호(0),

곡선회귀선(0),
곡선기울기(0),
곡선구분(0),
곡선유효(0),
곡선신뢰도(0),
곡선잔차(0),
평탄회귀선(0),
회귀방향(0),

마켓중심(0),
마켓기울기(0),
마켓방향(0),

호가점수(0),
호가기울기(0),
호가상태(0),
호가유효(0),
호가방향(0),

통합방향(0),
통합상태(0),
매수후보(0),
매도후보(0),
차단사유(0),
우선순위단계(0),

통합색상(0),
미래색상(0),
호가색상(0),
평활호가기울기(0)
;

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

세션첫봉 = 0;

If CurrentBar == 1 Then
    세션첫봉 = 1;

If CurrentBar > 1 && BDate <> BDate[1] Then
    세션첫봉 = 1;

/* V17과 동일하게 미래곡선 계산은 항상 수행 */
OSF_Htf_CurvePredict(
예측변수,
곡선하,
곡선상,
예측가격,
예측변화,
원시미래방향
);

/* 세션 첫 봉에만 전일 미래방향 표시를 차단 */
If 세션첫봉 == 1 Then
    세션미래방향 = 0;
Else
    세션미래방향 = 원시미래방향;

미래방향부호 =
IFF(세션미래방향 > 0, 1,
IFF(세션미래방향 < 0, -1, 0));

OSF_Mtf_LinRegV3(
(H + L) / 2,
곡선회귀선,
곡선기울기,
곡선구분,
곡선유효,
곡선신뢰도,
곡선잔차
);

평탄회귀선 = Round(곡선회귀선 / PriceScale, 0) * PriceScale;

회귀방향 = 0;

If 곡선유효 == 1 && 곡선신뢰도 >= 회귀최소신뢰도 Then
{
    If C > 평탄회귀선 Then
        회귀방향 = 1;

    If C < 평탄회귀선 Then
        회귀방향 = -1;
};

/* 마켓중심: 방향 필터 */
마켓중심 = Average((H + L) / 2, 마켓계산기간);
마켓기울기 = 0;
마켓방향 = 0;

If CurrentBar > 1 Then
    마켓기울기 = 마켓중심 - 마켓중심[1];

If C > 마켓중심 && 마켓기울기 > 0 Then
    마켓방향 = 1;

If C < 마켓중심 && 마켓기울기 < 0 Then
    마켓방향 = -1;

/* 호가 방향: #호가방향오실레이터V5 기준 */
호가점수 = 0;
호가기울기 = 0;
호가상태 = 0;
호가유효 = 0;
호가방향 = 0;

If 일분봉여부 == 1 Then
{
    OSF_OrderBookDirectionV2(
        호가관심기준,
        호가강세기준,
        호가부호반전,
        호가점수,
        호가기울기,
        호가상태,
        호가유효
    );
};

평활호가기울기 = Average(호가기울기, Max(1, 호가기울기평활기간));
호가색상 = RGB(150,150,150);

If 호가점수 > 0 Then
{
    If Abs(평활호가기울기) >= 호가강기울기기준 Then
        호가색상 = RGB(220,0,0);
    Else If Abs(평활호가기울기) >= 호가약기울기기준 Then
        호가색상 = RGB(255,100,70);
    Else
        호가색상 = RGB(255,185,185);
}
Else If 호가점수 < 0 Then
{
    If Abs(평활호가기울기) >= 호가강기울기기준 Then
        호가색상 = RGB(0,0,180);
    Else If Abs(평활호가기울기) >= 호가약기울기기준 Then
        호가색상 = RGB(60,130,255);
    Else
        호가색상 = RGB(180,210,255);
};

If 호가점수 > 0 Then
    호가방향 = 1;
Else If 호가점수 < 0 Then
    호가방향 = -1;

/* 우선순위: 마켓방향 → 평탄회귀 → 미래방향 → 호가방향 */
통합방향 = 0;
통합상태 = 0;
매수후보 = 0;
매도후보 = 0;
차단사유 = 0;
우선순위단계 = 0;

If 일분봉여부 == 1 Then
{
    If 마켓방향 == 1 Then
        우선순위단계 = 1;

    If 마켓방향 == 1 && 회귀방향 == 1 Then
        우선순위단계 = 2;

    If 마켓방향 == 1 && 회귀방향 == 1 && 미래방향부호 == 1 Then
        우선순위단계 = 3;

    If 마켓방향 == 1 && 회귀방향 == 1 && 미래방향부호 == 1 Then
    {
        If 호가유효 == 0 || 호가방향 >= 0 Then
        {
            통합방향 = 1;
            통합상태 = 100;
            매수후보 = 1;
            우선순위단계 = 4;
        }
        Else
            차단사유 = -1;
    };

    If 마켓방향 == -1 Then
        우선순위단계 = -1;

    If 마켓방향 == -1 && 회귀방향 == -1 Then
        우선순위단계 = -2;

    If 마켓방향 == -1 && 회귀방향 == -1 && 미래방향부호 == -1 Then
        우선순위단계 = -3;

    If 마켓방향 == -1 && 회귀방향 == -1 && 미래방향부호 == -1 Then
    {
        If 호가유효 == 0 || 호가방향 <= 0 Then
        {
            통합방향 = -1;
            통합상태 = -100;
            매도후보 = 1;
            우선순위단계 = -4;
        }
        Else
            차단사유 = 1;
    };
};

통합색상 =
IFF(통합상태 > 0, RGB(220,0,0),
IFF(통합상태 < 0, RGB(0,0,220),
RGB(130,130,130)));

미래색상 =
IFF(세션미래방향 > 0, RGB(255,130,0),
IFF(세션미래방향 < 0, RGB(0,120,255),
RGB(140,140,140)));

호가색상 =
IFF(호가점수 > 0, RGB(220,0,0),
IFF(호가점수 < 0, RGB(0,0,220),
RGB(140,140,140)));

Var : 단계화_1분_통합(0), 단계화_1분_색상(0),단계화_1분_세션미래방향(0);
단계화_1분_세션미래방향 = 
IFF(세션미래방향 > 0, 2,-2)+
IFF(세션미래방향 > 세션미래방향[1],  1, 0) +
IFF(세션미래방향 < 세션미래방향[1],  -1, 0);

단계화_1분_통합 = 단계화_1분_세션미래방향 + 마켓방향 + 회귀방향+호가방향;


If 단계화_1분_통합 > 0 Then
{
    If 단계화_1분_통합 > 4 Then
        단계화_1분_색상 = RGB(220,0,0);
    Else If 단계화_1분_통합 > 2 Then
        단계화_1분_색상 = RGB(255,100,70);
    Else
        단계화_1분_색상 = RGB(255,185,185);
}
Else If 단계화_1분_통합 < 0 Then
{
    If 단계화_1분_통합 < -4  Then
        단계화_1분_색상 = RGB(0,0,180);
    Else If 단계화_1분_통합 < -2  Then
        단계화_1분_색상 = RGB(60,130,255);
    Else
        단계화_1분_색상 = RGB(180,210,255);
}
Else
{
    단계화_1분_색상 = RGB(150,150,150);
};



If 일분봉여부 == 1 Then
{
 Plot1(단계화_1분_통합,"단계화_1분_통합",단계화_1분_색상,def,2+abs(단계화_1분_통합));

 }
Else
{
    NoPlot(1);
};
