// #우드스탁_미래곡선V16_1분봉용_해외_OSF (나스닥/MNQ 등 해외선물 테스트용)
// - 원본(WSF_ 접두, KP200용)에서 분기한 별도 지표. 원본은 절대 수정하지 않음.
// - 함수/식별자 명명 규칙 변경: 함수명에 한글이 들어가면 플랫폼이 인식하지
//   못하는 것을 실제 등록 테스트로 확인했음. 그래서 "WSF_" 대신 순수 영문
//   접두 "OSF_"(OverSeas Function)를 사용. 한글 접미(_해외)를 붙였던
//   이전 버전은 폐기하고 이 파일로 교체함.
// - 갭(Gap) 로직 우회: KP200은 주간장/야간장 사이에 실제 의미있는 갭이
//   생기지만, 나스닥/MNQ 등 해외선물은 사실상 거의 갭 없이 연속 거래된다.
//   그래서 OSF_GapRegimeV1 호출 자체를 제거하고, 항상 "갭 없음/풀비중"
//   상태로 고정한 뒤 OSF_1m_DailyAlignV2를 무조건(gate 없이) 호출한다.
//   = 일봉 추세와 1분 합의방향이 같은 방향일 때만 신호를 인정하는
//     KP200의 "일반장(갭 없음)" 분기를 상시 적용하는 것과 동일한 동작.
//   추후 해외 종목에도 갭 대응이 필요하다고 판단되면 OSF_GapRegimeV1.txt
//   자체는 이 폴더에 그대로 남겨두었으니 다시 연결하면 된다(현재는 미사용).
// 장중 1분봉 전용: 회귀예측·일봉추세 연결·호가 결합 운용
// 기존 예측.txt는 유지하고, 방향 전환 당시의 상단 3선·하단 3선을 Plot으로 보존
// 1분봉 조회 5,000봉 / 일봉 조회 5000봉
// 1분봉: 장 시작 회귀 초기화 + 일봉 전체연결(갭 무시) + 매매상태 표시
// 일봉  : 20일 자동 회귀 + 5/10/15일 미래가격 표시
//======================================================================

Input : 예측변수(10),
        예측봉수1(5),
        예측봉수2(10),
        예측봉수3(15),
        과거예측표시(1),
        현재미래표시(1),
        예측범위표시(1),
        기존보조선표시(0),
        최소신뢰도(0.40),
        매매상태표시(1),
        일봉회귀기간(10),
        일봉최소신뢰도(0.40),
        최종최소강도(40),
        중간갭기준(0.35),
        큰갭기준(0.75),
        큰갭재평가분(30),
        호가결합사용(1),
        호가관심기준(10),
        호가강세기준(35),
        가파른기울기틱(4),
        가파른추가확인봉(1),
        지속봉수(5),
        ADX기간(14),
        DI평활기간(3),
        DI차이최소(3),
        마켓계산기간(20),
        마켓밴드배수(1.0),
        마켓밴드표시(1),
        페이드기준시간비율(1.0),
        시간비율표시(1),
        최소전환유지봉수(3);

// 예측변수와 기존보조선표시는 기존 저장값의 입력 순서 호환을 위한 예약값

// 현재 진행 범위와 과거 저장 범위를 분리한다.
// 현재미래표시: 우측 현재 진행선 및 현재 6선
// 과거예측표시 + 예측범위표시: 과거 시점에 저장된 6선
// 분봉·일봉 구분 없이 현재 진행 범위와 과거 저장 범위를 각각 제어한다.
Var : 현재범위표시_적용(0), 과거범위표시_적용(0);

현재범위표시_적용 = IFF(현재미래표시 > 0, 1, 0);
과거범위표시_적용 =
    IFF(과거예측표시 > 0 && 예측범위표시 > 0, 1, 0);

Var: 곡선회귀선(0), 곡선회귀기울기(0),
     곡선회귀선_구분(0), 곡선회귀유효(0),
     곡선회귀신뢰도(0), 곡선회귀잔차(0);

// 다른 Plot 색상에서 공통으로 사용할 1분봉 최종 단계값
Var : 단계화_1분_통합(0), 단계화_1분_색상(0),
      핵심일분봉여부(0), 핵심세션첫봉(0),
      핵심곡선하(0), 핵심곡선상(0), 핵심예측가격(0), 핵심예측변화(0),
      핵심원시미래방향(0), 핵심세션미래방향(0), 핵심미래방향부호(0),
      핵심곡선회귀선(0), 핵심곡선기울기(0), 핵심곡선구분(0), 핵심곡선유효(0),
      핵심곡선신뢰도(0), 핵심곡선잔차(0), 핵심평탄회귀선(0), 핵심회귀방향(0),
      핵심마켓중심(0), 핵심마켓기울기(0), 핵심마켓방향(0),
      핵심호가점수(0), 핵심호가기울기(0), 핵심호가상태(0), 핵심호가유효(0), 핵심호가방향(0);

OSF_Mtf_LinRegV3((H+L)/2,
                 곡선회귀선, 곡선회귀기울기,
                 곡선회귀선_구분, 곡선회귀유효,
                 곡선회귀신뢰도, 곡선회귀잔차);

Var : MTF예측가격1(0), MTF예측가격2(0), MTF예측가격3(0),
      MTF예측방향1(0), MTF예측방향2(0), MTF예측방향3(0),
      MTF보정기울기(0), MTF기울기가속도(0), MTF예측변동성(0);

OSF_Mtf_LinRegPredictV4(
    곡선회귀선,
    곡선회귀기울기,
    곡선회귀유효,
    곡선회귀신뢰도,
    예측봉수1,
    예측봉수2,
    예측봉수3,
    MTF예측가격1,
    MTF예측가격2,
    MTF예측가격3,
    MTF예측방향1,
    MTF예측방향2,
    MTF예측방향3,
    MTF보정기울기,
    MTF기울기가속도,
    MTF예측변동성
);

// 회귀 잔차와 ATR을 결합한 5/10/15봉 예상 범위
Var : MTF기준오차(0),
      MTF범위1(0), MTF범위2(0), MTF범위3(0),
      MTF상단1(0), MTF하단1(0),
      MTF상단2(0), MTF하단2(0),
      MTF상단3(0), MTF하단3(0);

MTF기준오차 = Max(곡선회귀잔차, MTF예측변동성 * 0.25);

MTF범위1 = MTF기준오차 * SquareRoot(Max(1,예측봉수1));
MTF범위2 = MTF기준오차 * SquareRoot(Max(1,예측봉수2));
MTF범위3 = MTF기준오차 * SquareRoot(Max(1,예측봉수3));

MTF상단1 = MTF예측가격1 + MTF범위1;
MTF하단1 = MTF예측가격1 - MTF범위1;
MTF상단2 = MTF예측가격2 + MTF범위2;
MTF하단2 = MTF예측가격2 - MTF범위2;
MTF상단3 = MTF예측가격3 + MTF범위3;
MTF하단3 = MTF예측가격3 - MTF범위3;

// 운영 V2에서는 사용하지 않는 구형 곡선/지지저항 Plot과 계산을 제거
Var : 곡선회귀선_평탄(0), 곡선가격판정방향(0),
      회귀위치강도(0), 회귀위치단계(0), 회귀단계색상(0),
      회귀ADX값(0), 회귀상승DI(0), 회귀하락DI(0), 회귀ADX유효(0),
      회귀원DI차이(0), 회귀평활DI차이(0), 회귀DI방향유효(0),
      회귀선물일분봉여부(0), 회귀선물세션봉수(0), 회귀적용ADX기간(0), 회귀ADX표시유효(0);
곡선회귀선_평탄 = round(곡선회귀선/PriceScale,0)*PriceScale;
회귀위치강도 = 0;
회귀위치단계 = 0;
회귀단계색상 = RGB(130,130,130);

// 회귀위치와 색상은 하단 #우드스탁_미래곡선_ADXDI_V9_오실레이터와 동일하게 종가 기준으로 계산한다.
If 곡선회귀유효 == 0 Then 곡선가격판정방향 = 0;
If 곡선회귀유효 == 1 && L > 곡선회귀선_평탄 Then
    곡선가격판정방향 = 1;
If 곡선회귀유효 == 1 && H < 곡선회귀선_평탄 Then
    곡선가격판정방향 = -1;

// 평탄 회귀선과 캔들 위치를 상·하 3단계 색상으로 변환
If 곡선회귀유효 == 1 && MTF기준오차 > 0 Then
{
    회귀위치강도 = Max(-100,Min(100,((C-곡선회귀선_평탄)/MTF기준오차)*100));

    If C > 곡선회귀선_평탄 Then
    {
        If 회귀위치강도 >= 66 Then 회귀위치단계 = 100;
        Else If 회귀위치강도 >= 33 Then 회귀위치단계 = 66;
        Else 회귀위치단계 = 33;
    }
    Else If C < 곡선회귀선_평탄 Then
    {
        If 회귀위치강도 <= -66 Then 회귀위치단계 = -100;
        Else If 회귀위치강도 <= -33 Then 회귀위치단계 = -66;
        Else 회귀위치단계 = -33;
    }
    Else If CurrentBar > 1 Then
        회귀위치단계 = 회귀위치단계[1];

    If 회귀위치단계 == 100 Then 회귀단계색상 = RGB(220,0,0);
    If 회귀위치단계 == 66 Then 회귀단계색상 = RGB(255,100,70);
    If 회귀위치단계 == 33 Then 회귀단계색상 = RGB(255,185,185);
    If 회귀위치단계 == -100 Then 회귀단계색상 = RGB(0,0,180);
    If 회귀위치단계 == -66 Then 회귀단계색상 = RGB(60,130,255);
    If 회귀위치단계 == -33 Then 회귀단계색상 = RGB(180,210,255);
}

// V9: 메인 차트 회귀선에도 ADXDI_V9와 같은 3봉 평활 DI차이 색상 규칙을 적용한다.
회귀선물일분봉여부 = IFF(CodeCategory == 4 && DataCompress == 2 && BarInterval == 1,1,0);
If CurrentBar == 1 Then 회귀선물세션봉수 = 1;
Else If BDate <> BDate[1] Then 회귀선물세션봉수 = 1;
Else 회귀선물세션봉수 = 회귀선물세션봉수[1] + 1;

회귀적용ADX기간 = ADX기간;
If 회귀선물일분봉여부 == 1 Then 회귀적용ADX기간 = 2;
OSF_AutoSessionADXV1(회귀적용ADX기간,회귀ADX값,회귀상승DI,회귀하락DI,회귀ADX유효);
회귀ADX표시유효 = 회귀ADX유효;
If 회귀선물일분봉여부 == 1 && 회귀선물세션봉수 < 5 Then 회귀ADX표시유효 = 0;
회귀원DI차이 = 회귀상승DI - 회귀하락DI;
회귀평활DI차이 = Average(회귀원DI차이,Max(1,DI평활기간));
회귀DI방향유효 = IFF(회귀ADX표시유효 == 1 && Abs(회귀평활DI차이) >= DI차이최소,1,0);

If 곡선회귀유효 == 1 && 회귀ADX표시유효 == 1 Then
{
    // DI차이가 작으면 방향 불명확으로 회색 처리한다.
    If 회귀DI방향유효 == 0 Then
        회귀단계색상 = RGB(150,150,150);

    // DI 방향과 회귀단계가 반대일 때만 옅은 반대 계열로 표시한다.
    If 회귀DI방향유효 == 1 && 회귀평활DI차이 > 0 && 회귀위치단계 < 0 Then
        회귀단계색상 = RGB(255,190,190);
    If 회귀DI방향유효 == 1 && 회귀평활DI차이 < 0 && 회귀위치단계 > 0 Then
        회귀단계색상 = RGB(190,210,255);
}

//======================================================================
// 새 MTF 회귀예측 검증
// 과거 봉에서 계산한 예측값을 목표 봉에 Plot하여 실제 가격과 비교
//======================================================================
Var : MTF지난예측1(0), MTF지난예측2(0), MTF지난예측3(0),
      MTF지난색상1(0), MTF지난색상2(0), MTF지난색상3(0),
      MTF과거유효1(0), MTF과거유효2(0), MTF과거유효3(0),
      MTF과거신뢰1(0), MTF과거신뢰2(0), MTF과거신뢰3(0),
      MTF과거방향1(0), MTF과거방향2(0), MTF과거방향3(0),
      MTF세션번호(0), MTF세션최근봉(-1),
      MTF검증표시값(0), MTF검증색상2(0);

If DataCompress == 2 && BarInterval <= 30 &&
   DayIndex == 0 && Index != MTF세션최근봉 Then
{
    MTF세션번호 = MTF세션번호 + 1;
    MTF세션최근봉 = Index;
}

MTF지난예측1 = MTF예측가격1[예측봉수1];
MTF지난예측2 = MTF예측가격2[예측봉수2];
MTF지난예측3 = MTF예측가격3[예측봉수3];

MTF과거유효1 = 곡선회귀유효[예측봉수1];
MTF과거유효2 = 곡선회귀유효[예측봉수2];
MTF과거유효3 = 곡선회귀유효[예측봉수3];

MTF과거신뢰1 = 곡선회귀신뢰도[예측봉수1];
MTF과거신뢰2 = 곡선회귀신뢰도[예측봉수2];
MTF과거신뢰3 = 곡선회귀신뢰도[예측봉수3];

MTF과거방향1 = MTF예측방향1[예측봉수1];
MTF과거방향2 = MTF예측방향2[예측봉수2];
MTF과거방향3 = MTF예측방향3[예측봉수3];

// 30분 이하 분봉에서는 야간장/주간장을 포함해 세션을 넘긴 예측을 연결하지 않음
If DataCompress == 2 && BarInterval <= 30 && MTF세션번호 != MTF세션번호[예측봉수1] Then MTF과거유효1 = 0;
If DataCompress == 2 && BarInterval <= 30 && MTF세션번호 != MTF세션번호[예측봉수2] Then MTF과거유효2 = 0;
If DataCompress == 2 && BarInterval <= 30 && MTF세션번호 != MTF세션번호[예측봉수3] Then MTF과거유효3 = 0;

// 표시색은 방향으로 통일: 상승=빨강 계열, 하락=파랑 계열
// 신뢰도가 낮으면 같은 계열의 연한색, 방향이 없으면 회색

Var : 마켓중심가격(0),
      마켓상단가격1(0), 마켓상단가격2(0),

      마켓중심단계(0), 마켓중심색상(0),
      마켓세션봉수(0), 마켓계산봉수(0), 마켓j(0),
      마켓기준가격(0), 마켓누적거래량(0), 마켓누적가격거래량(0),
      마켓가중분산합(0), 마켓가중표준편차(0);

If DataCompress == 2 Then
{
    If CurrentBar == 1 Then
        마켓세션봉수 = 1;
    Else If DayIndex == 0 Then
        마켓세션봉수 = 1;
    Else
        마켓세션봉수 = 마켓세션봉수[1] + 1;

    마켓계산봉수 = Min(마켓계산기간,마켓세션봉수);
    마켓누적거래량 = 0;
    마켓누적가격거래량 = 0;

    For 마켓j = 0 To 마켓계산봉수-1
    {
        마켓기준가격 = (H[마켓j]+L[마켓j]+C[마켓j])/3;
        If V[마켓j] > 0 Then
        {
            마켓누적거래량 = 마켓누적거래량 + V[마켓j];
            마켓누적가격거래량 = 마켓누적가격거래량 + 마켓기준가격*V[마켓j];
        };
    };

    If 마켓누적거래량 > 0 && 마켓계산봉수 >= 2 Then
    {
        마켓중심가격 = 마켓누적가격거래량/마켓누적거래량;
        마켓가중분산합 = 0;
        For 마켓j = 0 To 마켓계산봉수-1
        {
            마켓기준가격 = (H[마켓j]+L[마켓j]+C[마켓j])/3;
            If V[마켓j] > 0 Then
                마켓가중분산합 = 마켓가중분산합 + (마켓기준가격-마켓중심가격)^2*V[마켓j];
        };
        마켓가중표준편차 = SquareRoot(마켓가중분산합/마켓누적거래량);
        마켓상단가격1 = 마켓중심가격 + 마켓가중표준편차*마켓밴드배수;
        마켓하단가격1 = 마켓중심가격 - 마켓가중표준편차*마켓밴드배수;
        마켓계산유효 = 1;
    };
}
