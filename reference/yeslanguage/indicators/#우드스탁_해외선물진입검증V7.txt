// #우드스탁_해외선물진입검증V7  (하단 지표, 1분봉 전용)  - 진입후보V10과 같은 후보 (WSF_FXEntryCandV7_CO)
// Plot13 S누적틱, Plot14 S승률(%), Plot15 S거래수 추가
// (아래는 V6 설명)
// #우드스탁_해외선물진입검증V6  (하단 지표, 1분봉 전용)  - 진입후보V9와 같은 후보 (WSF_FXEntryCandV6_CO, 횟수 제한 없음)
// (아래는 V5 설명)
// #우드스탁_해외선물진입검증V5  (하단 지표, 1분봉 전용)  - V4 + 구조해제포함(1), 진입후보V8과 같은 후보 (WSF_FXEntryCandV5_CO)
// (아래는 V4 설명)
// #우드스탁_해외선물진입검증V4  (하단 지표, 1분봉 전용)  - 작업 플랜 6단계: 청산·손익 검증
// V3 + 스윙 구조 이탈 후보(등급 4, WSF_FXEntryCandV4_CO). 진입후보V7과 같은 후보. 구조사용=0 이면 V3과 같음.
// Plot10 구조누적틱, Plot11 구조승률(%), Plot12 구조거래수 추가
// V2 + WSF_FXEntryCandV3_CO 입력(해제A사용·재돌파허용봉수·B보완사용·B사용). 새 입력을 0(B사용은 1)으로 두면 V2와 같음.
// V1 + 진입 시간대(기본 20:00~02:00)·세션당 최대진입횟수(기본 2). 청산 규칙은 V1과 같음(시간대 밖에서도 보유 중이면 규칙대로 청산).
// 판정방식(1~4 = 진입후보 V1~V4 규칙)의 후보로 가상 매매를 하고 손익을 틱으로 누적한다.
// 매매 규칙 (한 번에 1포지션)
//   진입: 후보 봉 종가
//   청산: 손절틱 / 목표틱 도달(같은 봉에서 둘 다 닿으면 손절로 처리) / 최대보유봉수 경과 시 종가
//         / 반대신호청산=1이면 반대 후보 봉 종가(그 봉에서 반대로 새 진입) / 세션 경계에서 직전 봉 종가
//   손익틱 = (청산가-진입가)/틱 x 방향 - 비용틱(수수료+슬리피지, 왕복)
// 주의: 봉 안의 실제 체결 순서는 알 수 없어 근사 계산이다. 손절 우선 처리로 보수적으로 잡았다.
// Plot1 전체누적틱, Plot2 A누적틱, Plot3 B누적틱, Plot4 C누적틱 (같은 단위라 등급별 비교 가능)
// Plot5 거래손익틱: 청산 봉에 그 거래의 손익 (막대 권장)
// Plot6 승률(%), Plot7 거래수, Plot8 A승률(%), Plot9 A거래수 : 제목줄 값 읽기용 (차트에서 꺼도 됨)
Input : 판정방식(4),
        예측변수(10),
        가치영역배수(1.5),
        최소봉수(5),
        ADX기간(14),
        ADX추세기준(20),
        ADX강세기준(35),
        매물대기준모드(1),
        폭계산봉수(30),
        비율기준봉수(120),
        좁음기준(40),
        단계1봉수(15),
        방향확정봉수(5),
        확정봉판정(0),
        해제유효봉수(10),
        등급C사용(-1),
        돌파여유틱(-1),
        돌파통합최소(50),
        돌파확인봉수(1),
        DI일치필수(1),
        확장확인봉수(5),
        진입시작시각(200000),
        진입종료시각(20000),
        해제A사용(1),
        재돌파허용봉수(3),
        B보완사용(1),
        B사용(1),
        구조사용(1),
        구조폭기준(50),
        구조유효봉수(10),
        구조매수사용(1),
        구조해제포함(1),
        S사용(1),
        S구조폭기준(50),
        S압축봉수(30),
        S폭비율기준(50),
        예측봉수1(5),
        예측봉수2(10),
        예측봉수3(15),
        예측봉수4(30),
        예측봉수5(60),
        최소신뢰도(0.4),
        지속봉수(5),
        마켓계산기간(20),
        최소전환유지봉수(3),
        스윙연결(0),
        기세무시틱(1),
        손절틱(12),
        목표틱(24),
        최대보유봉수(30),
        비용틱(1),
        반대신호청산(1);
Var : FX세션키(0), FX초기화(0), FX세션봉수(0),
      후보방향(0), 후보등급(0), 통합상태(0),
      보유방향(0), 보유등급(0), 보유가격(0), 보유봉수(0),
      손절가(0), 목표가(0), 청산발생(0), 청산가(0), 손익틱(0), 청산등급(0),
      누적전체(0), 누적A(0), 누적B(0), 누적C(0), 누적S(0), S거래수(0), S승수(0), S승률(0), 누적SS(0), SS거래수(0), SS승수(0), SS승률(0),
      거래수(0), 승수(0), A거래수(0), A승수(0), 승률(0), A승률(0);

FX세션키 = WSF_FXSessionKeyV1(sDate,sTime);
FX초기화 = 0;
If CurrentBar == 1 Then FX초기화 = 1;
Else If FX세션키 != FX세션키[1] Then FX초기화 = 1;
If FX초기화 == 1 Then FX세션봉수 = 1;
Else FX세션봉수 = FX세션봉수[1]+1;

If DataCompress == 2 && BarInterval == 1 Then
{
    WSF_FXEntryCandV7_CO(FX초기화,FX세션봉수,판정방식,
        예측변수,가치영역배수,최소봉수,ADX기간,ADX추세기준,ADX강세기준,매물대기준모드,
        폭계산봉수,비율기준봉수,좁음기준,단계1봉수,방향확정봉수,확정봉판정,해제유효봉수,
        등급C사용,돌파여유틱,돌파통합최소,돌파확인봉수,DI일치필수,확장확인봉수,
        진입시작시각,진입종료시각,
        해제A사용,재돌파허용봉수,B보완사용,B사용,
        구조사용,구조폭기준,구조유효봉수,구조매수사용,구조해제포함,
        S사용,S구조폭기준,S압축봉수,S폭비율기준,
        예측봉수1,예측봉수2,예측봉수3,예측봉수4,예측봉수5,
        최소신뢰도,지속봉수,마켓계산기간,최소전환유지봉수,스윙연결,기세무시틱,
        후보방향,후보등급,통합상태);

    // 상태 이어받기 (같은 봉 재평가는 직전 봉 상태에서 시작)
    If CurrentBar > 1 Then
    {
        보유방향 = 보유방향[1]; 보유등급 = 보유등급[1]; 보유가격 = 보유가격[1]; 보유봉수 = 보유봉수[1];
        누적전체 = 누적전체[1]; 누적A = 누적A[1]; 누적B = 누적B[1]; 누적C = 누적C[1];
        누적S = 누적S[1]; S거래수 = S거래수[1]; S승수 = S승수[1];
        누적SS = 누적SS[1]; SS거래수 = SS거래수[1]; SS승수 = SS승수[1];
        거래수 = 거래수[1]; 승수 = 승수[1]; A거래수 = A거래수[1]; A승수 = A승수[1];
    };

    // 청산 판정
    청산발생 = 0; 청산가 = 0; 손익틱 = 0; 청산등급 = 0;
    If 보유방향 != 0 Then
    {
        If FX초기화 == 1 Then
        {
            청산발생 = 1; 청산가 = C[1];
        }
        Else
        {
            보유봉수 = 보유봉수 + 1;
            손절가 = 보유가격 - 보유방향*손절틱*PriceScale;
            목표가 = 보유가격 + 보유방향*목표틱*PriceScale;
            If 보유방향 == 1 Then
            {
                If L <= 손절가 Then { 청산발생 = 1; 청산가 = 손절가; }
                Else If H >= 목표가 Then { 청산발생 = 1; 청산가 = 목표가; };
            }
            Else
            {
                If H >= 손절가 Then { 청산발생 = 1; 청산가 = 손절가; }
                Else If L <= 목표가 Then { 청산발생 = 1; 청산가 = 목표가; };
            };
            If 청산발생 == 0 && 보유봉수 >= 최대보유봉수 Then
            {
                청산발생 = 1; 청산가 = C;
            };
            If 청산발생 == 0 && 반대신호청산 == 1 && 후보방향 == -보유방향 Then
            {
                청산발생 = 1; 청산가 = C;
            };
        };
        If 청산발생 == 1 Then
        {
            손익틱 = (청산가-보유가격)/PriceScale*보유방향 - 비용틱;
            청산등급 = 보유등급;
            누적전체 = 누적전체 + 손익틱;
            If 보유등급 == 3 Then 누적A = 누적A + 손익틱;
            If 보유등급 == 2 Then 누적B = 누적B + 손익틱;
            If 보유등급 == 1 Then 누적C = 누적C + 손익틱;
            If 보유등급 == 5 Then
            {
                누적SS = 누적SS + 손익틱;
                SS거래수 = SS거래수 + 1;
                If 손익틱 > 0 Then SS승수 = SS승수 + 1;
            };
            If 보유등급 == 4 Then
            {
                누적S = 누적S + 손익틱;
                S거래수 = S거래수 + 1;
                If 손익틱 > 0 Then S승수 = S승수 + 1;
            };
            거래수 = 거래수 + 1;
            If 손익틱 > 0 Then 승수 = 승수 + 1;
            If 보유등급 == 3 Then
            {
                A거래수 = A거래수 + 1;
                If 손익틱 > 0 Then A승수 = A승수 + 1;
            };
            보유방향 = 0; 보유등급 = 0; 보유가격 = 0; 보유봉수 = 0;
        };
    };

    // 진입
    If 보유방향 == 0 && 후보방향 != 0 && FX초기화 == 0 Then
    {
        보유방향 = 후보방향; 보유등급 = 후보등급; 보유가격 = C; 보유봉수 = 0;
    };

    승률 = 0; A승률 = 0;
    If 거래수 > 0 Then 승률 = 승수/거래수*100;
    If A거래수 > 0 Then A승률 = A승수/A거래수*100;
    S승률 = 0;
    If S거래수 > 0 Then S승률 = S승수/S거래수*100;
    SS승률 = 0;
    If SS거래수 > 0 Then SS승률 = SS승수/SS거래수*100;

    Plot1(누적전체,"전체누적틱",RGB(60,60,60),Def,2);
    Plot2(누적A,"A누적틱",RGB(220,0,0),Def,1);
    Plot3(누적B,"B누적틱",RGB(255,140,0),Def,1);
    Plot4(누적C,"C누적틱",RGB(150,150,150),Def,1);
    If 청산발생 == 1 Then Plot5(손익틱,"거래손익틱",IFF(손익틱 > 0,RGB(0,160,0),RGB(220,0,0)),Def,3);
    Else NoPlot(5);
    Plot6(승률,"승률",RGB(0,120,200),Def,0);
    Plot7(거래수,"거래수",RGB(120,120,120),Def,0);
    Plot8(A승률,"A승률",RGB(200,0,120),Def,0);
    Plot9(A거래수,"A거래수",RGB(200,120,160),Def,0);
    Plot10(누적S,"구조누적틱",RGB(0,150,150),Def,1);
    Plot11(S승률,"구조승률",RGB(0,120,120),Def,0);
    Plot12(S거래수,"구조거래수",RGB(100,170,170),Def,0);
    Plot13(누적SS,"S누적틱",RGB(170,0,0),Def,2);
    Plot14(SS승률,"S승률",RGB(140,0,0),Def,0);
    Plot15(SS거래수,"S거래수",RGB(190,90,90),Def,0);
}
Else
{
    NoPlot(1); NoPlot(2); NoPlot(3); NoPlot(4); NoPlot(5);
    NoPlot(6); NoPlot(7); NoPlot(8); NoPlot(9);
    NoPlot(10); NoPlot(11); NoPlot(12);
    NoPlot(13); NoPlot(14); NoPlot(15);
};
