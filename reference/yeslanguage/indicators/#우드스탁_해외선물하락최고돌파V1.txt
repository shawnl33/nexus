// #우드스탁_해외선물하락최고돌파V2  (가격 차트 겹침 지표, 1분봉 기준)  - V1 + 조건 완화 단계
//   완화단계 = 1 : V1과 같음 (매수 기준선 핵심지난상승_618, 매도 기준선 핵심지난하락_382)
//   완화단계 = 2 : 매수 기준선 핵심지난상승_500, 매도 기준선 핵심지난하락_500 (기본)
//   완화단계 = 3 : 매수 기준선 핵심지난상승_382, 매도 기준선 핵심지난하락_618 (가장 느슨)
//   매수: 기준선 < 핵심지난하락최저가 && 핵심지난하락최고가 >= 핵심지난상승최고가 → 하락최고가 첫 종가 돌파
//   매도: 기준선 > 핵심지난상승최고가 && 핵심지난상승최저가 <= 핵심지난하락최저가 → 상승최저가 첫 종가 이탈
// (아래는 V1 설명)
// #우드스탁_해외선물하락최고돌파V1  (가격 차트 겹침 지표, 1분봉 기준)
// 매수 표시: 직전 봉 기준으로
//   핵심지난상승_618(차트 이름 "핵심지난상승_되돌림382") < 핵심지난하락최저가   (상승 스윙이 약해 하락 스윙 바닥 아래에 머묾)
//   && 핵심지난하락최고가 >= 핵심지난상승최고가                                  (하락 스윙 고점이 상승 스윙 고점 이상)
//   인 상황에서 종가가 핵심지난하락최고가를 위로 처음 돌파한 봉 (C > 하락최고[1] && C[1] <= 하락최고[1])
// 매도 표시 (대칭): 직전 봉 기준으로
//   핵심지난하락_382(하락 스윙 38.2% 반등 자리) > 핵심지난상승최고가
//   && 핵심지난상승최저가 <= 핵심지난하락최저가
//   인 상황에서 종가가 핵심지난상승최저가를 아래로 처음 이탈한 봉
// 값은 #우드스탁_해외선물미래곡선과 같은 WSF_FXFutureValuesV3_CO, 같은 기본 입력.
// Plot1 매수돌파: 저가 아래 금색 점 (크기 6)   Plot2 매도이탈: 고가 위 청록 점 (크기 6)
// Plot3 매수조건선: 매수 조건이 살아 있는 동안 핵심지난하락최고가 (금색 가는 선)
// Plot4 매도조건선: 매도 조건이 살아 있는 동안 핵심지난상승최저가 (청록 가는 선)
Input : 예측변수(10),
        예측봉수1(5),
        예측봉수2(10),
        예측봉수3(15),
        예측봉수4(30),
        예측봉수5(60),
        최소신뢰도(0.4),
        지속봉수(5),
        마켓계산기간(20),
        최소전환유지봉수(3),
        스윙연결(0),
        기세무시틱(1),
        표시여유틱(4),
        완화단계(2);
Var : 단계화_1분_통합(0), 곡선회귀선_평탄(0), 마켓중심가격(0),
      지속저장목표1(0), 지속저장목표2(0), 지속저장목표3(0), 지속저장목표4(0), 지속저장목표5(0),
      핵심지난상승최고가(0), 핵심지난상승최저가(0), 핵심지난상승_382(0), 핵심지난상승_500(0), 핵심지난상승_618(0),
      핵심지난하락최고가(0), 핵심지난하락최저가(0), 핵심지난하락_382(0), 핵심지난하락_500(0), 핵심지난하락_618(0),
      함수계산가능(0), 값유효(0), 매수조건(0), 매도조건(0), 매수기준선(0), 매도기준선(0);

함수계산가능 = WSF_FXFutureValuesV3_CO(
    예측변수,예측봉수1,예측봉수2,예측봉수3,예측봉수4,예측봉수5,
    최소신뢰도,지속봉수,마켓계산기간,최소전환유지봉수,
    단계화_1분_통합,곡선회귀선_평탄,마켓중심가격,
    지속저장목표1,지속저장목표2,지속저장목표3,지속저장목표4,지속저장목표5,
    핵심지난상승최고가,핵심지난상승최저가,핵심지난상승_382,핵심지난상승_500,핵심지난상승_618,
    핵심지난하락최고가,핵심지난하락최저가,핵심지난하락_382,핵심지난하락_500,핵심지난하락_618,
    스윙연결,기세무시틱);

값유효 = 0;
If 함수계산가능 == 1 && 핵심지난상승최고가 > 0 && 핵심지난상승최저가 > 0 &&
   핵심지난하락최고가 > 0 && 핵심지난하락최저가 > 0 Then 값유효 = 1;

// 조건 (돌파 전 상태)
매수조건 = 0; 매도조건 = 0;
If 값유효 == 1 Then
{
    매수기준선 = IFF(완화단계 >= 3,핵심지난상승_382,IFF(완화단계 == 2,핵심지난상승_500,핵심지난상승_618));
    매도기준선 = IFF(완화단계 >= 3,핵심지난하락_618,IFF(완화단계 == 2,핵심지난하락_500,핵심지난하락_382));
    If 매수기준선 < 핵심지난하락최저가 && 핵심지난하락최고가 >= 핵심지난상승최고가 Then 매수조건 = 1;
    If 매도기준선 > 핵심지난상승최고가 && 핵심지난상승최저가 <= 핵심지난하락최저가 Then 매도조건 = 1;
};

If 매수조건 == 1 Then Plot3(핵심지난하락최고가,"매수조건선",RGB(230,180,0),Def,1);
Else NoPlot(3);
If 매도조건 == 1 Then Plot4(핵심지난상승최저가,"매도조건선",RGB(0,170,170),Def,1);
Else NoPlot(4);

// 돌파 봉: 직전 봉 조건 성립 && 직전 봉 기준선을 종가로 처음 넘음
If CurrentBar > 1 && 매수조건[1] == 1 &&
   C > 핵심지난하락최고가[1] && C[1] <= 핵심지난하락최고가[1] Then
    Plot1(L-표시여유틱*PriceScale,"매수돌파",RGB(255,190,0),Def,6);
Else NoPlot(1);

If CurrentBar > 1 && 매도조건[1] == 1 &&
   C < 핵심지난상승최저가[1] && C[1] >= 핵심지난상승최저가[1] Then
    Plot2(H+표시여유틱*PriceScale,"매도이탈",RGB(0,200,200),Def,6);
Else NoPlot(2);
