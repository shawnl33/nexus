//======================================================================
// #CLV거래량흐름결합오실레이터V1_해외
// OSF_ClvVolFlowV1 확인용 오실레이터 (#호가방향오실레이터V5와 동일 표시 구조)
// CLV(빠른 트리거) + 거래량흐름(느린 필터) 결합 결과를 표시한다.
//======================================================================
Input : 관심기준(15),
        강세기준(35),
        부호반전(0),
        계산기간(20),
        기울기표시(1),
        기울기평활기간(3),
        약기울기기준(2),
        강기울기기준(5);
Var : 결합점수(0), 결합기울기(0), 결합상태(0), 결합유효(0),
      평활기울기(0), 점수색상(0), 기울기색상(0);

OSF_ClvVolFlowV1(
    관심기준,강세기준,부호반전,계산기간,
    결합점수,결합기울기,결합상태,결합유효
);

평활기울기 = Average(결합기울기,Max(1,기울기평활기간));
점수색상 = RGB(170,170,170);
기울기색상 = RGB(150,150,150);

If 결합점수 > 0 Then
{
    If Abs(평활기울기) >= 강기울기기준 Then
        점수색상 = RGB(220,0,0);
    Else If Abs(평활기울기) >= 약기울기기준 Then
        점수색상 = RGB(255,100,70);
    Else
        점수색상 = RGB(255,185,185);
}
Else If 결합점수 < 0 Then
{
    If Abs(평활기울기) >= 강기울기기준 Then
        점수색상 = RGB(0,0,180);
    Else If Abs(평활기울기) >= 약기울기기준 Then
        점수색상 = RGB(60,130,255);
    Else
        점수색상 = RGB(180,210,255);
}
Else
{
    점수색상 = RGB(150,150,150);
};

기울기색상 = IFF(평활기울기 > 0,RGB(255,100,100),
             IFF(평활기울기 < 0,RGB(100,135,255),
			 RGB(150,150,150)));

If 결합유효 == 1 Then
{
    Plot1(결합점수,"CLV+거래량흐름",점수색상,Def,4);
    If 기울기표시 == 1 Then
        Plot2(평활기울기,"평활결합기울기",기울기색상,Def,2);
    Else
        NoPlot(2);
}
Else
{
    NoPlot(1); NoPlot(2);
};

Plot3(0,"영선",RGB(90,90,90),Def,0);
Plot4(관심기준,"매수관심",RGB(255,170,170),Def,0);
Plot5(-관심기준,"매도관심",RGB(170,190,255),Def,0);
Plot6(강세기준,"매수강세",RGB(255,0,0),Def,0);
Plot7(-강세기준,"매도강세",RGB(0,0,255),Def,0);
