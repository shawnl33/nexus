//======================================================================
// WSF_ClvPressureV1
// 종가위치(Close Location Value) 기반 봉 내 매수/매도 압력 함수
// 국내선물을 포함해 어떤 종목에서도
// OHLC만으로 방향성을 추정한다.
//
// 계산식: CLV = ((종가-저가) - (고가-종가)) / (고가-저가) * 100  (-100 ~ +100)
//   - 종가가 고가 쪽에 붙을수록 +100에 가까워짐(매수압력), 저가 쪽이면 -100(매도압력)
// WSF_OrderBookDirectionV2와 동일하게 원시값/3봉평균/5봉평균을 60:25:15로 블렌딩해
// 같은 인터페이스(점수/기울기/상태/유효)로 출력한다.
//======================================================================
Input : 관심기준입력(Numeric),
        강세기준입력(Numeric),
        부호반전입력(Numeric),
        압력점수(NumericRef),
        압력기울기3(NumericRef),
        압력상태(NumericRef),
        압력유효(NumericRef);

Var : 봉범위(0),
      원시CLV(0),
      방향평균3(0),
      방향평균5(0),
      계산압력(0),
      계산기울기3(0),
      계산상태(0),
      관심기준값(0),
      강세기준값(0);

관심기준값 = Abs(관심기준입력);
If 관심기준값 < 1 Then
    관심기준값 = 1;

강세기준값 = Abs(강세기준입력);
If 강세기준값 < 관심기준값 Then
    강세기준값 = 관심기준값;

봉범위 = H - L;

If 봉범위 > 0 Then
    원시CLV = ((C - L) - (H - C)) / 봉범위 * 100;
Else
    원시CLV = 0;

If 부호반전입력 == 1 Then
    원시CLV = -원시CLV;

If CurrentBar >= 3 Then
    방향평균3 = (원시CLV + 원시CLV[1] + 원시CLV[2]) / 3;
Else
    방향평균3 = 원시CLV;

If CurrentBar >= 5 Then
    방향평균5 = (원시CLV + 원시CLV[1] + 원시CLV[2] + 원시CLV[3] + 원시CLV[4]) / 5;
Else
    방향평균5 = 방향평균3;

계산압력 = (원시CLV * 60 + 방향평균3 * 25 + 방향평균5 * 15) / 100;

If CurrentBar >= 4 Then
    계산기울기3 = 계산압력 - 계산압력[3];
Else
    계산기울기3 = 0;

If 계산기울기3 > 100 Then
    계산기울기3 = 100;
If 계산기울기3 < -100 Then
    계산기울기3 = -100;

계산상태 =
    IFF(계산압력 >= 강세기준값, 2,
    IFF(계산압력 >= 관심기준값, 1,
    IFF(계산압력 <= -강세기준값, -2,
    IFF(계산압력 <= -관심기준값, -1, 0))));

압력점수 = 계산압력;
압력기울기3 = 계산기울기3;
압력상태 = 계산상태;
압력유효 = 1;

Value1 = 1;
WSF_ClvPressureV1 = Value1;
