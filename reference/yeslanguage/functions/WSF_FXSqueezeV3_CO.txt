// WSF_FXSqueezeV3_CO  (_CO = Claude Opus 작성)
// V2 + 돌파 당시 박스 상하단 출력(돌파 확인용). V1·V2는 보존.
// V1 + 박스 돌파(빠른 해제).
// - 박스: 압축이 단계1봉수 이상 이어지는 동안 직전 봉의 최근N봉 매물대 1차 상하단을 기억
// - 돌파: 종가가 박스 상단+돌파여유틱 위면 +1, 하단-돌파여유틱 아래면 -1. 폭이 넓어지기를 기다리지 않음
//         압축이 풀린 뒤에도 방향확정봉수 동안은 마지막 박스로 돌파 판정. 박스(압축 1회)당 한 번만
// #우드스탁_해외선물매물대압축지속V3의 계산부를 함수로 분리 (표시 없음).
// - 압축: 최근 폭계산봉수 매물대(WSF_FXSessionProfileV2_CO) 1차 폭이 최근 비율기준봉수 최대 폭 대비 좁음기준% 미만
// - 해제: 단계1봉수 이상 이어진 압축이 풀리는 봉. 종가가 압축 당시 박스 1차 상단 위 +1, 하단 아래 -1, 사이 0
// - 지연확정: 해제 방향 0이면 방향확정봉수 이내 첫 박스 이탈 방향 (+1/-1)
// - 확정봉판정 = 1이면 해제·확정을 직전 완료봉 기준으로 판정
// 반환: 이번 봉 압축해제 여부 (1/0)
Input : 세션초기화(Numeric),
        세션봉수입력(Numeric),
        가치영역배수(Numeric),
        최소봉수(Numeric),
        폭계산봉수(Numeric),
        비율기준봉수(Numeric),
        좁음기준(Numeric),
        단계1봉수(Numeric),
        방향확정봉수(Numeric),
        확정봉판정(Numeric),
        돌파여유틱(Numeric),
        압축지속봉수출력(NumericRef),
        폭비율출력(NumericRef),
        즉시해제방향출력(NumericRef),
        지연확정방향출력(NumericRef),
        해제길이출력(NumericRef),
        박스돌파방향출력(NumericRef),
        박스돌파길이출력(NumericRef),
        돌파박스상단출력(NumericRef),
        돌파박스하단출력(NumericRef);
Var : 매물대중심(0), 매물대상단1(0), 매물대하단1(0), 매물대상단2(0), 매물대하단2(0),
      매물대5상태(0), 매물대유효(0),
      폭(0), 최대폭(0), 폭비율(0), 기준창(0), j(0),
      좁음(0), 압축지속봉수(0),
      압축해제(0), 해제방향(0), 해제길이(0), 해제상단기준(0), 해제하단기준(0),
      대기활성(0), 대기상단(0), 대기하단(0), 대기경과(0), 대기길이(0),
      확정방향(0), 판정가격(0),
      박스활성(0), 박스상단(0), 박스하단(0), 박스경과(0), 박스길이(0), 박스완료(0),
      돌파원방향(0), 돌파원길이(0), 돌파여유(0), 돌파원상단(0), 돌파원하단(0);

WSF_FXSessionProfileV2_CO(세션초기화,가치영역배수,최소봉수,폭계산봉수,
    매물대중심,매물대상단1,매물대하단1,매물대상단2,매물대하단2,
    매물대5상태,매물대유효);

폭 = 0;
If 매물대유효 == 1 Then 폭 = 매물대상단1-매물대하단1;
기준창 = Min(Max(1,Round(비율기준봉수,0)),Max(1,세션봉수입력));
최대폭 = 0;
For j = 0 To 기준창-1 { 최대폭 = Max(최대폭,폭[j]); };
폭비율 = 0;
If 최대폭 > 0 Then 폭비율 = 폭/최대폭*100;

좁음 = 0;
If 매물대유효 == 1 && 최대폭 > 0 && 세션봉수입력 >= 폭계산봉수 && 폭비율 < 좁음기준 Then 좁음 = 1;

If 세션초기화 == 1 Then 압축지속봉수 = 좁음;
Else If 좁음 == 1 Then 압축지속봉수 = 압축지속봉수[1]+1;
Else 압축지속봉수 = 0;

압축해제 = 0; 해제방향 = 0; 해제길이 = 0;
If 확정봉판정 == 1 Then
{
    If CurrentBar > 2 && 세션초기화 == 0 && 세션초기화[1] == 0 &&
       좁음[1] == 0 && 압축지속봉수[2] >= 단계1봉수 Then
    {
        압축해제 = 1;
        해제길이 = 압축지속봉수[2];
        해제상단기준 = 매물대상단1[2]; 해제하단기준 = 매물대하단1[2];
        If C[1] > 해제상단기준 Then 해제방향 = 1;
        Else If C[1] < 해제하단기준 Then 해제방향 = -1;
    };
}
Else
{
    If CurrentBar > 1 && 세션초기화 == 0 && 좁음 == 0 && 압축지속봉수[1] >= 단계1봉수 Then
    {
        압축해제 = 1;
        해제길이 = 압축지속봉수[1];
        해제상단기준 = 매물대상단1[1]; 해제하단기준 = 매물대하단1[1];
        If C > 해제상단기준 Then 해제방향 = 1;
        Else If C < 해제하단기준 Then 해제방향 = -1;
    };
};

// 방향 확정 대기. 같은 봉 재평가는 직전 봉 상태에서 시작.
If CurrentBar > 1 Then
{
    대기활성 = 대기활성[1]; 대기상단 = 대기상단[1]; 대기하단 = 대기하단[1];
    대기경과 = 대기경과[1]; 대기길이 = 대기길이[1];
};
확정방향 = 0;
If 세션초기화 == 1 Then 대기활성 = 0;
판정가격 = C;
If 확정봉판정 == 1 Then 판정가격 = C[1];
If 대기활성 == 1 Then
{
    대기경과 = 대기경과 + 1;
    If 판정가격 > 대기상단 Then 확정방향 = 1;
    Else If 판정가격 < 대기하단 Then 확정방향 = -1;
    If 확정방향 != 0 || 대기경과 >= 방향확정봉수 Then 대기활성 = 0;
};
If 압축해제 == 1 && 해제방향 == 0 && 방향확정봉수 > 0 Then
{
    대기활성 = 1;
    대기상단 = 해제상단기준; 대기하단 = 해제하단기준;
    대기경과 = 0;
    대기길이 = 해제길이;
};

// 박스 돌파. 같은 봉 재평가는 직전 봉 상태에서 시작.
If CurrentBar > 1 Then
{
    박스활성 = 박스활성[1]; 박스상단 = 박스상단[1]; 박스하단 = 박스하단[1];
    박스경과 = 박스경과[1]; 박스길이 = 박스길이[1]; 박스완료 = 박스완료[1];
};
If 세션초기화 == 1 Then 박스활성 = 0;
돌파원방향 = 0; 돌파원길이 = 0; 돌파원상단 = 0; 돌파원하단 = 0;
돌파여유 = Max(0,돌파여유틱)*PriceScale;
// 1) 직전 봉까지의 박스로 돌파 판정
If 박스활성 == 1 && 박스완료 == 0 && 세션초기화 == 0 Then
{
    If C > 박스상단 + 돌파여유 Then 돌파원방향 = 1;
    Else If C < 박스하단 - 돌파여유 Then 돌파원방향 = -1;
    If 돌파원방향 != 0 Then
    {
        돌파원길이 = 박스길이;
        돌파원상단 = 박스상단; 돌파원하단 = 박스하단;
        박스완료 = 1;
        박스활성 = 0;
    };
};
// 2) 이번 봉 기준으로 박스 갱신
If 압축지속봉수 >= 단계1봉수 Then
{
    If 압축지속봉수 == 단계1봉수 || 박스활성 == 0 Then
    {
        // 새 압축 구간 시작 (돌파가 이미 난 압축 구간이면 다시 열지 않음)
        If 압축지속봉수 == 단계1봉수 || 박스완료 == 0 Then
        {
            박스활성 = 1; 박스완료 = 0;
        };
    };
    If 박스활성 == 1 Then
    {
        박스상단 = 매물대상단1; 박스하단 = 매물대하단1;
        박스경과 = 0; 박스길이 = 압축지속봉수;
    };
}
Else If 박스활성 == 1 Then
{
    박스경과 = 박스경과 + 1;
    If 박스경과 > 방향확정봉수 Then 박스활성 = 0;
};
If 확정봉판정 == 1 Then
{
    박스돌파방향출력 = 0; 박스돌파길이출력 = 0; 돌파박스상단출력 = 0; 돌파박스하단출력 = 0;
    If CurrentBar > 1 && 세션초기화 == 0 Then
    {
        박스돌파방향출력 = 돌파원방향[1]; 박스돌파길이출력 = 돌파원길이[1];
        돌파박스상단출력 = 돌파원상단[1]; 돌파박스하단출력 = 돌파원하단[1];
    };
}
Else
{
    박스돌파방향출력 = 돌파원방향; 박스돌파길이출력 = 돌파원길이;
    돌파박스상단출력 = 돌파원상단; 돌파박스하단출력 = 돌파원하단;
};

압축지속봉수출력 = 압축지속봉수;
폭비율출력 = 폭비율;
즉시해제방향출력 = 해제방향;
지연확정방향출력 = 확정방향;
해제길이출력 = 해제길이;
If 확정방향 != 0 Then 해제길이출력 = 대기길이;

Value1 = 압축해제;
WSF_FXSqueezeV3_CO = Value1;
