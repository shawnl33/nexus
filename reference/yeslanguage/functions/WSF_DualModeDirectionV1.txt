Input : 회귀방향입력(Numeric),
        호가방향입력(Numeric),
        호가유효입력(Numeric),
        호가사용입력(Numeric),
        운용모드출력(NumericRef),
        최종방향출력(NumericRef),
        진입허용출력(NumericRef),
        필터상태출력(NumericRef);

Var : 회귀방향값(0),
      호가방향값(0);

// 입력값을 -1, 0, 1로 정규화한다.
회귀방향값 = IFF(회귀방향입력 > 0, 1,
             IFF(회귀방향입력 < 0, -1, 0));

호가방향값 = IFF(호가방향입력 > 0, 1,
             IFF(호가방향입력 < 0, -1, 0));

// 기본값: 미래곡선 회귀방향은 항상 유지한다.
운용모드출력 = 0;
최종방향출력 = 회귀방향값;
진입허용출력 = 0;
필터상태출력 = 0;

// 일봉: 미래곡선 회귀방향만 사용하는 스윙 모드
If DataCompress == 3 Then
{
    운용모드출력 = 2;
    최종방향출력 = 회귀방향값;

    If 회귀방향값 <> 0 Then
        진입허용출력 = 1;
}
Else
{
    // 1분봉: 미래곡선이 방향, 호가는 진입 필터
    If DataCompress == 2 && BarInterval == 1 Then
    {
        운용모드출력 = 1;
        최종방향출력 = 회귀방향값;

        If 회귀방향값 <> 0 Then
        {
            // 호가 필터를 끈 경우에는 미래곡선 방향만으로 진입을 허용한다.
            If 호가사용입력 <> 1 Then
            {
                진입허용출력 = 1;
            }
            Else
            {
                // 호가 필터를 켠 경우 유효한 호가 방향이 회귀 방향과
                // 일치할 때만 1분봉 진입 표시를 허용한다.
                If 호가유효입력 == 1 Then
                {
                    If 호가방향값 == 회귀방향값 Then
                    {
                        필터상태출력 = 1;
                        진입허용출력 = 1;
                    }
                    Else
                    {
                        If 호가방향값 == 0 Then
                            필터상태출력 = 0;
                        Else
                            필터상태출력 = -1;
                    }
                }
            }
        }
    }
}
