Input : 기간입력(Numeric),
        ADX출력(NumericRef),
        상승DI출력(NumericRef),
        하락DI출력(NumericRef),
        ADX유효출력(NumericRef);

Var : 기간값(0),
      계산봉수(0),
      전일종가(0),
      상승이동(0),
      하락이동(0),
      TR값(0),
      상승DM(0),
      하락DM(0),
      평활TR(0),
      평활상승DM(0),
      평활하락DM(0),
      상승DI값(0),
      하락DI값(0),
      DX값(0),
      ADX값(0);

기간값 = Abs(기간입력);

If 기간값 < 2 Then
    기간값 = 2;

/* 일봉에서는 BDate 변경 초기화를 사용하면 안 된다.
   일봉의 각 봉마다 날짜가 달라 매 봉 초기화되기 때문이다. */
If CurrentBar == 1 Then
{
    계산봉수 = 1;
    전일종가 = C;

    TR값 = H - L;
    상승DM = 0;
    하락DM = 0;

    평활TR = TR값;
    평활상승DM = 0;
    평활하락DM = 0;

    상승DI값 = 0;
    하락DI값 = 0;
    DX값 = 0;
    ADX값 = 0;
}
Else
{
    계산봉수 = 계산봉수[1] + 1;
    전일종가 = C[1];

    TR값 = Max(H - L,
             Max(Abs(H - 전일종가),
                 Abs(L - 전일종가)));

    상승이동 = H - H[1];
    하락이동 = L[1] - L;

    상승DM = IFF(상승이동 > 하락이동 && 상승이동 > 0,
                 상승이동, 0);

    하락DM = IFF(하락이동 > 상승이동 && 하락이동 > 0,
                 하락이동, 0);

    If 계산봉수 <= 기간값 Then
    {
        평활TR = 평활TR[1] + TR값;
        평활상승DM = 평활상승DM[1] + 상승DM;
        평활하락DM = 평활하락DM[1] + 하락DM;
    }
    Else
    {
        평활TR = 평활TR[1] - 평활TR[1] / 기간값 + TR값;
        평활상승DM = 평활상승DM[1] - 평활상승DM[1] / 기간값 + 상승DM;
        평활하락DM = 평활하락DM[1] - 평활하락DM[1] / 기간값 + 하락DM;
    };

    If 평활TR > 0 Then
    {
        상승DI값 = 평활상승DM / 평활TR * 100;
        하락DI값 = 평활하락DM / 평활TR * 100;
    }
    Else
    {
        상승DI값 = 0;
        하락DI값 = 0;
    };

    If 상승DI값 + 하락DI값 > 0 Then
    {
        DX값 = Abs(상승DI값 - 하락DI값)
               / (상승DI값 + 하락DI값) * 100;
    }
    Else
    {
        DX값 = 0;
    };

    If 계산봉수 <= 기간값 Then
    {
        ADX값 = 0;
    }
    Else
    {
        If 계산봉수 <= 기간값 * 2 Then
        {
            ADX값 = (ADX값[1] * (계산봉수 - 기간값 - 1) + DX값)
                    / (계산봉수 - 기간값);
        }
        Else
        {
            ADX값 = (ADX값[1] * (기간값 - 1) + DX값) / 기간값;
        };
    };
};

ADX유효출력 = IFF(계산봉수 >= 기간값 * 2, 1, 0);

If ADX유효출력 == 1 Then
{
    ADX출력 = ADX값;
    상승DI출력 = 상승DI값;
    하락DI출력 = 하락DI값;
}
Else
{
    ADX출력 = 0;
    상승DI출력 = 0;
    하락DI출력 = 0;
};

Value1 = ADX값;
WSF_DailyADXV1 = Value1;
