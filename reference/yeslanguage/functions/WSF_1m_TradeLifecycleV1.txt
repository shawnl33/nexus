Input : 후보방향입력(Numeric),
        통합상태입력(Numeric),
        미래방향입력(Numeric),
        현재회귀선입력(Numeric),
        호가상태입력(Numeric),
        호가유효입력(Numeric),
        ADX입력(Numeric),
        ADX청산기준입력(Numeric),
        고정손절틱입력(Numeric),
        수익보호시작틱입력(Numeric),
        수익보호폭틱입력(Numeric),
        포지션상태(NumericRef),
        진입신호(NumericRef),
        청산신호(NumericRef),
        진입가격(NumericRef),
        현재손익틱(NumericRef),
        최고손익틱(NumericRef),
        청산사유(NumericRef);

Var : 일분봉여부(0),
      내부포지션(0),
      내부진입가격(0),
      내부최고손익틱(0),
      계산손익틱(0),
      틱크기(0),
      ADX청산기준값(0),
      고정손절틱값(0),
      수익보호시작틱값(0),
      수익보호폭틱값(0),
      신규진입여부(0),
      청산발생(0),
      계산청산사유(0);

/*
   1분봉 진입·보유·청산 생명주기

   포지션상태 :  1 매수 보유, 0 미보유, -1 매도 보유
   진입신호   :  1 매수 진입, 0 없음, -1 매도 진입
   청산신호   :  1 매수 포지션 청산, 0 없음, -1 매도 포지션 청산

   청산사유
      1 : 고정 손절
      2 : 미래곡선 또는 통합상태 반전
      3 : 회귀선 완전 이탈
      4 : 유효한 호가 방향 반전
      5 : ADX 약화
      6 : 수익 보호

   원칙
   1. 1분봉에서만 작동한다.
   2. 장이 바뀌면 포지션 기억을 초기화한다.
   3. 포지션이 없으면 청산신호와 청산사유는 반드시 0이다.
   4. 신규 진입 봉에서는 같은 봉의 청산 조건을 검사하지 않는다.
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

틱크기 = PriceScale;
If 틱크기 <= 0 Then
    틱크기 = 1;

ADX청산기준값 = Abs(ADX청산기준입력);
고정손절틱값 = Abs(고정손절틱입력);
수익보호시작틱값 = Abs(수익보호시작틱입력);
수익보호폭틱값 = Abs(수익보호폭틱입력);

/* 출력 펄스는 매 봉 초기화한다. */
진입신호 = 0;
청산신호 = 0;
청산사유 = 0;
신규진입여부 = 0;
청산발생 = 0;
계산청산사유 = 0;
계산손익틱 = 0;

If CurrentBar == 1 or BDate <> BDate[1] or 일분봉여부 == 0 Then
{
    내부포지션 = 0;
    내부진입가격 = 0;
    내부최고손익틱 = 0;
}
Else
{
    내부포지션 = 내부포지션[1];
    내부진입가격 = 내부진입가격[1];
    내부최고손익틱 = 내부최고손익틱[1];
};

If 일분봉여부 == 1 Then
{
    /* 미보유 상태에서 후보가 처음 발생한 경우에만 진입한다. */
    If 내부포지션 == 0 Then
    {
        내부진입가격 = 0;
        내부최고손익틱 = 0;

        If 후보방향입력 > 0 Then
        {
            내부포지션 = 1;
            내부진입가격 = C;
            진입신호 = 1;
            신규진입여부 = 1;
        };

        If 후보방향입력 < 0 Then
        {
            내부포지션 = -1;
            내부진입가격 = C;
            진입신호 = -1;
            신규진입여부 = 1;
        };
    };

    If 내부포지션 <> 0 Then
    {
        계산손익틱 = (C - 내부진입가격) / 틱크기 * 내부포지션;

        If 계산손익틱 > 내부최고손익틱 Then
            내부최고손익틱 = 계산손익틱;

        /* 신규 진입 봉에는 청산을 판정하지 않는다. */
        If 신규진입여부 == 0 Then
        {
            /* 1순위: 고정 손절 */
            If 고정손절틱값 > 0 && 계산손익틱 <= -고정손절틱값 Then
            {
                청산발생 = 1;
                계산청산사유 = 1;
            };

            /* 2순위: 미래곡선 또는 통합상태가 반대 방향으로 전환 */
            If 청산발생 == 0 Then
            {
                If 내부포지션 == 1 &&
                   (미래방향입력 < 0 or 통합상태입력 <= -50) Then
                {
                    청산발생 = 1;
                    계산청산사유 = 2;
                };

                If 내부포지션 == -1 &&
                   (미래방향입력 > 0 or 통합상태입력 >= 50) Then
                {
                    청산발생 = 1;
                    계산청산사유 = 2;
                };
            };

            /* 3순위: 봉 전체가 회귀선 반대편으로 이탈 */
            If 청산발생 == 0 && 현재회귀선입력 > 0 Then
            {
                If 내부포지션 == 1 && H < 현재회귀선입력 Then
                {
                    청산발생 = 1;
                    계산청산사유 = 3;
                };

                If 내부포지션 == -1 && L > 현재회귀선입력 Then
                {
                    청산발생 = 1;
                    계산청산사유 = 3;
                };
            };

            /* 4순위: 실제 호가가 제공된 경우의 강한 반대 방향 */
            If 청산발생 == 0 && 호가유효입력 == 1 Then
            {
                If 내부포지션 == 1 && 호가상태입력 <= -2 Then
                {
                    청산발생 = 1;
                    계산청산사유 = 4;
                };

                If 내부포지션 == -1 && 호가상태입력 >= 2 Then
                {
                    청산발생 = 1;
                    계산청산사유 = 4;
                };
            };

            /* 5순위: ADX 값이 실제로 존재하면서 청산 기준 미만 */
            If 청산발생 == 0 && ADX청산기준값 > 0 &&
               ADX입력 > 0 && ADX입력 < ADX청산기준값 Then
            {
                청산발생 = 1;
                계산청산사유 = 5;
            };

            /* 6순위: 일정 수익 이후 최고 수익에서 지정 폭만큼 후퇴 */
            If 청산발생 == 0 && 수익보호시작틱값 > 0 &&
               수익보호폭틱값 > 0 &&
               내부최고손익틱 >= 수익보호시작틱값 &&
               계산손익틱 <= 내부최고손익틱 - 수익보호폭틱값 Then
            {
                청산발생 = 1;
                계산청산사유 = 6;
            };
        };

        If 청산발생 == 1 Then
        {
            청산신호 = 내부포지션;
            청산사유 = 계산청산사유;

            /* 청산 봉의 손익은 출력한 뒤 내부 보유 상태만 초기화한다. */
            내부포지션 = 0;
            내부진입가격 = 0;
            내부최고손익틱 = 0;
        };
    };
};

포지션상태 = 내부포지션;
진입가격 = 내부진입가격;

If 청산발생 == 1 Then
{
    현재손익틱 = 계산손익틱;
    최고손익틱 = Max(계산손익틱, 내부최고손익틱[1]);
}
Else
{
    If 내부포지션 <> 0 Then
    {
        현재손익틱 = 계산손익틱;
        최고손익틱 = 내부최고손익틱;
    }
    Else
    {
        현재손익틱 = 0;
        최고손익틱 = 0;
    };
};

Value1 = 내부포지션;
WSF_1m_TradeLifecycleV1 = Value1;
