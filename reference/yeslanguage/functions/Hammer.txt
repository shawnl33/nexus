Input	: Length(Numeric), Tail(Numeric);
Var		: Minv(0), Maxv(0);

Minv = Min(Close, Open);
Maxv = Max(Close, Open);
Hammer = False;

If Minv > MedianPrice() AND Open <> Close Then Begin
	If Minv - Low > (Maxv - Minv) * Tail AND High - Maxv < Maxv - Minv 
	AND LRL(Close, Length) < LRL(Close, Length)[1] Then 
		Hammer = True;
End;
