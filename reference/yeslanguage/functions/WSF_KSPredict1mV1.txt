// 국내선물 Data1 1분봉 전용. 장 시작은 BDate/DayIndex 기준.
// 해외선물 전용 사본. 공용 원본을 덮어쓰지 말 것.
Input : 세션초기화(Numeric), 세션봉수(Numeric), 현재회귀선입력(Numeric),
        회귀기울기입력(Numeric),
        회귀유효입력(Numeric),
        회귀신뢰도입력(Numeric),
        예측봉수1입력(Numeric),
        예측봉수2입력(Numeric),
        예측봉수3입력(Numeric),
        예측가격1(NumericRef),
        예측가격2(NumericRef),
        예측가격3(NumericRef),
        예측방향1(NumericRef),
        예측방향2(NumericRef),
        예측방향3(NumericRef),
        보정기울기(NumericRef),
        기울기가속도(NumericRef),
        예측변동성(NumericRef),
        예측봉수4입력(Numeric), 예측봉수5입력(Numeric),
        예측가격4(NumericRef), 예측가격5(NumericRef),
        예측방향4(NumericRef), 예측방향5(NumericRef);

//======================================================================
// WSF_KSPredict1mV1
// - V3의 R²/ATR 제한을 유지
// - 최근 3봉의 기울기 둔화를 반영하여 전환부 과대 외삽을 완화
// - 같은 방향의 가속은 사용하지 않고, 둔화/반전 성분만 사용
//======================================================================

Var : 예측봉수1(0), 예측봉수2(0), 예측봉수3(0),
      신뢰계수(0), ATR값(0), 최대기울기(0), 최대가속도(0),
      원가속도(0), 이동값1(0), 이동값2(0), 이동값3(0),
      최대이동1(0), 최대이동2(0), 최대이동3(0);

Var : 예측봉수4(0), 예측봉수5(0), 이동값4(0), 이동값5(0), 최대이동4(0), 최대이동5(0);
Var : 세션TR(0), 세션ATR(0);
세션TR = H-L;
If 세션초기화 == 0 Then 세션TR = Max(H-L,Max(Abs(H-C[1]),Abs(L-C[1])));
If 세션초기화 == 1 Then 세션ATR = 세션TR;
Else If 세션봉수 <= 14 Then 세션ATR = (세션ATR[1]*(세션봉수-1)+세션TR)/세션봉수;
Else 세션ATR = (세션ATR[1]*13+세션TR)/14;

예측봉수1 = Max(0,예측봉수1입력);
예측봉수2 = Max(0,예측봉수2입력);
예측봉수3 = Max(0,예측봉수3입력);
예측봉수4 = Max(0,예측봉수4입력);
예측봉수5 = Max(0,예측봉수5입력);

예측가격1 = 현재회귀선입력;
예측가격2 = 현재회귀선입력;
예측가격3 = 현재회귀선입력;
예측가격4 = 현재회귀선입력;
예측가격5 = 현재회귀선입력;
예측방향1 = 0;
예측방향2 = 0;
예측방향3 = 0;
예측방향4 = 0;
예측방향5 = 0;
보정기울기 = 0;
기울기가속도 = 0;
예측변동성 = 세션ATR;

If 회귀유효입력 == 1 Then
{
    신뢰계수 = (회귀신뢰도입력-0.20)/0.50;
    신뢰계수 = Max(0,Min(1,신뢰계수));

    ATR값 = 세션ATR;
    최대기울기 = ATR값*0.50;
    최대가속도 = ATR값*0.08;

    보정기울기 = 회귀기울기입력*신뢰계수;
    보정기울기 = Max(-최대기울기,Min(최대기울기,보정기울기));

    원가속도 = (회귀기울기입력-회귀기울기입력[3])/3;
    기울기가속도 = 원가속도*신뢰계수;
    기울기가속도 = Max(-최대가속도,Min(최대가속도,기울기가속도));

    // 추세를 더 과장하지 않고 둔화되는 성분만 반영
    If 보정기울기*기울기가속도 >= 0 Then 기울기가속도 = 0;

    // 30분 이하에서는 전일 기울기를 가속도 계산에 섞지 않음
    If 세션봉수 <= 3 Then
        기울기가속도 = 0;

    이동값1 = 보정기울기*예측봉수1
              + 0.5*기울기가속도*예측봉수1*예측봉수1;
    이동값2 = 보정기울기*예측봉수2
              + 0.5*기울기가속도*예측봉수2*예측봉수2;
    이동값3 = 보정기울기*예측봉수3
              + 0.5*기울기가속도*예측봉수3*예측봉수3;
    이동값4 = 보정기울기*예측봉수4
              + 0.5*기울기가속도*예측봉수4*예측봉수4;
    이동값5 = 보정기울기*예측봉수5
              + 0.5*기울기가속도*예측봉수5*예측봉수5;

    최대이동1 = ATR값*1.50*SquareRoot(Max(1,예측봉수1));
    최대이동2 = ATR값*1.50*SquareRoot(Max(1,예측봉수2));
    최대이동3 = ATR값*1.50*SquareRoot(Max(1,예측봉수3));
    최대이동4 = ATR값*1.50*SquareRoot(Max(1,예측봉수4));
    최대이동5 = ATR값*1.50*SquareRoot(Max(1,예측봉수5));

    이동값1 = Max(-최대이동1,Min(최대이동1,이동값1));
    이동값2 = Max(-최대이동2,Min(최대이동2,이동값2));
    이동값3 = Max(-최대이동3,Min(최대이동3,이동값3));
    이동값4 = Max(-최대이동4,Min(최대이동4,이동값4));
    이동값5 = Max(-최대이동5,Min(최대이동5,이동값5));

    예측가격1 = 현재회귀선입력+이동값1;
    예측가격2 = 현재회귀선입력+이동값2;
    예측가격3 = 현재회귀선입력+이동값3;
    예측가격4 = 현재회귀선입력+이동값4;
    예측가격5 = 현재회귀선입력+이동값5;

    예측방향1 = IFF(이동값1 > 0,1,IFF(이동값1 < 0,-1,0));
    예측방향2 = IFF(이동값2 > 0,1,IFF(이동값2 < 0,-1,0));
    예측방향3 = IFF(이동값3 > 0,1,IFF(이동값3 < 0,-1,0));
    예측방향4 = IFF(이동값4 > 0,1,IFF(이동값4 < 0,-1,0));
    예측방향5 = IFF(이동값5 > 0,1,IFF(이동값5 < 0,-1,0));
    예측변동성 = ATR값;
}

Value1 = 1;
WSF_KSPredict1mV1 = Value1;
