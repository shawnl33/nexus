Input : 청산요청입력(Numeric),
        저장진입방향입력(Numeric),
        저장진입가격입력(Numeric),
        청산가격입력(Numeric),
        틱크기입력(Numeric),
        틱가치입력(Numeric),
        저장청산가격출력(NumericRef),
        저장청산일자출력(NumericRef),
        저장청산시간출력(NumericRef),
        저장청산봉번호출력(NumericRef),
        틱손익출력(NumericRef),
        원화손익출력(NumericRef);

Var : 내부청산가격(0),
      내부청산일자(0),
      내부청산시간(0),
      내부청산봉번호(0),
      내부틱손익(0),
      내부원화손익(0),
      적용청산가격(0),
      적용틱크기(0);

/*
   WSF_1m_ExitSnapshotV1

   청산요청이 발생한 순간의 가격, 일자, 시간, 봉 번호를 저장한다.
   저장된 진입 방향과 진입가격을 기준으로 틱손익과 원화손익을 계산한다.
*/

If CurrentBar == 1 Then
{
    내부청산가격 = 0;
    내부청산일자 = 0;
    내부청산시간 = 0;
    내부청산봉번호 = 0;
    내부틱손익 = 0;
    내부원화손익 = 0;
}
Else
{
    내부청산가격 = 내부청산가격[1];
    내부청산일자 = 내부청산일자[1];
    내부청산시간 = 내부청산시간[1];
    내부청산봉번호 = 내부청산봉번호[1];
    내부틱손익 = 내부틱손익[1];
    내부원화손익 = 내부원화손익[1];
};

If CurrentBar > 1 && BDate <> BDate[1] Then
{
    내부청산가격 = 0;
    내부청산일자 = 0;
    내부청산시간 = 0;
    내부청산봉번호 = 0;
    내부틱손익 = 0;
    내부원화손익 = 0;
};

If 청산요청입력 == 1 && 저장진입방향입력 <> 0 && 저장진입가격입력 > 0 Then
{
    적용청산가격 = 청산가격입력;

    If 적용청산가격 <= 0 Then
        적용청산가격 = C;

    적용틱크기 = 틱크기입력;

    If 적용틱크기 <= 0 Then
        적용틱크기 = PriceScale;

    내부청산가격 = 적용청산가격;
    내부청산일자 = BDate;
    내부청산시간 = STime;
    내부청산봉번호 = CurrentBar;
    내부틱손익 =
        ((적용청산가격 - 저장진입가격입력) * 저장진입방향입력)
        / 적용틱크기;
    내부원화손익 = 내부틱손익 * 틱가치입력;
};

저장청산가격출력 = 내부청산가격;
저장청산일자출력 = 내부청산일자;
저장청산시간출력 = 내부청산시간;
저장청산봉번호출력 = 내부청산봉번호;
틱손익출력 = 내부틱손익;
원화손익출력 = 내부원화손익;

Value1 = 틱손익출력;
WSF_1m_ExitSnapshotV1 = Value1;
