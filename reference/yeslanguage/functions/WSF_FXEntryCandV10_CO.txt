// 새 돌파 판정
S돌파방향 = 0;
If S사용 == 1 && CurrentBar > 2 && 세션초기화 == 0 &&
   구조유효[1] == 1 && 구조유효[2] == 1 && 폭퍼센트[1] < S구조폭기준 &&
   압축지속봉수[1] >= S압축봉수 && 압축폭비율 < S폭비율기준 Then
{
    If C > 전체최고[1] && C[1] <= 전체최고[1] Then S돌파방향 = 1;
    If C < 전체최저[1] && C[1] >= 전체최저[1] Then S돌파방향 = -1;
};
If S돌파방향 != 0 Then
{
    If 압축지속봉수 == 0 || 압축지속봉수 >= S진행중봉수 Then S방향 = S돌파방향;
    Else If S대기봉수 > 0 Then
    {
        S대기방향 = S돌파방향; S대기경과 = 0;
        S대기기준 = IFF(S돌파방향 == 1,전체최고[1],전체최저[1]);
    };
};
If S방향 != 0 Then
{
    후보방향 = S방향; 후보등급 = 5;
};

// 진입 시간대 제한 (자정을 넘는 시간대 지원). 횟수 제한 없음
시간대안 = 0;
If 진입시작시각 < 진입종료시각 Then
{
    If sTime >= 진입시작시각 && sTime < 진입종료시각 Then 시간대안 = 1;
}
Else
{
    If sTime >= 진입시작시각 || sTime < 진입종료시각 Then 시간대안 = 1;
};
If 진입시작시각 == 진입종료시각 Then 시간대안 = 1;
If 후보방향 != 0 && 시간대안 == 0 Then
{
    후보방향 = 0; 후보등급 = 0;
};
// 반대 신호 차단 (반대차단시작시각 ~ 07:00, 자정 넘김 지원)
반대차단시간 = 0;
If 반대차단시작시각 >= 70000 Then
{
    If sTime >= 반대차단시작시각 || sTime < 70000 Then 반대차단시간 = 1;
}
Else
{
    If sTime >= 반대차단시작시각 && sTime < 70000 Then 반대차단시간 = 1;
};
If 후보방향 != 0 && 반대차단봉수 > 0 && 반대차단시간 == 1 &&
   직전후보방향 == -후보방향 && 직전후보경과 < 반대차단봉수 Then
{
    후보방향 = 0; 후보등급 = 0;
};
If 후보방향 != 0 Then
{
    직전후보방향 = 후보방향; 직전후보경과 = 0;
    If 후보등급 == 4 Then { 구조확정방향 = 0; 구조확정경과 = 0; };
};

후보방향출력 = 후보방향;
후보등급출력 = 후보등급;
통합상태출력 = 통합상태;
Value1 = 후보방향;
WSF_FXEntryCandV10_CO = Value1;
