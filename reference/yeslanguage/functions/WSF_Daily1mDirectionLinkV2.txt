Input : 일봉통합상태입력(Numeric),
        일분봉통합상태입력(Numeric),
        연결통합상태출력(NumericRef),
        방향일치출력(NumericRef),
        매수허용출력(NumericRef),
        매도허용출력(NumericRef);

Var : 일봉방향(0),
      일분봉방향(0);

/* 일봉과 1분봉 방향이 같을 때만 1분봉 통합상태를 통과시킨다. */
연결통합상태출력 = 0;
방향일치출력 = 0;
매수허용출력 = 0;
매도허용출력 = 0;

일봉방향 = IFF(일봉통합상태입력 > 0, 1,
             IFF(일봉통합상태입력 < 0, -1, 0));

일분봉방향 = IFF(일분봉통합상태입력 > 0, 1,
               IFF(일분봉통합상태입력 < 0, -1, 0));

If 일봉방향 <> 0 && 일봉방향 == 일분봉방향 Then
{
    방향일치출력 = 일봉방향;
    연결통합상태출력 = 일분봉통합상태입력;

    If 연결통합상태출력 == 100 Then
        매수허용출력 = 1;

    If 연결통합상태출력 == -100 Then
        매도허용출력 = 1;
};

Value1 = 연결통합상태출력;
WSF_Daily1mDirectionLinkV2 = Value1;
