Input : 일봉회귀기간입력(Numeric),
        최소신뢰도입력(Numeric),
        일봉회귀선(NumericRef),
        일봉회귀기울기(NumericRef),
        일봉회귀신뢰도(NumericRef),
        일봉회귀잔차(NumericRef),
        일봉추세방향(NumericRef),
        일봉추세상태(NumericRef),
        일봉추세강도(NumericRef),
        일봉추세유효(NumericRef);

//======================================================================
// WSF_1m_DailyTrendLinkV1
// - 1분봉에서 각 세션의 고가/저가를 직접 집계해 완성 일봉 생성
// - 오늘 진행 중인 일봉은 제외하고 직전 N개 완성 일봉만 사용
// - 마지막 완성 일봉 회귀값에서 오늘 위치까지 1봉 투영
// - 일봉 방향은 하루 동안 고정되고, 강/약 상태는 현재가격으로 확인
//======================================================================

Var : n(20), j(0), X값(0), Y값(0),
      SumX(0), SumY(0), SumXY(0), SumX2(0), SumY2(0),
      분모(0), LRS값(0), B값(0), 마지막회귀값(0),
      R2분자(0), R2분모(0), SST(0), SSE(0),
      연결유효(0), 현재장고가(0), 현재장저가(0),
      집계초기화(0), 완전장시작확인(0),
      최근장시작봉(-1), 완성일수(0);

Array : 완성일봉중간[100](0);

n = Min(100,Max(5,일봉회귀기간입력));

일봉회귀선 = C;
일봉회귀기울기 = 0;
일봉회귀신뢰도 = 0;
일봉회귀잔차 = 0;
일봉추세방향 = 0;
일봉추세상태 = 0;
일봉추세강도 = 0;
일봉추세유효 = 0;
연결유효 = 0;

If DataCompress == 2 && BarInterval == 1 Then
{
    If 집계초기화 == 0 Then
    {
        현재장고가 = H;
        현재장저가 = L;
        집계초기화 = 1;

        If DayIndex == 0 Then
        {
            완전장시작확인 = 1;
            최근장시작봉 = Index;
        }
    }
    Else
    {
        If DayIndex == 0 && Index != 최근장시작봉 Then
        {
            // 처음 불러온 세션이 중간부터 시작한 경우에는 저장하지 않음
            If 완전장시작확인 == 1 Then
            {
                For j = 98 DownTo 0
                {
                    완성일봉중간[j+1] = 완성일봉중간[j];
                }

                완성일봉중간[0] = (현재장고가+현재장저가)/2;
                완성일수 = Min(100,완성일수+1);
            }

            현재장고가 = H;
            현재장저가 = L;
            완전장시작확인 = 1;
            최근장시작봉 = Index;
        }
        Else
        {
            현재장고가 = Max(현재장고가,H);
            현재장저가 = Min(현재장저가,L);
        }
    }

    If 완성일수 >= n Then
    {
        SumX = 0;
        SumY = 0;
        SumXY = 0;
        SumX2 = 0;
        SumY2 = 0;

        // 배열 0은 최신 완성 일봉, n-1은 가장 오래된 완성 일봉
        For j = 0 To n-1
        {
            X값 = j;
            Y값 = 완성일봉중간[n-1-j];

            SumX = SumX + X값;
            SumY = SumY + Y값;
            SumXY = SumXY + X값*Y값;
            SumX2 = SumX2 + X값*X값;
            SumY2 = SumY2 + Y값*Y값;
        }

        분모 = n*SumX2-SumX*SumX;

        If 분모 != 0 Then
        {
            LRS값 = (n*SumXY-SumX*SumY)/분모;
            B값 = (SumY*SumX2-SumX*SumXY)/분모;

            마지막회귀값 = LRS값*(n-1)+B값;
            일봉회귀선 = 마지막회귀값+LRS값;
            일봉회귀기울기 = LRS값;

            R2분자 = n*SumXY-SumX*SumY;
            R2분모 = SquareRoot(Max(0,
                       (n*SumX2-SumX*SumX)*(n*SumY2-SumY*SumY)));

            If R2분모 > 0 Then
                일봉회귀신뢰도 = Min(1,Max(0,(R2분자/R2분모)^2));

            SST = SumY2-SumY*SumY/n;
            SSE = Max(0,SST*(1-일봉회귀신뢰도));

            If n > 2 Then
                일봉회귀잔차 = SquareRoot(SSE/(n-2));

            연결유효 = 1;
        }
    }
}

WSF_Daily_LinRegTrendV1(
    일봉회귀선,
    일봉회귀기울기,
    연결유효,
    일봉회귀신뢰도,
    일봉회귀잔차,
    C,
    최소신뢰도입력,
    일봉추세방향,
    일봉추세상태,
    일봉추세강도,
    일봉추세유효
);

Value1 = 1;
WSF_1m_DailyTrendLinkV1 = Value1;
