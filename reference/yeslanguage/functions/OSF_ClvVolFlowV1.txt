//======================================================================
// OSF_ClvVolFlowV1
// CLV압력(OSF_ClvPressureV1, 빠른 트리거) + 거래량흐름(OSF_VolFlowV1, 느린 필터) 결합
//
// 결합 규칙 (v2 - 방향은 거래량흐름이 확정, CLV는 강도만 조절):
//   - 최종 부호(방향)는 항상 거래량흐름점수의 부호를 따른다
//     -> CLV가 순간적으로 반대로 튀어도 결합 결과의 방향 자체는 뒤집히지 않는다
//        (v1에서는 부호가 다를 때 크기만 줄였을 뿐 방향은 CLV가 이길 수 있어
//         결합오실레이터가 CLV만큼 자주 뒤집히는 문제가 있었음)
//   - CLV가 거래량흐름과 같은 방향이면(확인) 거래량흐름값을 1.2배로 강화
//   - CLV가 거래량흐름과 다른 방향이면(비확인/이견) 거래량흐름값을 0.5배로 약화
//   - 거래량흐름이 정확히 0(추세 없음)이면 결합값도 0(중립)
// 유효 여부는 거래량흐름의 워밍업 기간(계산기간입력)을 그대로 따른다
//======================================================================
Input : 관심기준입력(Numeric),
        강세기준입력(Numeric),
        부호반전입력(Numeric),
        계산기간입력(Numeric),
        결합점수(NumericRef),
        결합기울기3(NumericRef),
        결합상태(NumericRef),
        결합유효(NumericRef);

Var : CLV점수(0), CLV기울기(0), CLV상태(0), CLV유효(0),
      흐름점수(0), 흐름기울기(0), 흐름상태(0), 흐름유효(0),
      부호일치(0),
      계산결합(0),
      계산기울기3(0),
      계산상태(0),
      관심기준값(0),
      강세기준값(0);

관심기준값 = Abs(관심기준입력);
If 관심기준값 < 1 Then
    관심기준값 = 1;

강세기준값 = Abs(강세기준입력);
If 강세기준값 < 관심기준값 Then
    강세기준값 = 관심기준값;

// 두 하위 함수는 방향반전 없이(0) 원점수로 호출하고, 반전은 결합 마지막 단계에서 한 번만 적용
OSF_ClvPressureV1(
    관심기준입력, 강세기준입력, 0,
    CLV점수, CLV기울기, CLV상태, CLV유효
);

OSF_VolFlowV1(
    관심기준입력, 강세기준입력, 0, 계산기간입력,
    흐름점수, 흐름기울기, 흐름상태, 흐름유효
);

부호일치 =
    IFF(CLV점수 > 0 && 흐름점수 > 0, 1,
    IFF(CLV점수 < 0 && 흐름점수 < 0, 1, 0));

If 흐름점수 == 0 Then
    계산결합 = 0;
Else If 부호일치 == 1 Then
    계산결합 = 흐름점수 * 1.2;
Else
    계산결합 = 흐름점수 * 0.5;

If 부호반전입력 == 1 Then
    계산결합 = -계산결합;

If 계산결합 > 100 Then
    계산결합 = 100;
If 계산결합 < -100 Then
    계산결합 = -100;

If CurrentBar >= Round(계산기간입력,0) + 3 Then
    계산기울기3 = 계산결합 - 계산결합[3];
Else
    계산기울기3 = 0;

If 계산기울기3 > 100 Then
    계산기울기3 = 100;
If 계산기울기3 < -100 Then
    계산기울기3 = -100;

계산상태 =
    IFF(계산결합 >= 강세기준값, 2,
    IFF(계산결합 >= 관심기준값, 1,
    IFF(계산결합 <= -강세기준값, -2,
    IFF(계산결합 <= -관심기준값, -1, 0))));

결합점수 = 계산결합;
결합기울기3 = 계산기울기3;
결합상태 = 계산상태;
결합유효 = 흐름유효;

Value1 = 1;
OSF_ClvVolFlowV1 = Value1;
