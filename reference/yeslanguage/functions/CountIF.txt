Inputs	: Criteria(logical), Length(Numeric);
Var		: Accumv(0), Counter(0);

For Counter = 0 To Length - 1 Begin	
	If Criteria[Counter] Then
		Accumv = Accumv + 1;
End;

CountIF = Accumv;
Accumv 	= 0;
