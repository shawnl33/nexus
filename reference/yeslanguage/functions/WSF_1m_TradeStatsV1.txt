Input : 청산요청입력(Numeric),
        틱손익입력(Numeric),
        원화손익입력(Numeric),
        거래횟수출력(NumericRef),
        승리횟수출력(NumericRef),
        패배횟수출력(NumericRef),
        무승부횟수출력(NumericRef),
        승률출력(NumericRef),
        누적틱손익출력(NumericRef),
        누적원화손익출력(NumericRef);

Var : 내부거래횟수(0),
      내부승리횟수(0),
      내부패배횟수(0),
      내부무승부횟수(0),
      내부승률(0),
      내부누적틱손익(0),
      내부누적원화손익(0);

/*
   WSF_1m_TradeStatsV1

   청산이 발생할 때마다 당일 거래 결과를 한 번 누적한다.
   새 거래일에는 모든 누적값을 초기화한다.
*/

If CurrentBar == 1 Then
{
    내부거래횟수 = 0;
    내부승리횟수 = 0;
    내부패배횟수 = 0;
    내부무승부횟수 = 0;
    내부승률 = 0;
    내부누적틱손익 = 0;
    내부누적원화손익 = 0;
}
Else
{
    내부거래횟수 = 내부거래횟수[1];
    내부승리횟수 = 내부승리횟수[1];
    내부패배횟수 = 내부패배횟수[1];
    내부무승부횟수 = 내부무승부횟수[1];
    내부승률 = 내부승률[1];
    내부누적틱손익 = 내부누적틱손익[1];
    내부누적원화손익 = 내부누적원화손익[1];
};

If CurrentBar > 1 && BDate <> BDate[1] Then
{
    내부거래횟수 = 0;
    내부승리횟수 = 0;
    내부패배횟수 = 0;
    내부무승부횟수 = 0;
    내부승률 = 0;
    내부누적틱손익 = 0;
    내부누적원화손익 = 0;
};

If 청산요청입력 == 1 Then
{
    내부거래횟수 = 내부거래횟수 + 1;
    내부누적틱손익 = 내부누적틱손익 + 틱손익입력;
    내부누적원화손익 = 내부누적원화손익 + 원화손익입력;

    If 틱손익입력 > 0 Then
        내부승리횟수 = 내부승리횟수 + 1;

    If 틱손익입력 < 0 Then
        내부패배횟수 = 내부패배횟수 + 1;

    If 틱손익입력 == 0 Then
        내부무승부횟수 = 내부무승부횟수 + 1;
};

If 내부거래횟수 > 0 Then
    내부승률 = 내부승리횟수 / 내부거래횟수 * 100;
Else
    내부승률 = 0;

거래횟수출력 = 내부거래횟수;
승리횟수출력 = 내부승리횟수;
패배횟수출력 = 내부패배횟수;
무승부횟수출력 = 내부무승부횟수;
승률출력 = 내부승률;
누적틱손익출력 = 내부누적틱손익;
누적원화손익출력 = 내부누적원화손익;

Value1 = 누적틱손익출력;
WSF_1m_TradeStatsV1 = Value1;
