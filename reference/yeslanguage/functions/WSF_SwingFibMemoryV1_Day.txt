        시간비율(NumericRef);

//======================================================================
// WSF_SwingFibMemoryV1
// - 입력으로 받은 세션미래방향(핵심세션미래방향)의 0선 전환 시점을

//   38.2/50.0/61.8% 되돌림 레벨을 "확정값"으로 기억해서 돌려준다.
// - 시간비율 = 지금 진행 중인 반대방향 구간의 경과봉수 ÷ 그 구간이
//   되돌리고 있는 직전 확정구간의 총 경과봉수
//   (가격 되돌림과 같은 발상을 "시간" 축에 적용한 것)
// - 이 함수는 Plot을 하지 않는다. 호출부(인디케이터)에서
//   핵심방향구간방향으로 "지금 되돌리는 중인 구간만" 표시 조건을
//   걸고, 시간비율로 페이드 여부를 결정해서 그린다.
//
// 설계 메모
// - 지속 상태(직전 봉 값, 진행 중 구간 등)는 전부 내부 Var(상태_...)
//   에만 저장하고, NumericRef 출력 파라미터는 매 호출 끝에서 값만
//   "쓰기"로 채운다. 출력 파라미터를 다시 "읽어서" 상태로 쓰는 방식은
//   WSF_OrderBookDirectionV1 등 기존 함수들에서 쓰인 적이 없어
//   피했다(검증되지 않은 패턴).
// - 같은 이유로, 입력 파라미터 핵심세션미래방향에 [1] 같은 과거값
//   참조를 직접 쓰지 않고, 내부 Var(상태_직전미래방향)에 매 봉 끝에서
//   현재값을 저장해뒀다가 다음 호출에서 "직전값"으로 비교한다.
//   (Input 파라미터에 [n] 인덱싱이 지원되는지 매뉴얼로 확인하지
//   못했음 ? 이 방식이면 그 여부와 무관하게 안전하다.)
// - 호출부는 반드시 매 봉(해당 봉주기에서) 이 함수를 호출해야 한다.
//   1분봉 전용으로 쓰려면 호출 자체를 DataCompress==2 && BarInterval==1
//   조건 안에서만 하면 된다(기존 V16 §3-9-A의 핵심일분봉여부 게이트와
//   동일한 방식).
//======================================================================

Var : 상태_핵심상승최고가(0), 상태_핵심상승최저가(0), 상태_핵심상승Count(0),
      상태_핵심하락최고가(0), 상태_핵심하락최저가(0), 상태_핵심하락Count(0),

      상태_핵심지난상승최고가(0), 상태_핵심지난상승최저가(0), 상태_핵심지난상승Count(0),
      상태_핵심지난상승_크기(0), 상태_핵심지난상승_382(0), 상태_핵심지난상승_500(0), 상태_핵심지난상승_618(0),

      상태_핵심지난하락최고가(0), 상태_핵심지난하락최저가(0), 상태_핵심지난하락Count(0),
      상태_핵심지난하락_크기(0), 상태_핵심지난하락_382(0), 상태_핵심지난하락_500(0), 상태_핵심지난하락_618(0),

      상태_핵심방향구간방향(0), 상태_핵심양수전환(0), 상태_핵심음수전환(0),
      상태_시간비율(0), 상태_직전미래방향(0);

If barIndex == 0 Then
{
    상태_핵심상승최고가 = 0; 상태_핵심상승최저가 = 0; 상태_핵심상승Count = 0;
    상태_핵심하락최고가 = 0; 상태_핵심하락최저가 = 0; 상태_핵심하락Count = 0;

    상태_핵심지난상승최고가 = 0; 상태_핵심지난상승최저가 = 0; 상태_핵심지난상승Count = 0;
    상태_핵심지난상승_크기 = 0; 상태_핵심지난상승_382 = 0; 상태_핵심지난상승_500 = 0; 상태_핵심지난상승_618 = 0;

    상태_핵심지난하락최고가 = 0; 상태_핵심지난하락최저가 = 0; 상태_핵심지난하락Count = 0;
    상태_핵심지난하락_크기 = 0; 상태_핵심지난하락_382 = 0; 상태_핵심지난하락_500 = 0; 상태_핵심지난하락_618 = 0;

    상태_핵심방향구간방향 = 0;
    상태_시간비율 = 0;
    상태_직전미래방향 = 0;
}
Else
{
    상태_핵심양수전환 = IFF(상태_직전미래방향 <= 0 && 핵심세션미래방향 > 0,1,0);
    상태_핵심음수전환 = IFF(상태_직전미래방향 >= 0 && 핵심세션미래방향 < 0,1,0);

    If 상태_핵심양수전환 == 1 Then
    {
        // 이전 하락 구간이 있었다면, 새 상승 구간을 열기 전에 확정 저장
        If 상태_핵심방향구간방향 == -1 Then
        {
            상태_핵심지난하락최고가 = 상태_핵심하락최고가;
            상태_핵심지난하락최저가 = 상태_핵심하락최저가;
            상태_핵심지난하락_크기  = 상태_핵심지난하락최고가 - 상태_핵심지난하락최저가;
            상태_핵심지난하락_382  = 상태_핵심지난하락최저가 + 상태_핵심지난하락_크기*0.382;
            상태_핵심지난하락_500  = 상태_핵심지난하락최저가 + 상태_핵심지난하락_크기*0.500;
            상태_핵심지난하락_618  = 상태_핵심지난하락최저가 + 상태_핵심지난하락_크기*0.618;
            상태_핵심지난하락Count = 상태_핵심하락Count;
        };
        상태_핵심상승최고가 = H;
        상태_핵심상승최저가 = L;
        상태_핵심상승Count = 1;
        상태_핵심방향구간방향 = 1;
    }
    Else If 상태_핵심음수전환 == 1 Then
    {
        // 이전 상승 구간이 있었다면, 새 하락 구간을 열기 전에 확정 저장
        If 상태_핵심방향구간방향 == 1 Then
        {
            상태_핵심지난상승최고가 = 상태_핵심상승최고가;
            상태_핵심지난상승최저가 = 상태_핵심상승최저가;
            상태_핵심지난상승_크기  = 상태_핵심지난상승최고가 - 상태_핵심지난상승최저가;
            상태_핵심지난상승_382  = 상태_핵심지난상승최저가 + 상태_핵심지난상승_크기*0.382;
            상태_핵심지난상승_500  = 상태_핵심지난상승최저가 + 상태_핵심지난상승_크기*0.500;
            상태_핵심지난상승_618  = 상태_핵심지난상승최저가 + 상태_핵심지난상승_크기*0.618;
            상태_핵심지난상승Count = 상태_핵심상승Count;
        };
        상태_핵심하락최고가 = H;
        상태_핵심하락최저가 = L;
        상태_핵심하락Count = 1;
        상태_핵심방향구간방향 = -1;
    }
    Else If 상태_핵심방향구간방향 == 1 && 핵심세션미래방향 > 0 Then
    {
        상태_핵심상승최고가 = Max(상태_핵심상승최고가,H);
        상태_핵심상승최저가 = Min(상태_핵심상승최저가,L);
        상태_핵심상승Count = 상태_핵심상승Count + 1;
    }
    Else If 상태_핵심방향구간방향 == -1 && 핵심세션미래방향 < 0 Then
    {
        상태_핵심하락최고가 = Max(상태_핵심하락최고가,H);
        상태_핵심하락최저가 = Min(상태_핵심하락최저가,L);
        상태_핵심하락Count = 상태_핵심하락Count + 1;
    };

    // 시간비율 ? 지금 되돌리고 있는 반대 구간 기준으로만 계산
    상태_시간비율 = 0;
    If 상태_핵심방향구간방향 == -1 && 상태_핵심지난상승Count > 0 Then
        상태_시간비율 = 상태_핵심하락Count / 상태_핵심지난상승Count;
    If 상태_핵심방향구간방향 == 1 && 상태_핵심지난하락Count > 0 Then
        상태_시간비율 = 상태_핵심상승Count / 상태_핵심지난하락Count;

    상태_직전미래방향 = 핵심세션미래방향;
};

핵심지난하락_382  = 상태_핵심지난하락_382;

핵심방향구간방향 = 상태_핵심방향구간방향;
