Input 	: Period(NumericSimple);
Var 	: Value_Lowest(0), Value_Highest(0);

Value_Lowest 	= Lowest(L, Period);
Value_Highest 	= Highest(H, Period);

FastK = (Close - Value_Lowest) / (Value_Highest - Value_Lowest) * 100;
