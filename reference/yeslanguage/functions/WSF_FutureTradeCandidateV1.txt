Input : 통합상태입력(Numeric),
        매수후보(NumericRef),
        매도후보(NumericRef),
        후보방향(NumericRef);

Var : 현재통합상태(0),
      이전통합상태(0),
      일분봉여부(0);

일분봉여부 =
    IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

현재통합상태 = 통합상태입력;

/* 1분봉은 장 시작 시 이전 상태를 초기화한다.
   일봉은 전일 상태와 계속 비교한다. */
If 일분봉여부 == 1 && BDate <> BDate[1] Then
{
    이전통합상태 = 0;
}
Else
{
    이전통합상태 = 통합상태입력[1];
};

매수후보 = 0;
매도후보 = 0;
후보방향 = 0;

/* +100 상태에 처음 진입한 봉 */
If 현재통합상태 == 100 &&
   이전통합상태 <> 100 Then
{
    매수후보 = 1;
    후보방향 = 1;
};

/* -100 상태에 처음 진입한 봉 */
If 현재통합상태 == -100 &&
   이전통합상태 <> -100 Then
{
    매도후보 = 1;
    후보방향 = -1;
};

Value1 = 후보방향;
WSF_FutureTradeCandidateV1 = Value1;
