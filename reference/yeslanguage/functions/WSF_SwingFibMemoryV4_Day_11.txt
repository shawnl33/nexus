                        상태_핵심지난하락_382유지 = 0; 상태_핵심지난하락_500유지 = 0; 상태_핵심지난하락_618유지 = 0;
                        상태_핵심지난하락_붕괴 = 0;
                        상태_핵심지난하락_확장100 = 0; 상태_핵심지난하락_확장1382 = 0; 상태_핵심지난하락_확장1618 = 0;
                        상태_핵심지난하락_확장200 = 0; 상태_핵심지난하락_확장2382 = 0;
                        If 상태_핵심지난하락Count > 0 Then
                        {
                            상태_핵심지난하락_확장100  = 상태_핵심지난하락최고가 - 상태_핵심지난하락_크기*1.000;
                            상태_핵심지난하락_확장1382 = 상태_핵심지난하락최고가 - 상태_핵심지난하락_크기*1.382;
                            상태_핵심지난하락_확장1618 = 상태_핵심지난하락최고가 - 상태_핵심지난하락_크기*1.618;
                            상태_핵심지난하락_확장200  = 상태_핵심지난하락최고가 - 상태_핵심지난하락_크기*2.000;
                            상태_핵심지난하락_확장2382 = 상태_핵심지난하락최고가 - 상태_핵심지난하락_크기*2.382;

                            If 상태_핵심지난상승최고가 < 상태_핵심지난하락_618 Then
                                상태_핵심지난하락_618유지 = 1;
                            If 상태_핵심지난상승최고가 < 상태_핵심지난하락_500 Then
                                상태_핵심지난하락_500유지 = 1;
                            If 상태_핵심지난상승최고가 < 상태_핵심지난하락_382 Then
                                상태_핵심지난하락_382유지 = 1;

                            // 붕괴: 이번 반등이 기준 하락 자체의 시작가(직전 고점,
                            // 즉 100% 되돌림선)까지 뚫었으면 구조가 완전히 무효화된
                            // 것으로 본다 ? 확장 자체가 의미 없어지므로 별도 플래그로 표시.
                            If 상태_핵심지난상승최고가 > 상태_핵심지난하락최고가 Then
                                상태_핵심지난하락_붕괴 = 1;
                        };

                        // 지금 확정되는 핵심지난상승(이번 상승) 자체의 구조약화 여부:
                        // 그 반등이 직전 하락의 50%도 못 넘어섰다면(=500유지가 섰다면)
                        // 이 상승구간 자체가 약한 반등이었던 것으로 본다.
                        상태_핵심지난상승_구조약화 = 상태_핵심지난하락_500유지;
                    };
                    상태_핵심하락최고가 = 상태_대기최고가;
                    상태_핵심하락최저가 = 상태_대기최저가;
                    상태_핵심하락Count = 상태_대기Count;
                    상태_핵심방향구간방향 = -1;
                };

                상태_대기방향 = 0; 상태_대기Count = 0; 상태_대기최고가 = 0; 상태_대기최저가 = 0;
            };
        }
        Else If (상태_대기방향 == 1 && 핵심세션미래방향 < 0) || (상태_대기방향 == -1 && 핵심세션미래방향 > 0) Then
        {
            If 상태_핵심방향구간방향 == 1 Then
            {
                상태_핵심상승최고가 = Max(상태_핵심상승최고가,Max(상태_대기최고가,H));
                상태_핵심상승최저가 = Min(상태_핵심상승최저가,Min(상태_대기최저가,L));
                상태_핵심상승Count = 상태_핵심상승Count + 상태_대기Count + 1;
            }
            Else If 상태_핵심방향구간방향 == -1 Then
            {
                상태_핵심하락최고가 = Max(상태_핵심하락최고가,Max(상태_대기최고가,H));
                상태_핵심하락최저가 = Min(상태_핵심하락최저가,Min(상태_대기최저가,L));
                상태_핵심하락Count = 상태_핵심하락Count + 상태_대기Count + 1;
            };

            상태_대기방향 = 0; 상태_대기Count = 0; 상태_대기최고가 = 0; 상태_대기최저가 = 0;
        };
    }
    Else
    {
        If 상태_핵심양수전환 == 1 && 상태_핵심방향구간방향 != 1 Then
        {
            상태_대기방향 = 1;
            상태_대기Count = 1;
            상태_대기최고가 = H;
            상태_대기최저가 = L;
        }
        Else If 상태_핵심음수전환 == 1 && 상태_핵심방향구간방향 != -1 Then
        {
            상태_대기방향 = -1;
            상태_대기Count = 1;
            상태_대기최고가 = H;
            상태_대기최저가 = L;
        }
        Else If 상태_핵심방향구간방향 == 1 && 핵심세션미래방향 > 0 Then
        {
            상태_핵심상승최고가 = Max(상태_핵심상승최고가,H);
            상태_핵심상승최저가 = Min(상태_핵심상승최저가,L);
            상태_핵심상승Count = 상태_핵심상승Count + 1;
        }
        Else If 상태_핵심방향구간방향 == -1 && 핵심세션미래방향 < 0 Then
        {
            상태_핵심하락최고가 = Max(상태_핵심하락최고가,H);
            상태_핵심하락최저가 = Min(상태_핵심하락최저가,L);
            상태_핵심하락Count = 상태_핵심하락Count + 1;
        };
    };

    // 시간비율 ? 지금 되돌리고 있는 반대 구간 기준으로만 계산
    상태_시간비율 = 0;
    If 상태_핵심방향구간방향 == -1 && 상태_핵심지난상승Count > 0 Then
        상태_시간비율 = 상태_핵심하락Count / 상태_핵심지난상승Count;
    If 상태_핵심방향구간방향 == 1 && 상태_핵심지난하락Count > 0 Then
        상태_시간비율 = 상태_핵심상승Count / 상태_핵심지난하락Count;

    상태_직전미래방향 = 핵심세션미래방향;
};

// 1=중간목표(138.2/200%), 2=깊은목표(161.8/238.2%)
// 구조약화면 한 단계 낮춘다. 붕괴(되돌림이 기준구간 시작가까지 뚫림)면

    IFF(상태_핵심지난상승_붕괴 == 1, -2,
