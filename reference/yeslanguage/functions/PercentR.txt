Input	: Length(NumericSimple);
Var		: HiLoDiff(0), ValueHigiest(0);

ValueHigiest 	= Highest(High, Length);
HiLoDiff 		= ValueHigiest - Lowest(Low, Length);

If HiLoDiff != 0 Then 
	PercentR = 100 - ((ValueHigiest - Close) / HiLoDiff) * 100;
Else
	PercentR = 0;
