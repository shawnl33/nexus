Input : 통합상태입력(Numeric),
        청산신호입력(Numeric),
        틱크기입력(Numeric),
        틱가치입력(Numeric),
        포지션상태(NumericRef),
        신규진입신호(NumericRef),
        신규청산신호(NumericRef),
        진입가격출력(NumericRef),
        청산가격출력(NumericRef),
        실현틱손익(NumericRef),
        실현원화손익(NumericRef),
        누적원화손익(NumericRef),
        보유봉수(NumericRef);

Var : 현재포지션(0),
      저장진입가격(0),
      저장누적원화손익(0),
      이전통합상태(0),
      틱크기(0),
      틱가치(0),
      계산가격차이(0),
      계산틱손익(0),
      계산원화손익(0),
      일분봉여부(0),
      새세션여부(0);

/*
   WSF_1m_TradeLifecycleV2

   통합상태입력
      100  : 매수 확인
       50  : 매수 관찰
        0  : 중립
      -50  : 매도 관찰
     -100  : 매도 확인

   청산신호입력
        1  : 매수 포지션 청산
       -1  : 매도 포지션 청산
        2  : 방향과 관계없이 보유 포지션 청산
        0  : 유지

   포지션상태
        1  : 매수 보유
        0  : 미보유
       -1  : 매도 보유

   원칙
   1. 1분봉에서만 작동한다.
   2. 장이 바뀌면 포지션과 당일 누적손익을 초기화한다.
   3. 통합상태가 처음 100/-100에 진입할 때만 신규 진입한다.
   4. 같은 확인 상태가 이어져도 중복 진입하지 않는다.
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);
새세션여부 = IFF(CurrentBar == 1 || BDate <> BDate[1], 1, 0);

틱크기 = Abs(틱크기입력);
If 틱크기 <= 0 Then
    틱크기 = PriceScale;

틱가치 = Abs(틱가치입력);

신규진입신호 = 0;
신규청산신호 = 0;
청산가격출력 = 0;
실현틱손익 = 0;
실현원화손익 = 0;

If 일분봉여부 == 0 Then
{
    현재포지션 = 0;
    저장진입가격 = 0;
    저장누적원화손익 = 0;
    보유봉수 = 0;
}
Else
{
    If 새세션여부 == 1 Then
    {
        현재포지션 = 0;
        저장진입가격 = 0;
        저장누적원화손익 = 0;
        보유봉수 = 0;
        이전통합상태 = 0;
    }
    Else
    {
        현재포지션 = 현재포지션[1];
        저장진입가격 = 저장진입가격[1];
        저장누적원화손익 = 저장누적원화손익[1];
        보유봉수 = 보유봉수[1];
        이전통합상태 = 통합상태입력[1];
    };

    /* 기존 포지션의 청산을 진입보다 먼저 처리한다. */
    If 현재포지션 == 1 &&
       (청산신호입력 == 1 || 청산신호입력 == 2) Then
    {
        계산가격차이 = C - 저장진입가격;
        계산틱손익 = 계산가격차이 / 틱크기;
        계산원화손익 = 계산틱손익 * 틱가치;

        신규청산신호 = 1;
        청산가격출력 = C;
        실현틱손익 = 계산틱손익;
        실현원화손익 = 계산원화손익;
        저장누적원화손익 = 저장누적원화손익 + 계산원화손익;

        현재포지션 = 0;
        저장진입가격 = 0;
        보유봉수 = 0;
    };

    If 현재포지션 == -1 &&
       (청산신호입력 == -1 || 청산신호입력 == 2) Then
    {
        계산가격차이 = 저장진입가격 - C;
        계산틱손익 = 계산가격차이 / 틱크기;
        계산원화손익 = 계산틱손익 * 틱가치;

        신규청산신호 = -1;
        청산가격출력 = C;
        실현틱손익 = 계산틱손익;
        실현원화손익 = 계산원화손익;
        저장누적원화손익 = 저장누적원화손익 + 계산원화손익;

        현재포지션 = 0;
        저장진입가격 = 0;
        보유봉수 = 0;
    };

    /* 청산한 봉에서는 즉시 반대 방향으로 재진입하지 않는다. */
    If 현재포지션 == 0 && 신규청산신호 == 0 Then
    {
        If 통합상태입력 == 100 && 이전통합상태 <> 100 Then
        {
            현재포지션 = 1;
            저장진입가격 = C;
            신규진입신호 = 1;
            보유봉수 = 0;
        };

        If 통합상태입력 == -100 && 이전통합상태 <> -100 Then
        {
            현재포지션 = -1;
            저장진입가격 = C;
            신규진입신호 = -1;
            보유봉수 = 0;
        };
    };

    If 현재포지션 <> 0 Then
        보유봉수 = 보유봉수 + 1;
};

포지션상태 = 현재포지션;
진입가격출력 = 저장진입가격;
누적원화손익 = 저장누적원화손익;

Value1 = 현재포지션;
WSF_1m_TradeLifecycleV2 = Value1;
