                꺾임 = 0; 반대기울기최대 = 0;
            };
        };

        If 신호방향 != 0 Then
        {
            중기준사용 = 중가격기준;
            If 구조등급상향 == 1 && 구조중 == 1 Then 중기준사용 = Max(중가격기준,구조중가격기준);
            신호등급 = 1;
            If 가격조정비 <= 강가격기준 && 시간조정비 >= 강시간기준 Then 신호등급 = 3;
            Else If 가격조정비 <= 중기준사용 Then 신호등급 = 2;
            If 시작가B만갱신 != 1 || 신호종류 == 2 Then 추세시작가 = 조정극값;
            조정중 = 0; 꺾임 = 0;
        };
    };
};
