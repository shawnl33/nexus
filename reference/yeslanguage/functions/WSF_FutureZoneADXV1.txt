Input : 미래방향입력(Numeric),
        매물대위치입력(Numeric),
        ADX입력(Numeric),
        ADX기준입력(Numeric),
        통합상태(NumericRef),
        매물대필터상태(NumericRef),
        ADX상태(NumericRef);

Var : 미래방향(0),
      매물대위치(0),
      ADX값(0),
      ADX기준값(0);

/*
   입력 규칙

   미래방향입력
      1  : 미래곡선 상승
      0  : 미래곡선 중립
     -1  : 미래곡선 하락

   매물대위치입력
      1  : 상승에 유리한 위치
      0  : 매물대 내부 또는 중립
     -1  : 하락에 유리한 위치

   중요
   1. 미래곡선만 최종 방향을 결정한다.
   2. 매물대와 ADX는 방향을 뒤집지 않는다.
   3. 매물대 충돌 또는 ADX 약세면 확인 상태를 관찰 상태로 낮춘다.

   출력 통합상태
      100  : 매수 확인
       50  : 매수 관찰
        0  : 중립
      -50  : 매도 관찰
     -100  : 매도 확인
*/

미래방향 = IFF(미래방향입력 > 0, 1,
            IFF(미래방향입력 < 0, -1, 0));

매물대위치 = IFF(매물대위치입력 > 0, 1,
              IFF(매물대위치입력 < 0, -1, 0));

ADX값 = Abs(ADX입력);
ADX기준값 = Abs(ADX기준입력);

If ADX기준값 < 1 Then
    ADX기준값 = 1;

통합상태 = 0;
매물대필터상태 = 0;
ADX상태 = 0;

If ADX값 >= ADX기준값 Then
    ADX상태 = 1;

/* 상승 방향에서 매물대 위치 판정 */
If 미래방향 == 1 Then
{
    If 매물대위치 >= 0 Then
        매물대필터상태 = 1;
    Else
        매물대필터상태 = -1;

    If 매물대필터상태 == 1 && ADX상태 == 1 Then
        통합상태 = 100;
    Else
        통합상태 = 50;
};

/* 하락 방향에서 매물대 위치 판정 */
If 미래방향 == -1 Then
{
    If 매물대위치 <= 0 Then
        매물대필터상태 = -1;
    Else
        매물대필터상태 = 1;

    If 매물대필터상태 == -1 && ADX상태 == 1 Then
        통합상태 = -100;
    Else
        통합상태 = -50;
};

Value1 = 통합상태;
WSF_FutureZoneADXV1 = Value1;
