Input : 가치영역배수입력(Numeric),
        최소봉수입력(Numeric),
        매물대중심(NumericRef),
        매물대상단(NumericRef),
        매물대하단(NumericRef),
        매물대위치(NumericRef),
        매물대유효(NumericRef);

Var : 일분봉여부(0),
      새장여부(0),
      세션봉수(0),
      대표가격(0),
      거래가중치(0),
      누적거래량(0),
      누적가격거래량(0),
      누적편차거래량(0),
      평균편차(0),
      가치영역배수(0),
      최소봉수(0);

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

가치영역배수 = Abs(가치영역배수입력);
If 가치영역배수 <= 0 Then
    가치영역배수 = 1.5;

최소봉수 = Abs(최소봉수입력);
If 최소봉수 < 1 Then
    최소봉수 = 1;

매물대중심 = 0;
매물대상단 = 0;
매물대하단 = 0;
매물대위치 = 0;
매물대유효 = 0;

If 일분봉여부 == 1 Then
{
    새장여부 = 0;

    If CurrentBar == 1 Then
        새장여부 = 1;

    If CurrentBar > 1 && BDate <> BDate[1] Then
        새장여부 = 1;

    대표가격 = (H + L + C) / 3;
    거래가중치 = V;

    If 거래가중치 <= 0 Then
        거래가중치 = 1;

    If 새장여부 == 1 Then
    {
        세션봉수 = 1;
        누적거래량 = 거래가중치;
        누적가격거래량 = 대표가격 * 거래가중치;
        누적편차거래량 = 0;
        매물대중심 = 대표가격;
    }
    Else
    {
        세션봉수 = 세션봉수[1] + 1;
        누적거래량 = 누적거래량[1] + 거래가중치;
        누적가격거래량 = 누적가격거래량[1]
                           + 대표가격 * 거래가중치;

        매물대중심 = 누적가격거래량 / 누적거래량;

        누적편차거래량 = 누적편차거래량[1]
                         + Abs(대표가격 - 매물대중심) * 거래가중치;
    };

    If 누적거래량 > 0 Then
        평균편차 = 누적편차거래량 / 누적거래량;
    Else
        평균편차 = 0;

    If 평균편차 < PriceScale Then
        평균편차 = PriceScale;

    매물대상단 = 매물대중심 + 평균편차 * 가치영역배수;
    매물대하단 = 매물대중심 - 평균편차 * 가치영역배수;

    If 세션봉수 >= 최소봉수 Then
    {
        매물대유효 = 1;

        If C > 매물대상단 Then
            매물대위치 = 1;
        Else If C < 매물대하단 Then
            매물대위치 = -1;
        Else
            매물대위치 = 0;
    };
};

Value1 = 매물대위치;
WSF_1m_SessionMarketProfileV1 = Value1;
