Input : 방향입력(Numeric),
        상태입력(Numeric),
        유효입력(Numeric),
        표준방향(NumericRef),
        표준상태(NumericRef),
        표준유효(NumericRef);

//======================================================================
// WSF_CommonDirectionStateV1
//
// 공통 출력 규격
//   방향 : 1, 0, -1
//   상태 : 100, 50, 0, -50, -100
//   유효 : 1, 0
//
// 미래곡선 방향과 반대되는 상태는 중립으로 정리한다.
//======================================================================

표준방향 = 0;
표준상태 = 0;
표준유효 = 0;

If 유효입력 > 0 Then
{
    표준유효 = 1;

    표준방향 =
        IFF(방향입력 > 0, 1,
        IFF(방향입력 < 0, -1, 0));

    표준상태 =
        IFF(상태입력 == 100, 100,
        IFF(상태입력 == 50, 50,
        IFF(상태입력 == -50, -50,
        IFF(상태입력 == -100, -100, 0))));

    If 표준방향 == 1 && 표준상태 < 0 Then
        표준상태 = 0;

    If 표준방향 == -1 && 표준상태 > 0 Then
        표준상태 = 0;

    If 표준방향 == 0 Then
        표준상태 = 0;
};

Value1 = 표준방향;
WSF_CommonDirectionStateV1 = Value1;
