Input : 분봉예측방향1입력(Numeric),
        분봉예측방향2입력(Numeric),
        분봉예측방향3입력(Numeric),
        분봉회귀유효입력(Numeric),
        분봉회귀신뢰도입력(Numeric),
        일봉추세방향입력(Numeric),
        일봉추세상태입력(Numeric),
        일봉추세강도입력(Numeric),
        일봉추세유효입력(Numeric),
        분봉최소신뢰도입력(Numeric),
        일봉최소강도입력(Numeric),
        분봉합의방향(NumericRef),
        분봉합의상태(NumericRef),
        일치방향(NumericRef),
        일치상태(NumericRef),
        일치강도(NumericRef),
        일치유효(NumericRef);

//======================================================================
// WSF_1m_DailyAlignV1
// - 1분봉 5/10/15분 예측 중 2개 이상 같은 방향일 때 분봉 합의
// - 일봉 방향과 분봉 합의 방향이 같을 때만 매매 방향 유효
//
// 일치상태 : 2 강매수, 1 매수, 0 관망, -1 매도, -2 강매도
//======================================================================

Var : 상승개수(0), 하락개수(0), 기본강도(0);

분봉합의방향 = 0;
분봉합의상태 = 0;
일치방향 = 0;
일치상태 = 0;
일치강도 = 0;
일치유효 = 0;

상승개수 = 0;
하락개수 = 0;

If 분봉예측방향1입력 > 0 Then 상승개수 = 상승개수 + 1;
If 분봉예측방향2입력 > 0 Then 상승개수 = 상승개수 + 1;
If 분봉예측방향3입력 > 0 Then 상승개수 = 상승개수 + 1;

If 분봉예측방향1입력 < 0 Then 하락개수 = 하락개수 + 1;
If 분봉예측방향2입력 < 0 Then 하락개수 = 하락개수 + 1;
If 분봉예측방향3입력 < 0 Then 하락개수 = 하락개수 + 1;

If 상승개수 >= 2 Then
{
    분봉합의방향 = 1;
    분봉합의상태 = 1;
    If 상승개수 == 3 Then 분봉합의상태 = 2;
}

If 하락개수 >= 2 Then
{
    분봉합의방향 = -1;
    분봉합의상태 = -1;
    If 하락개수 == 3 Then 분봉합의상태 = -2;
}

If 분봉회귀유효입력 == 1 &&
   분봉회귀신뢰도입력 >= 분봉최소신뢰도입력 &&
   일봉추세유효입력 == 1 &&
   일봉추세강도입력 >= 일봉최소강도입력 &&
   분봉합의방향 != 0 &&
   분봉합의방향 == 일봉추세방향입력 Then
{
    일치유효 = 1;
    일치방향 = 분봉합의방향;

    기본강도 = Min(분봉회귀신뢰도입력*100,일봉추세강도입력);

    If Abs(분봉합의상태) == 1 Then 기본강도 = 기본강도*0.75;
    If Abs(일봉추세상태입력) == 1 Then 기본강도 = 기본강도*0.80;

    일치강도 = 기본강도;
    일치상태 = 일치방향;

    If Abs(분봉합의상태) == 2 &&
       Abs(일봉추세상태입력) == 2 &&
       일치강도 >= 60 Then
        일치상태 = 일치방향*2;
}

Value1 = 1;
WSF_1m_DailyAlignV1 = Value1;
