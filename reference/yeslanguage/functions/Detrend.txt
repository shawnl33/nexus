Inputs	: Price(NumericSeries), Length(NumericSimple);
Var		: AvgVal(0), DFact(0);

AvgVal 	= Ma(Price, Length);
DFact 	= (Length * 0.5) + 1;

If CurrentBar > DFact Then 
	Detrend = Price - AvgVal[DFact];
Else
	Detrend = 0;
