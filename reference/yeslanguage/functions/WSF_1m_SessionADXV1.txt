Input : ADX기간입력(Numeric),
        세션ADX(NumericRef),
        상승DI(NumericRef),
        하락DI(NumericRef),
        ADX유효(NumericRef);

Var : 일분봉여부(0),
      기간값(0),
      세션봉수(0),
      원상승이동(0),
      원하락이동(0),
      상승DM(0),
      하락DM(0),
      실제범위(0),
      평활TR(0),
      평활상승DM(0),
      평활하락DM(0),
      계산상승DI(0),
      계산하락DI(0),
      DX값(0),
      DX누적(0),
      계산ADX(0),
      계산유효(0);

/* 1분봉에서만 계산한다. */
일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

기간값 = Round(Abs(ADX기간입력), 0);
If 기간값 < 2 Then
    기간값 = 2;

/* 출력값은 매 봉 먼저 초기화한다. */
세션ADX = 0;
상승DI = 0;
하락DI = 0;
ADX유효 = 0;

If 일분봉여부 == 1 Then
{
    /* 날짜가 바뀌면 전일 값을 사용하지 않고 새로 시작한다. */
    If BDate <> BDate[1] Then
    {
        세션봉수 = 1;
        원상승이동 = 0;
        원하락이동 = 0;
        상승DM = 0;
        하락DM = 0;
        실제범위 = H - L;
        평활TR = 실제범위;
        평활상승DM = 0;
        평활하락DM = 0;
        계산상승DI = 0;
        계산하락DI = 0;
        DX값 = 0;
        DX누적 = 0;
        계산ADX = 0;
        계산유효 = 0;
    }
    Else
    {
        세션봉수 = 세션봉수[1] + 1;

        원상승이동 = H - H[1];
        원하락이동 = L[1] - L;

        상승DM = IFF(원상승이동 > 원하락이동 && 원상승이동 > 0,
                     원상승이동, 0);
        하락DM = IFF(원하락이동 > 원상승이동 && 원하락이동 > 0,
                     원하락이동, 0);

        실제범위 = Max(H - L,
                       Max(Abs(H - C[1]), Abs(L - C[1])));

        /* 세션 초반에는 합계를 만들고 이후 Wilder 방식으로 평활한다. */
        If 세션봉수 <= 기간값 Then
        {
            평활TR = 평활TR[1] + 실제범위;
            평활상승DM = 평활상승DM[1] + 상승DM;
            평활하락DM = 평활하락DM[1] + 하락DM;
        }
        Else
        {
            평활TR = 평활TR[1] - 평활TR[1] / 기간값 + 실제범위;
            평활상승DM = 평활상승DM[1] - 평활상승DM[1] / 기간값 + 상승DM;
            평활하락DM = 평활하락DM[1] - 평활하락DM[1] / 기간값 + 하락DM;
        };

        If 평활TR > 0 Then
        {
            계산상승DI = 평활상승DM / 평활TR * 100;
            계산하락DI = 평활하락DM / 평활TR * 100;
        }
        Else
        {
            계산상승DI = 0;
            계산하락DI = 0;
        };

        If 계산상승DI + 계산하락DI > 0 Then
        {
            DX값 = Abs(계산상승DI - 계산하락DI) /
                   (계산상승DI + 계산하락DI) * 100;
        }
        Else
        {
            DX값 = 0;
        };

        /* ADX기간만큼 DX를 모은 뒤 최초 ADX를 만든다. */
        If 세션봉수 > 기간값 && 세션봉수 < 기간값 * 2 Then
        {
            DX누적 = DX누적[1] + DX값;
            계산ADX = 0;
            계산유효 = 0;
        }
        Else If 세션봉수 == 기간값 * 2 Then
        {
            DX누적 = DX누적[1] + DX값;
            계산ADX = DX누적 / 기간값;
            계산유효 = 1;
        }
        Else If 세션봉수 > 기간값 * 2 Then
        {
            DX누적 = DX누적[1];
            계산ADX = (계산ADX[1] * (기간값 - 1) + DX값) / 기간값;
            계산유효 = 1;
        };
    };

    If 계산유효 == 1 Then
    {
        세션ADX = 계산ADX;
        상승DI = 계산상승DI;
        하락DI = 계산하락DI;
        ADX유효 = 1;
    };
};

Value1 = ADX유효;
WSF_1m_SessionADXV1 = Value1;
