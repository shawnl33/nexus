// Korean chart calendar date YYYYMMDD and time HHMMSS. No timezone conversion.
Input : 날짜입력(Numeric), 시각입력(Numeric);
Var : 연도(0), 월(0), 일(0), 기준날짜(0), 구간(0);
일 = 날짜입력 % 100;
월 = ((날짜입력-일)/100) % 100;
연도 = (날짜입력-월*100-일)/10000;
기준날짜 = 날짜입력;
구간 = 0;
If 시각입력 >= 153000 Then 구간 = 1;
If 시각입력 < 70000 Then
{
    일 = 일-1;
    If 일 == 0 Then
    {
        월 = 월-1;
        If 월 == 0 Then { 월 = 12; 연도 = 연도-1; };
        일 = 31;
        If 월 == 4 || 월 == 6 || 월 == 9 || 월 == 11 Then 일 = 30;
        If 월 == 2 Then
        {
            일 = 28;
            If 연도 % 4 == 0 && (연도 % 100 != 0 || 연도 % 400 == 0) Then 일 = 29;
        };
    };
    기준날짜 = 연도*10000+월*100+일;
    구간 = 1;
};
WSF_FXSessionKeyV1 = 기준날짜*2+구간;
