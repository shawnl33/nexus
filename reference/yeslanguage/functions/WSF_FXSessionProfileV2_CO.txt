// WSF_FXSessionProfileV2_CO  (_CO = Claude Opus 작성)
// WSF_FXSessionProfileV1_CO + 계산기간 입력 (V1은 보존)
// - 계산기간입력 = 0 : V1과 같음. 세션 시작부터 누적한 거래량가중 중심 ± 평균편차
// - 계산기간입력 = N : 최근 N봉(세션 안, Min(N,세션봉수))만으로 중심과 평균편차 계산
// - 1차 상하단 = 중심 ± 평균편차 x 가치영역배수, 2차 = 그 2배
// - 매물대5상태: 2차상단 위 2, 1차상단 위 1, 안 0, 1차하단 아래 -1, 2차하단 아래 -2
// 반환: 매물대유효 (1=유효, 0=준비중/1분봉 아님)
Input : 세션초기화(Numeric),
        가치영역배수입력(Numeric),
        최소봉수입력(Numeric),
        계산기간입력(Numeric),
        매물대중심(NumericRef),
        매물대상단1(NumericRef),
        매물대하단1(NumericRef),
        매물대상단2(NumericRef),
        매물대하단2(NumericRef),
        매물대5상태(NumericRef),
        매물대유효(NumericRef);
Var : 세션봉수(0),
      대표가격(0),
      거래가중치(0),
      누적거래량(0),
      누적가격거래량(0),
      누적편차거래량(0),
      중심값(0),
      평균편차(0),
      가치영역배수(0),
      최소봉수(0),
      계산창(0),
      j(0),
      창거래량(0),
      창가격거래량(0),
      창편차거래량(0);

가치영역배수 = Abs(가치영역배수입력);
If 가치영역배수 <= 0 Then 가치영역배수 = 1.5;
최소봉수 = Abs(최소봉수입력);
If 최소봉수 < 1 Then 최소봉수 = 1;

매물대중심 = 0;
매물대상단1 = 0;
매물대하단1 = 0;
매물대상단2 = 0;
매물대하단2 = 0;
매물대5상태 = 0;
매물대유효 = 0;

If DataCompress == 2 && BarInterval == 1 Then
{
    대표가격 = (H + L + C) / 3;
    거래가중치 = V;
    // 거래량이 없는 봉도 가격 흐름에서 빠지지 않게 가중치 1
    If 거래가중치 <= 0 Then 거래가중치 = 1;

    // 세션 누적 (V1과 같은 계산)
    If CurrentBar == 1 || 세션초기화 == 1 Then
    {
        세션봉수 = 1;
        누적거래량 = 거래가중치;
        누적가격거래량 = 대표가격 * 거래가중치;
        누적편차거래량 = 0;
        중심값 = 대표가격;
    }
    Else
    {
        세션봉수 = 세션봉수[1] + 1;
        누적거래량 = 누적거래량[1] + 거래가중치;
        누적가격거래량 = 누적가격거래량[1] + 대표가격 * 거래가중치;
        중심값 = 누적가격거래량 / 누적거래량;
        누적편차거래량 = 누적편차거래량[1] + Abs(대표가격 - 중심값) * 거래가중치;
    };
    평균편차 = 0;
    If 누적거래량 > 0 Then 평균편차 = 누적편차거래량 / 누적거래량;

    // 최근 N봉 모드: 중심과 평균편차를 최근 창으로 다시 계산
    If 계산기간입력 > 0 Then
    {
        계산창 = Min(Round(계산기간입력,0),세션봉수);
        창거래량 = 0;
        창가격거래량 = 0;
        For j = 0 To 계산창-1
        {
            창거래량 = 창거래량 + 거래가중치[j];
            창가격거래량 = 창가격거래량 + 대표가격[j] * 거래가중치[j];
        };
        If 창거래량 > 0 Then
        {
            중심값 = 창가격거래량 / 창거래량;
            창편차거래량 = 0;
            For j = 0 To 계산창-1
            {
                창편차거래량 = 창편차거래량 + Abs(대표가격[j] - 중심값) * 거래가중치[j];
            };
            평균편차 = 창편차거래량 / 창거래량;
        };
    };

    // 밴드가 완전히 겹치는 것을 막는다
    If 평균편차 < PriceScale Then 평균편차 = PriceScale;

    매물대중심 = 중심값;
    매물대상단1 = 중심값 + 평균편차 * 가치영역배수;
    매물대하단1 = 중심값 - 평균편차 * 가치영역배수;
    매물대상단2 = 중심값 + 평균편차 * 가치영역배수 * 2;
    매물대하단2 = 중심값 - 평균편차 * 가치영역배수 * 2;

    If 세션봉수 >= 최소봉수 Then
    {
        매물대유효 = 1;
        If C > 매물대상단2 Then 매물대5상태 = 2;
        Else If C > 매물대상단1 Then 매물대5상태 = 1;
        Else If C < 매물대하단2 Then 매물대5상태 = -2;
        Else If C < 매물대하단1 Then 매물대5상태 = -1;
        Else 매물대5상태 = 0;
    };
};

Value1 = 매물대유효;
WSF_FXSessionProfileV2_CO = Value1;
