Input : 회귀선입력(Numeric),
        회귀기울기입력(Numeric),
        회귀유효입력(Numeric),
        회귀신뢰도입력(Numeric),
        회귀잔차입력(Numeric),
        현재가격입력(Numeric),
        최소신뢰도입력(Numeric),
        일봉추세방향(NumericRef),
        일봉추세상태(NumericRef),
        일봉추세강도(NumericRef),
        일봉추세유효(NumericRef);

//======================================================================
// OSF_Daily_LinRegTrendV1
// - 일봉 선형회귀 기울기를 최우선으로 추세 방향 판정
// - R²가 최소신뢰도 미만이면 관망
// - 현재가격의 회귀선 상하 위치로 강/약 상태 구분
//
// 출력
// 일봉추세방향 :  1 상승, 0 관망, -1 하락
// 일봉추세상태 :  2 강상승, 1 상승, 0 관망, -1 하락, -2 강하락
// 일봉추세강도 :  0 ~ 100
// 일봉추세유효 :  1 유효, 0 준비/관망
//======================================================================

Var : 기준잔차(0), 최소기울기(0), 기울기비율(0),
      기울기강도(0), 신뢰강도(0);

일봉추세방향 = 0;
일봉추세상태 = 0;
일봉추세강도 = 0;
일봉추세유효 = 0;

기준잔차 = Max(PriceScale,회귀잔차입력);
최소기울기 = Max(PriceScale,기준잔차*0.02);

If 회귀유효입력 == 1 &&
   회귀신뢰도입력 >= 최소신뢰도입력 &&
   Abs(회귀기울기입력) >= 최소기울기 Then
{
    일봉추세유효 = 1;

    기울기비율 = Abs(회귀기울기입력)/기준잔차;
    기울기강도 = Min(1,기울기비율*5);
    신뢰강도 = Min(1,Max(0,회귀신뢰도입력));

    일봉추세강도 = (신뢰강도*0.60 + 기울기강도*0.40)*100;

    If 회귀기울기입력 > 0 Then
    {
        일봉추세방향 = 1;

        If 현재가격입력 >= 회귀선입력 Then
        {
            일봉추세상태 = 2;
        }
        Else
        {
            일봉추세상태 = 1;
            일봉추세강도 = 일봉추세강도*0.65;
        }
    }
    Else
    {
        일봉추세방향 = -1;

        If 현재가격입력 <= 회귀선입력 Then
        {
            일봉추세상태 = -2;
        }
        Else
        {
            일봉추세상태 = -1;
            일봉추세강도 = 일봉추세강도*0.65;
        }
    }
}

Value1 = 1;
OSF_Daily_LinRegTrendV1 = Value1;
