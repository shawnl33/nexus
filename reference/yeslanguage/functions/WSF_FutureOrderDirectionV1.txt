Input : 미래방향입력(Numeric),
        관심기준입력(Numeric),
        강세기준입력(Numeric),
        지속확인봉수입력(Numeric),
        부호반전입력(Numeric),
        최종방향출력(NumericRef),
        호가상태출력(NumericRef),
        호가점수출력(NumericRef),
        호가유효출력(NumericRef);

Var : 일분봉여부(0),
      미래방향(0),
      호가방향점수(0),
      호가기울기3(0),
      호가방향상태(0),
      호가유효(0),
      확인필요봉수(0),
      매수확인수(0),
      매도확인수(0),
      계산최종방향(0);

/*
   출력값 의미
   +100 : 미래곡선 상승 + 호가 상승 확인 완료
    +50 : 미래곡선 상승, 호가 미확인 또는 중립
      0 : 미래곡선과 유효 호가가 충돌
    -50 : 미래곡선 하락, 호가 미확인 또는 중립
   -100 : 미래곡선 하락 + 호가 하락 확인 완료

   호가 자료는 1분봉에서만 사용한다.
   일봉 및 다른 주기에서는 미래곡선 방향을 그대로 통과시킨다.
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

미래방향 =
    IFF(미래방향입력 > 0, 1,
    IFF(미래방향입력 < 0, -1, 0));

확인필요봉수 = IntPortion(Abs(지속확인봉수입력));
If 확인필요봉수 < 1 Then
    확인필요봉수 = 1;

호가방향점수 = 0;
호가기울기3 = 0;
호가방향상태 = 0;
호가유효 = 0;
계산최종방향 = 0;

If BDate <> BDate[1] Then
{
    매수확인수 = 0;
    매도확인수 = 0;
};

If 일분봉여부 == 1 Then
{
    WSF_OrderBookDirectionV1(
        관심기준입력,
        강세기준입력,
        부호반전입력,
        호가방향점수,
        호가기울기3,
        호가방향상태,
        호가유효
    );

    If 미래방향 > 0 Then
    {
        매도확인수 = 0;

        If 호가유효 == 1 && 호가방향상태 > 0 Then
            매수확인수 = 매수확인수[1] + 1;
        Else
            매수확인수 = 0;

        If 호가유효 == 1 && 호가방향상태 < 0 Then
            계산최종방향 = 0;
        Else If 매수확인수 >= 확인필요봉수 Then
            계산최종방향 = 100;
        Else
            계산최종방향 = 50;
    }
    Else If 미래방향 < 0 Then
    {
        매수확인수 = 0;

        If 호가유효 == 1 && 호가방향상태 < 0 Then
            매도확인수 = 매도확인수[1] + 1;
        Else
            매도확인수 = 0;

        If 호가유효 == 1 && 호가방향상태 > 0 Then
            계산최종방향 = 0;
        Else If 매도확인수 >= 확인필요봉수 Then
            계산최종방향 = -100;
        Else
            계산최종방향 = -50;
    }
    Else
    {
        매수확인수 = 0;
        매도확인수 = 0;
        계산최종방향 = 0;
    };
}
Else
{
    매수확인수 = 0;
    매도확인수 = 0;

    If 미래방향 > 0 Then
        계산최종방향 = 100;
    Else If 미래방향 < 0 Then
        계산최종방향 = -100;
    Else
        계산최종방향 = 0;
};

최종방향출력 = 계산최종방향;
호가상태출력 = 호가방향상태;
호가점수출력 = 호가방향점수;
호가유효출력 = 호가유효;

Value1 = 1;
WSF_FutureOrderDirectionV1 = Value1;
