Input : 가치영역배수입력(Numeric),
        최소봉수입력(Numeric),
        매물대중심(NumericRef),
        매물대상단1(NumericRef),
        매물대상단2(NumericRef),
        매물대하단1(NumericRef),
        매물대하단2(NumericRef),
        매물대위치(NumericRef),
        매물대유효(NumericRef);

Var : 계산중심(0),
      계산상단1(0),
      계산하단1(0),
      계산위치(0),
      계산유효(0);

매물대중심 = 0;
매물대상단1 = 0;
매물대상단2 = 0;
매물대하단1 = 0;
매물대하단2 = 0;
매물대위치 = 0;
매물대유효 = 0;

WSF_1m_SessionMarketProfileV1(
    가치영역배수입력,
    최소봉수입력,
    계산중심,
    계산상단1,
    계산하단1,
    계산위치,
    계산유효
);

If 계산유효 == 1 Then
{
    매물대중심 = 계산중심;
    매물대상단1 = 계산상단1;
    매물대하단1 = 계산하단1;

    매물대상단2 =
        계산중심 + (계산상단1 - 계산중심) * 2;

    매물대하단2 =
        계산중심 - (계산중심 - 계산하단1) * 2;

    매물대위치 = 계산위치;
    매물대유효 = 1;
};

Value1 = 매물대위치;
WSF_1m_SessionMarketProfileV2 = Value1;
