// WSF_FXEntryCandV1_CO  (_CO = Claude Opus 작성)
// 진입 후보 판정 함수. 판정방식으로 진입후보 V1~V4 규칙을 고른다 (검증 지표에서 버전 비교용).
//   1 = 진입후보V1: 통합상태 첫 ±100 또는 ±100 중 같은 방향 해제. 등급 A(즉시해제)/B(지연확정)/C(해제 없음)
//   2 = 진입후보V2: 박스 돌파 봉 즉시 A + 통합 첫 ±100은 B/C
//   3 = 진입후보V3: 돌파 뒤 돌파확인봉수 확인 + DI 일치일 때 A
//   4 = 진입후보V4: V3 + 변동성 확장 확인 (확인된 돌파 뒤 확장확인봉수 이내 압축이 풀려야 A, 박스 안 복귀 시 취소)
// 자동값: 등급C사용입력 < 0 이면 방식1만 C 사용, 돌파여유틱입력 < 0 이면 방식2는 1틱, 방식3·4는 2틱
// 출력: 후보방향(+1/-1/0), 후보등급(3=A, 2=B, 1=C), 통합상태(±100/±50/0)
// 반환: 후보방향
Input : 세션초기화(Numeric), 세션봉수(Numeric), 판정방식(Numeric),
        예측변수(Numeric), 가치영역배수(Numeric), 최소봉수(Numeric),
        ADX기간(Numeric), ADX추세기준(Numeric), ADX강세기준(Numeric), 매물대기준모드(Numeric),
        폭계산봉수(Numeric), 비율기준봉수(Numeric), 좁음기준(Numeric), 단계1봉수(Numeric),
        방향확정봉수(Numeric), 확정봉판정(Numeric), 해제유효봉수(Numeric),
        등급C사용입력(Numeric), 돌파여유틱입력(Numeric), 돌파통합최소(Numeric),
        돌파확인봉수입력(Numeric), DI일치필수입력(Numeric), 확장확인봉수(Numeric),
        후보방향출력(NumericRef), 후보등급출력(NumericRef), 통합상태출력(NumericRef);
Var : 방식(0), 등급C사용(0), 돌파여유틱(0), 돌파확인봉수(0), DI일치필수(0),
      곡선하(0), 곡선상(0), 곡선예측가격(0), 곡선예측변화(0), 곡선예측방향(0), 미래방향(0),
      매물대중심(0), 매물대상단1(0), 매물대하단1(0), 매물대상단2(0), 매물대하단2(0),
      매물대5상태(0), 매물대유효(0),
      세션ADX(0), 상승DI(0), 하락DI(0), ADX유효(0),
      통합상태(0), 매물대상태(0), ADX상태(0), DI방향(0),
      압축해제(0), 압축지속봉수(0), 폭비율(0), 즉시해제방향(0), 지연확정방향(0), 해제길이(0),
      박스돌파방향(0), 박스돌파길이(0), 돌파박스상단(0), 돌파박스하단(0),
      최근해제방향(0), 최근해제종류(0), 최근해제경과(999), 차단방향(0),
      예비방향(0), 예비상단(0), 예비하단(0), 예비확인(0),
      확장대기방향(0), 확장상단(0), 확장하단(0), 확장경과(0),
      확인방향(0), A판정방향(0), DI통과(0), 트리거방향(0), 후보방향(0), 후보등급(0);

방식 = Round(판정방식,0);
If 방식 < 1 Then 방식 = 1;
If 방식 > 4 Then 방식 = 4;
등급C사용 = 등급C사용입력;
If 등급C사용입력 < 0 Then 등급C사용 = IFF(방식 == 1,1,0);
돌파여유틱 = 돌파여유틱입력;
If 돌파여유틱입력 < 0 Then 돌파여유틱 = IFF(방식 == 2,1,2);
돌파확인봉수 = 0; DI일치필수 = 0;
If 방식 >= 3 Then
{
    돌파확인봉수 = 돌파확인봉수입력; DI일치필수 = DI일치필수입력;
};

후보방향 = 0; 후보등급 = 0; 통합상태 = 0;

If DataCompress == 2 && BarInterval == 1 Then
{
    // 1분 통합판정
    WSF_FXCurveV1(세션초기화,세션봉수,예측변수,
                  곡선하,곡선상,곡선예측가격,곡선예측변화,곡선예측방향);
    미래방향 = IFF(곡선예측방향 > 0, 1, IFF(곡선예측방향 < 0, -1, 0));
    If 세션초기화 == 1 Then 미래방향 = 0;
    WSF_FXSessionProfileV1_CO(세션초기화,가치영역배수,최소봉수,
        매물대중심,매물대상단1,매물대하단1,매물대상단2,매물대하단2,
        매물대5상태,매물대유효);
    WSF_FXADXV1(세션초기화,세션봉수,ADX기간,세션ADX,상승DI,하락DI,ADX유효);
    WSF_FXDecisionV2_CO(미래방향,매물대5상태,매물대유효,매물대중심,C,매물대기준모드,
        세션ADX,상승DI,하락DI,ADX유효,ADX추세기준,ADX강세기준,
        통합상태,매물대상태,ADX상태,DI방향);

    // 압축 해제·박스 돌파
    압축해제 = WSF_FXSqueezeV3_CO(세션초기화,세션봉수,가치영역배수,최소봉수,
        폭계산봉수,비율기준봉수,좁음기준,단계1봉수,방향확정봉수,확정봉판정,돌파여유틱,
        압축지속봉수,폭비율,즉시해제방향,지연확정방향,해제길이,
        박스돌파방향,박스돌파길이,돌파박스상단,돌파박스하단);

    // 상태 이어받기 (같은 봉 재평가는 직전 봉 상태에서 시작)
    If CurrentBar > 1 Then
    {
        최근해제방향 = 최근해제방향[1]; 최근해제종류 = 최근해제종류[1]; 최근해제경과 = 최근해제경과[1];
        차단방향 = 차단방향[1];
        예비방향 = 예비방향[1]; 예비상단 = 예비상단[1]; 예비하단 = 예비하단[1]; 예비확인 = 예비확인[1];
        확장대기방향 = 확장대기방향[1]; 확장상단 = 확장상단[1]; 확장하단 = 확장하단[1]; 확장경과 = 확장경과[1];
    };
    If 세션초기화 == 1 Then
    {
        최근해제방향 = 0; 최근해제종류 = 0; 최근해제경과 = 999; 차단방향 = 0;
        예비방향 = 0; 예비확인 = 0; 확장대기방향 = 0;
    };

    // 최근 이벤트 기억 (종류 1=박스돌파·즉시해제, 2=지연확정). 방식1은 박스돌파를 쓰지 않음.
    최근해제경과 = 최근해제경과 + 1;
    If 방식 >= 2 && 박스돌파방향 != 0 Then
    {
        최근해제방향 = 박스돌파방향; 최근해제종류 = 1; 최근해제경과 = 0;
    }
    Else If 즉시해제방향 != 0 Then
    {
        최근해제방향 = 즉시해제방향; 최근해제종류 = 1; 최근해제경과 = 0;
    }
    Else If 지연확정방향 != 0 Then
    {
        최근해제방향 = 지연확정방향; 최근해제종류 = 2; 최근해제경과 = 0;
    };

    // 중복 금지 해제
    If 차단방향 == 1 && 통합상태 <= 0 Then 차단방향 = 0;
    If 차단방향 == -1 && 통합상태 >= 0 Then 차단방향 = 0;

    If 세션초기화 == 0 Then
    {
        // ---- A 판정 (방식 2~4) ----
        확인방향 = 0; A판정방향 = 0;
        If 방식 >= 2 Then
        {
            // 돌파 확인
            If 예비방향 != 0 Then
            {
                If (예비방향 == 1 && C > 예비상단) || (예비방향 == -1 && C < 예비하단) Then
                {
                    예비확인 = 예비확인 + 1;
                    If 예비확인 >= 돌파확인봉수 Then
                    {
                        확인방향 = 예비방향;
                        예비방향 = 0;
                    };
                }
                Else
                {
                    예비방향 = 0; 예비확인 = 0;
                };
            };
            If 박스돌파방향 != 0 Then
            {
                예비상단 = 돌파박스상단; 예비하단 = 돌파박스하단;
                If 돌파확인봉수 <= 0 Then 확인방향 = 박스돌파방향;
                Else
                {
                    예비방향 = 박스돌파방향; 예비확인 = 0;
                };
            };

            If 방식 == 4 Then
            {
                // 변동성 확장 대기: 압축이 풀릴 때(압축지속봉수 0) 확정, 박스 안 복귀·기한 초과 시 취소
                If 확장대기방향 != 0 Then
                {
                    확장경과 = 확장경과 + 1;
                    If (확장대기방향 == 1 && C <= 확장상단) || (확장대기방향 == -1 && C >= 확장하단) Then
                        확장대기방향 = 0;
                    Else If 압축지속봉수 == 0 Then
                    {
                        A판정방향 = 확장대기방향;
                        확장대기방향 = 0;
                    }
                    Else If 확장경과 >= 확장확인봉수 Then
                        확장대기방향 = 0;
                };
                If 확인방향 != 0 Then
                {
                    If 압축지속봉수 == 0 Then A판정방향 = 확인방향;
                    Else
                    {
                        확장대기방향 = 확인방향; 확장경과 = 0;
                        확장상단 = 예비상단; 확장하단 = 예비하단;
                    };
                };
            }
            Else A판정방향 = 확인방향;

            DI통과 = 1;
            If DI일치필수 == 1 Then
            {
                DI통과 = 0;
                If ADX유효 == 1 && A판정방향 == 1 && 상승DI > 하락DI Then DI통과 = 1;
                If ADX유효 == 1 && A판정방향 == -1 && 하락DI > 상승DI Then DI통과 = 1;
            };
            If A판정방향 == 1 && 미래방향 == 1 && 통합상태 >= 돌파통합최소 && DI통과 == 1 && 차단방향 != 1 Then
            {
                후보방향 = 1; 후보등급 = 3;
            }
            Else If A판정방향 == -1 && 미래방향 == -1 && 통합상태 <= -돌파통합최소 && DI통과 == 1 && 차단방향 != -1 Then
            {
                후보방향 = -1; 후보등급 = 3;
            };
        };

        // ---- B/C 판정 (모든 방식) ----
        If 후보방향 == 0 Then
        {
            트리거방향 = 0;
            If 통합상태 == 100 && 통합상태[1] != 100 Then 트리거방향 = 1;
            Else If 통합상태 == -100 && 통합상태[1] != -100 Then 트리거방향 = -1;
            Else If 방식 == 1 && 통합상태 == 100 && 최근해제경과 == 0 && 최근해제방향 == 1 Then 트리거방향 = 1;
            Else If 방식 == 1 && 통합상태 == -100 && 최근해제경과 == 0 && 최근해제방향 == -1 Then 트리거방향 = -1;
            If 트리거방향 != 0 && 차단방향 != 트리거방향 Then
            {
                If 최근해제방향 != 0 && 최근해제경과 <= 해제유효봉수 Then
                {
                    If 최근해제방향 == 트리거방향 Then
                    {
                        후보방향 = 트리거방향;
                        후보등급 = 2;
                        If 방식 == 1 && 최근해제종류 == 1 Then 후보등급 = 3;
                    };
                }
                Else If 등급C사용 == 1 Then
                {
                    후보방향 = 트리거방향; 후보등급 = 1;
                };
            };
        };

        If 후보방향 != 0 Then 차단방향 = 후보방향;
    };
};

후보방향출력 = 후보방향;
후보등급출력 = 후보등급;
통합상태출력 = 통합상태;
Value1 = 후보방향;
WSF_FXEntryCandV1_CO = Value1;
