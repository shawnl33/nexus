// WSF_FXWaveAdjV1_CO  (_CO = Claude Opus 작성)
// 파동 조정 분석: 평탄선 기울기 추세 구간 뒤 조정(시간·가격)을 재고, 추세 재개 신호를 낸다. 상승·하락 대칭.
// 카운트는 WSF_FXFlatCountV2_CO (위치Cnt, 기울기Cnt) 사용.
//   추세: 기울기가 같은 방향으로 최소추세봉수 이상 → 추세방향 ±1, 추세L = 그 기울기 길이,
//         추세시작가 = 그 기울기 구간의 저가(상승)/고가(하락), 추세극점 = 최고가(상승)/최저가(하락)
//   조정 시작: 위치가 반대쪽이 되거나 기울기가 반대로 꺾인 봉
//   조정 중: 시간조정비 = 조정경과/추세L x100, 가격조정비 = |극점-조정극값|/|극점-시작가| x100,
//            반대기울기비 = 조정 중 가장 긴 반대 기울기 길이/추세L x100
//   전환: 조정극값이 추세시작가를 넘어감 (상승: 저점 붕괴, 하락: 고점 돌파) → 추세 해제
//   조정 종료(신호 없음): 종가가 추세극점을 넘어서 추세 재개
//   신호 종류 1 (A, 숨고르기 지속): 기울기가 한 번도 안 꺾이고 위치가 추세 쪽으로 복귀확인봉수 이상 돌아옴
//   신호 종류 2 (B, 기울기 재전환): 기울기가 최소반대봉수 이상 반대로 꺾였다가 추세 방향으로 다시 돌아선 첫 봉
//            (최소반대봉수 미만의 짧은 꺾임은 잡음으로 보고 꺾임 기록을 지움)
//   신호 등급: 3 강 = 가격조정 <= 강가격기준 && 시간조정 >= 강시간기준 / 2 중 = 가격조정 <= 중가격기준 / 1 약
//   신호 뒤: 조정 해제, 추세시작가 = 이번 조정극값 (새 파동 시작점). 추세L은 새 같은 방향 기울기가 최소추세봉수 이상이 되면 갱신.
// 출력: 추세방향, 조정중, 시간조정비, 가격조정비, 반대기울기비, 신호방향, 신호종류, 신호등급, 전환(끝난 추세 방향)
// 반환: 카운트 유효 1/0
Input : 세션초기화(Numeric), 평탄값(Numeric), 최소구간봉수(Numeric),
        최소추세봉수(Numeric), 최소반대봉수(Numeric), 복귀확인봉수(Numeric),
        강가격기준(Numeric), 강시간기준(Numeric), 중가격기준(Numeric),
        추세방향출력(NumericRef), 조정중출력(NumericRef),
        시간조정비출력(NumericRef), 가격조정비출력(NumericRef), 반대기울기비출력(NumericRef),
        신호방향출력(NumericRef), 신호종류출력(NumericRef), 신호등급출력(NumericRef), 전환출력(NumericRef);
Var : 카운트유효(0), 위치Cnt(0), 직전상승위치(0), 직전하락위치(0), 위치진폭(0),
      기울기Cnt(0), 직전상승기울기(0), 직전하락기울기(0),
      기울기부호(0), 기울기길이(0), 런고(0), 런저(0),
      추세방향(0), 추세L(0), 추세시작가(0), 추세극점(0),
      조정중(0), 조정경과(0), 조정극값(0), 반대기울기최대(0), 꺾임(0),
      시간조정비(0), 가격조정비(0), 반대기울기비(0), 추세폭(0),
      신호방향(0), 신호종류(0), 신호등급(0), 전환(0);

카운트유효 = WSF_FXFlatCountV2_CO(세션초기화,평탄값,최소구간봉수,
    위치Cnt,직전상승위치,직전하락위치,위치진폭,
    기울기Cnt,직전상승기울기,직전하락기울기);

// 같은 봉 재평가는 직전 봉 상태에서 시작
If CurrentBar > 1 Then
{
    런고 = 런고[1]; 런저 = 런저[1];
    추세방향 = 추세방향[1]; 추세L = 추세L[1]; 추세시작가 = 추세시작가[1]; 추세극점 = 추세극점[1];
    조정중 = 조정중[1]; 조정경과 = 조정경과[1]; 조정극값 = 조정극값[1];
    반대기울기최대 = 반대기울기최대[1]; 꺾임 = 꺾임[1];
};
If 세션초기화 == 1 Then
{
    런고 = 0; 런저 = 0;
    추세방향 = 0; 추세L = 0; 추세시작가 = 0; 추세극점 = 0;
    조정중 = 0; 조정경과 = 0; 조정극값 = 0; 반대기울기최대 = 0; 꺾임 = 0;
};

신호방향 = 0; 신호종류 = 0; 신호등급 = 0; 전환 = 0;
시간조정비 = 0; 가격조정비 = 0; 반대기울기비 = 0;
기울기부호 = IFF(기울기Cnt > 0,1,IFF(기울기Cnt < 0,-1,0));
기울기길이 = Abs(기울기Cnt);

If 카운트유효 == 1 && 세션초기화 == 0 && 기울기부호 != 0 Then
{
    // 현재 기울기 구간의 고저 (구간 첫 봉에서 새로 시작)
    If 기울기길이 == 1 Then
    {
        런고 = H; 런저 = L;
    }
    Else
    {
        런고 = Max(런고,H); 런저 = Min(런저,L);
    };

    If 추세방향 == 0 Then
    {
        // 추세 인정
        If 기울기길이 >= 최소추세봉수 Then
        {
            추세방향 = 기울기부호; 추세L = 기울기길이;
            추세시작가 = IFF(기울기부호 == 1,런저,런고);
            추세극점 = IFF(기울기부호 == 1,런고,런저);
            조정중 = 0;
        };
    }
    Else If 조정중 == 0 Then
    {
        // 추세 진행: 극점 갱신, 같은 방향 기울기가 기준 이상이면 길이 갱신
        If 추세방향 == 1 Then 추세극점 = Max(추세극점,H);
        Else 추세극점 = Min(추세극점,L);
        If 기울기부호 == 추세방향 && 기울기길이 >= 최소추세봉수 Then 추세L = 기울기길이;
        // 조정 시작
        If 위치Cnt*추세방향 < 0 || 기울기부호 == -추세방향 Then
        {
            조정중 = 1; 조정경과 = 0;
            조정극값 = IFF(추세방향 == 1,L,H);
            반대기울기최대 = 0; 꺾임 = 0;
        };
    };

    If 추세방향 != 0 && 조정중 == 1 Then
    {
        조정경과 = 조정경과+1;
        If 추세방향 == 1 Then 조정극값 = Min(조정극값,L);
        Else 조정극값 = Max(조정극값,H);
        If 기울기부호 == -추세방향 Then
        {
            꺾임 = 1;
            반대기울기최대 = Max(반대기울기최대,기울기길이);
        };
        If 추세L > 0 Then
        {
            시간조정비 = 조정경과/추세L*100;
            반대기울기비 = 반대기울기최대/추세L*100;
        };
        추세폭 = Abs(추세극점-추세시작가);
        If 추세폭 > 0 Then 가격조정비 = Abs(추세극점-조정극값)/추세폭*100;

        If (추세방향 == 1 && 조정극값 < 추세시작가) || (추세방향 == -1 && 조정극값 > 추세시작가) Then
        {
            // 전환: 파동 시작점 이탈
            전환 = 추세방향;
            추세방향 = 0; 조정중 = 0; 꺾임 = 0;
        }
        Else If (추세방향 == 1 && C > 추세극점) || (추세방향 == -1 && C < 추세극점) Then
        {
            // 극점 돌파로 조정 종료 (신호 없음)
            조정중 = 0; 꺾임 = 0;
            추세시작가 = 조정극값;
            If 추세방향 == 1 Then 추세극점 = Max(추세극점,H);
            Else 추세극점 = Min(추세극점,L);
        }
        Else If 꺾임 == 0 && 위치Cnt*추세방향 >= 복귀확인봉수 Then
        {
            신호방향 = 추세방향; 신호종류 = 1;
        }
        Else If 꺾임 == 1 && 기울기부호 == 추세방향 && 기울기길이 == 1 Then
        {
            If 반대기울기최대 >= 최소반대봉수 Then
            {
                신호방향 = 추세방향; 신호종류 = 2;
            }
            Else
            {
                // 짧은 꺾임은 잡음: 꺾임 기록 지움
                꺾임 = 0; 반대기울기최대 = 0;
            };
        };

        If 신호방향 != 0 Then
        {
            신호등급 = 1;
            If 가격조정비 <= 강가격기준 && 시간조정비 >= 강시간기준 Then 신호등급 = 3;
            Else If 가격조정비 <= 중가격기준 Then 신호등급 = 2;
            조정중 = 0; 꺾임 = 0;
            추세시작가 = 조정극값;
        };
    };
};

추세방향출력 = 추세방향;
조정중출력 = 조정중;
시간조정비출력 = 시간조정비;
가격조정비출력 = 가격조정비;
반대기울기비출력 = 반대기울기비;
신호방향출력 = 신호방향;
신호종류출력 = 신호종류;
신호등급출력 = 신호등급;
전환출력 = 전환;
Value1 = 카운트유효;
WSF_FXWaveAdjV1_CO = Value1;
