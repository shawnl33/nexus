Input : 포지션방향입력(Numeric),
        현재가격입력(Numeric),
        현재고가입력(Numeric),
        현재저가입력(Numeric),
        진입가격입력(Numeric),
        미래방향입력(Numeric),
        회귀이탈입력(Numeric),
        호가상태입력(Numeric),
        호가유효입력(Numeric),
        ADX입력(Numeric),
        ADX이전입력(Numeric),
        반대매물대접근입력(Numeric),
        가격단위입력(Numeric),
        고정손절틱입력(Numeric),
        수익보호시작틱입력(Numeric),
        수익보호폭틱입력(Numeric),
        청산요청(NumericRef),
        청산대상방향(NumericRef),
        청산강도(NumericRef),
        청산사유(NumericRef),
        평가손익틱(NumericRef),
        최고수익틱출력(NumericRef);

Var : 일분봉여부(0),
      포지션방향(0),
      가격단위(0),
      고정손절틱(0),
      수익보호시작틱(0),
      수익보호폭틱(0),
      최고유리가격(0),
      최고수익틱(0),
      수익후퇴틱(0),
      이전포지션방향(0),
      이전진입가격(0),
      호가역행(0),
      ADX약화(0),
      보조역행개수(0),
      신규포지션(0);

/*
   WSF_1m_ExitDecisionV1

   청산사유
     0 : 청산 없음
     1 : 미래곡선 반전
     2 : 회귀선 완전 이탈
     3 : 호가/ADX/반대 매물대 복합 역행
     4 : 고정 손절
     5 : 최고수익 대비 수익보호선 이탈

   회귀이탈입력
     매수 포지션에서 회귀선 아래 완전 이탈 : -1
     매도 포지션에서 회귀선 위 완전 이탈 :  1

   반대매물대접근입력
     반대편 강한 매물대에 접근했으면 1, 아니면 0
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

포지션방향 = IFF(포지션방향입력 > 0, 1,
               IFF(포지션방향입력 < 0, -1, 0));

가격단위 = Abs(가격단위입력);
If 가격단위 <= 0 Then
    가격단위 = PriceScale;
If 가격단위 <= 0 Then
    가격단위 = 1;

고정손절틱 = Abs(고정손절틱입력);
수익보호시작틱 = Abs(수익보호시작틱입력);
수익보호폭틱 = Abs(수익보호폭틱입력);

청산요청 = 0;
청산대상방향 = 0;
청산강도 = 0;
청산사유 = 0;
평가손익틱 = 0;
최고수익틱출력 = 0;

신규포지션 = 0;

If CurrentBar == 1 Then
    신규포지션 = 1;
Else
{
    이전포지션방향 = 포지션방향입력[1];
    이전진입가격 = 진입가격입력[1];

    If BDate <> BDate[1] Then
        신규포지션 = 1;

    If 포지션방향 <> IFF(이전포지션방향 > 0, 1,
                       IFF(이전포지션방향 < 0, -1, 0)) Then
        신규포지션 = 1;

    If 진입가격입력 <> 이전진입가격 Then
        신규포지션 = 1;
};

If 일분봉여부 == 1 && 포지션방향 <> 0 && 진입가격입력 > 0 Then
{
    청산대상방향 = 포지션방향;

    평가손익틱 = (현재가격입력 - 진입가격입력)
                   / 가격단위 * 포지션방향;

    If 신규포지션 == 1 Then
    {
        If 포지션방향 == 1 Then
            최고유리가격 = Max(진입가격입력, 현재고가입력);
        Else
            최고유리가격 = Min(진입가격입력, 현재저가입력);
    }
    Else
    {
        If 포지션방향 == 1 Then
            최고유리가격 = Max(최고유리가격[1], 현재고가입력);
        Else
            최고유리가격 = Min(최고유리가격[1], 현재저가입력);
    };

    최고수익틱 = (최고유리가격 - 진입가격입력)
                   / 가격단위 * 포지션방향;

    If 최고수익틱 < 0 Then
        최고수익틱 = 0;

    수익후퇴틱 = 최고수익틱 - 평가손익틱;
    최고수익틱출력 = 최고수익틱;

    /* 1순위: 고정 손절 */
    If 고정손절틱 > 0 && 평가손익틱 <= -고정손절틱 Then
    {
        청산요청 = 1;
        청산강도 = 100;
        청산사유 = 4;
    };

    /* 2순위: 미래곡선이 포지션 반대 방향으로 확정 */
    If 청산요청 == 0 &&
       미래방향입력 * 포지션방향 < 0 Then
    {
        청산요청 = 1;
        청산강도 = 100;
        청산사유 = 1;
    };

    /* 3순위: 회귀선을 포지션 반대편으로 완전 이탈 */
    If 청산요청 == 0 &&
       회귀이탈입력 * 포지션방향 < 0 Then
    {
        청산요청 = 1;
        청산강도 = 100;
        청산사유 = 2;
    };

    /* 4순위: 충분한 수익 후 정해진 폭만큼 후퇴 */
    If 청산요청 == 0 &&
       수익보호시작틱 > 0 &&
       수익보호폭틱 > 0 &&
       최고수익틱 >= 수익보호시작틱 &&
       수익후퇴틱 >= 수익보호폭틱 Then
    {
        청산요청 = 1;
        청산강도 = 100;
        청산사유 = 5;
    };

    /* 5순위: 호가, ADX 약화, 반대 매물대를 복합 확인 */
    호가역행 = 0;
    ADX약화 = 0;
    보조역행개수 = 0;

    If 호가유효입력 == 1 && 호가상태입력 * 포지션방향 < 0 Then
        호가역행 = 1;

    If ADX입력 < ADX이전입력 Then
        ADX약화 = 1;

    보조역행개수 = 호가역행
                   + ADX약화
                   + IFF(반대매물대접근입력 == 1, 1, 0);

    If 청산요청 == 0 && 보조역행개수 >= 2 Then
    {
        청산강도 = 50;
        청산사유 = 3;

        If 보조역행개수 >= 3 Then
        {
            청산요청 = 1;
            청산강도 = 100;
        };
    };
}
Else
{
    최고유리가격 = 0;
    최고수익틱 = 0;
    최고수익틱출력 = 0;
};

Value1 = 청산요청;
WSF_1m_ExitDecisionV1 = Value1;
