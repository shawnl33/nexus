Input : 변동기간입력(Numeric),
        중간갭기준입력(Numeric),
        큰갭기준입력(Numeric),
        갭비율(NumericRef),
        갭등급(NumericRef),
        일봉적용비중(NumericRef),
        장시작경과분(NumericRef),
        갭방향(NumericRef),
        갭판별유효(NumericRef);

//======================================================================
// WSF_GapRegimeV1
// - 1분봉 전용 갭 환경 판별 함수
// - 5,000봉 조회 환경을 고려하여 최근 5~10개 완성 세션만 사용
// - 갭비율 = Abs(금일 시가-직전 완성 세션 종가) / 평균 TR
//
// 갭등급       : 0 일반, 1 중간 갭, 2 큰 갭
// 일봉적용비중 : 일반 1.0, 중간 0.5, 큰 갭 0.0
// 갭방향       : 상승 갭 1, 하락 갭 -1, 보합 0
//======================================================================

Var : n(10), j(0),
      현재장고가(0), 현재장저가(0), 현재장종가(0), 현재장시가(0),
      직전완성종가(0), 직전직전종가(0), 완성TR(0), 평균TR(0),
      집계초기화(0), 완전장시작확인(0),
      최근장시작봉(-1), 완성일수(0), TR개수(0),
      장시작시각(0), 현재분(0), 시작분(0);

Array : 완성TR배열[20](0);

n = Min(10,Max(5,변동기간입력));

갭비율 = 0;
갭등급 = 0;
일봉적용비중 = 0;
장시작경과분 = 0;
갭방향 = 0;
갭판별유효 = 0;

If DataCompress == 2 && BarInterval == 1 Then
{
    If 집계초기화 == 0 Then
    {
        현재장시가 = O;
        현재장고가 = H;
        현재장저가 = L;
        현재장종가 = C;
        장시작시각 = stime;
        집계초기화 = 1;

        If DayIndex == 0 Then
        {
            완전장시작확인 = 1;
            최근장시작봉 = Index;
        }
    }
    Else
    {
        If DayIndex == 0 && Index != 최근장시작봉 Then
        {
            // 첫 로딩 세션이 중간부터 시작했다면 완성 세션으로 저장하지 않음
            If 완전장시작확인 == 1 Then
            {
                직전직전종가 = 직전완성종가;
                직전완성종가 = 현재장종가;

                If 완성일수 == 0 Then
                    완성TR = 현재장고가-현재장저가;
                Else
                    완성TR = Max(현재장고가-현재장저가,
                               Max(Abs(현재장고가-직전직전종가),
                                   Abs(현재장저가-직전직전종가)));

                For j = 18 DownTo 0
                {
                    완성TR배열[j+1] = 완성TR배열[j];
                }

                완성TR배열[0] = 완성TR;
                TR개수 = Min(20,TR개수+1);
                완성일수 = 완성일수+1;
            }

            현재장시가 = O;
            현재장고가 = H;
            현재장저가 = L;
            현재장종가 = C;
            장시작시각 = stime;
            완전장시작확인 = 1;
            최근장시작봉 = Index;
        }
        Else
        {
            현재장고가 = Max(현재장고가,H);
            현재장저가 = Min(현재장저가,L);
            현재장종가 = C;
        }
    }

    // HHMM 형식의 시간을 분으로 환산해 자정 경과도 처리
    현재분 = TimeToMinutes(stime);
    시작분 = TimeToMinutes(장시작시각);
    장시작경과분 = 현재분-시작분;
    If 장시작경과분 < 0 Then 장시작경과분 = 장시작경과분+1440;

    If TR개수 >= n && 직전완성종가 > 0 Then
    {
        평균TR = 0;

        For j = 0 To n-1
        {
            평균TR = 평균TR+완성TR배열[j];
        }

        평균TR = 평균TR/n;

        If 평균TR > 0 Then
        {
            갭비율 = Abs(현재장시가-직전완성종가)/평균TR;

            If 현재장시가 > 직전완성종가 Then 갭방향 = 1;
            If 현재장시가 < 직전완성종가 Then 갭방향 = -1;

            갭등급 = 0;
            일봉적용비중 = 1.0;

            If 갭비율 >= 중간갭기준입력 Then
            {
                갭등급 = 1;
                일봉적용비중 = 0.5;
            }

            If 갭비율 >= 큰갭기준입력 Then
            {
                갭등급 = 2;
                일봉적용비중 = 0.0;
            }

            갭판별유효 = 1;
        }
    }
}

Value1 = 1;
WSF_GapRegimeV1 = Value1;
