//======================================================================
// WSF_MarketWidthDir5V1
// #우드스탁_마켓폭방향5단계오실레이터V2의 계산 로직을 재사용 가능한
// 함수로 뽑아낸 것. 다른 WSF_ 함수들과 같은 관례로, 계산에 필요한 값만
// Input(Numeric)으로 받고 결과는 NumericRef 출력 하나(단계)로만 돌려줌
// ? 색상/Plot 등 표시 관련 로직은 호출부(지표) 쪽에서 처리.
//
// 계산 내용(V2와 동일):
//   1) 거래량가중 VWAP 중심가격 + 표준편차 밴드(밴드배수) ? 1분봉은
//      세션 리셋, 그 외 주기는 WSF_DailyMarketProfileV1 사용.
//   2) 위치강도 = 종가가 밴드 상단까지 거리 대비 얼마나 벗어났는지(%)
//   3) 밴드폭비율 = 현재 밴드폭 / 최근 폭비교기간 평균 밴드폭 (밴드
//      확장/수축 정도)
//   4) 결합점수 = 위치강도 * (1 + 폭가중치*(밴드폭비율-100)/100)
//   5) WSF_RelVolV1로 구한 RVOL상태로 결합점수를 가중/감쇠
//      (2→×1.2, 1→그대로, 0→×0.5)
//   6) 중심기울기+종가위치로 방향을 정하고, 레벨1~4기준으로 |결합점수|를
//      나눠 ±1~±5 단계를 매김(방향 불일치 시 0)
//======================================================================
// [업데이트] 결합점수에 2봉 EMA 스무딩 추가 ? 밴드가 좁아 봉 하나 만에
// 위치강도/결합점수가 확 튀면서 특정 레벨 구간(예: 4단계, 70~90점)을
// 그냥 건너뛰는 현상을 완화하기 위함. V1/V2는 원본 로직 그대로 유지
// (비교용 베이스라인), 이 함수(및 이를 호출하는 V3)에만 적용함.
//======================================================================
Input : 계산기간(Numeric),
        밴드배수(Numeric),
        폭비교기간(Numeric),
        폭가중치(Numeric),
        레벨1기준(Numeric),
        레벨2기준(Numeric),
        레벨3기준(Numeric),
        레벨4기준(Numeric),
        RVOL계산기간(Numeric),
        RVOL관심기준(Numeric),
        RVOL강세기준(Numeric),
        단계(NumericRef);

Var : 중심가격(0), 상단가격1(0), 하단가격1(0), 계산유효(0),
      세션봉수(0), 계산봉수(0), j(0),
      기준가격(0), 누적거래량(0), 누적가격거래량(0),
      가중분산합(0), 가중표준편차(0),
      중심기울기(0), 거리기준(0), 위치강도(0),
      밴드폭(0), 밴드폭평균(0), 밴드폭비율(0),
      결합점수(0), 결합점수원본(0), 결합점수평활(0), 계산단계(0),
      RVOL점수(0), RVOL기울기(0), RVOL상태(0), RVOL유효(0);

계산유효 = 0;

If DataCompress == 2 Then
{
    If CurrentBar == 1 Then
        세션봉수 = 1;
    Else If DayIndex == 0 Then
        세션봉수 = 1;
    Else
        세션봉수 = 세션봉수[1] + 1;

    계산봉수 = Min(계산기간,세션봉수);
    누적거래량 = 0;
    누적가격거래량 = 0;

    For j = 0 To 계산봉수-1
    {
        기준가격 = (H[j]+L[j]+C[j])/3;
        If V[j] > 0 Then
        {
            누적거래량 = 누적거래량 + V[j];
            누적가격거래량 = 누적가격거래량 + 기준가격*V[j];
        };
    };

    If 누적거래량 > 0 && 계산봉수 >= 2 Then
    {
        중심가격 = 누적가격거래량/누적거래량;
        가중분산합 = 0;
        For j = 0 To 계산봉수-1
        {
            기준가격 = (H[j]+L[j]+C[j])/3;
            If V[j] > 0 Then
                가중분산합 = 가중분산합 + (기준가격-중심가격)^2*V[j];
        };
        가중표준편차 = SquareRoot(가중분산합/누적거래량);
        상단가격1 = 중심가격 + 가중표준편차*밴드배수;
        하단가격1 = 중심가격 - 가중표준편차*밴드배수;
        계산유효 = 1;
    };
}
Else
{
    WSF_DailyMarketProfileV1(
        계산기간,
        밴드배수,
        중심가격,
        상단가격1,
        하단가격1,
        계산유효
    );
};

중심기울기 = 중심가격 - 중심가격[1];
거리기준 = Max(PriceScale,Abs(상단가격1-중심가격));
밴드폭 = Max(0,상단가격1-하단가격1);
밴드폭평균 = Average(밴드폭,Max(1,폭비교기간));

If 밴드폭평균 > 0 Then
    밴드폭비율 = 밴드폭/밴드폭평균*100;
Else
    밴드폭비율 = 100;

WSF_RelVolV1(
    RVOL관심기준,RVOL강세기준,RVOL계산기간,
    RVOL점수,RVOL기울기,RVOL상태,RVOL유효
);

위치강도 = 0;
결합점수 = 0;
결합점수원본 = 0;
계산단계 = 0;

If 계산유효 == 1 && 거리기준 > 0 Then
{
    위치강도 = Max(-100,Min(100,((C-중심가격)/거리기준)*100));
    결합점수원본 = Max(-100,Min(100, 위치강도*(1+폭가중치*(밴드폭비율-100)/100)));

    If 결합점수원본 == 0 || RVOL유효 == 0 Then
        결합점수 = 결합점수원본;
    Else If RVOL상태 == 2 Then
        결합점수 = 결합점수원본 * 1.2;
    Else If RVOL상태 == 1 Then
        결합점수 = 결합점수원본;
    Else
        결합점수 = 결합점수원본 * 0.5;

    결합점수 = Max(-100,Min(100,결합점수));

    // 4단계처럼 폭이 좁은 구간을 봉당 큰 점프로 건너뛰는 현상을 완화하기
    // 위해 2봉 EMA 스무딩 적용(가중치 2/3). 세션/계산 시작 첫 유효봉은
    // 스무딩 없이 원값을 그대로 사용.
    If 계산유효[1] == 1 Then
        결합점수평활 = 결합점수*0.6667 + 결합점수평활[1]*0.3333;
    Else
        결합점수평활 = 결합점수;

    If 중심기울기 > 0 && C > 중심가격 Then
    {
        If Abs(결합점수평활) >= 레벨4기준 Then 계산단계 = 5;
        Else If Abs(결합점수평활) >= 레벨3기준 Then 계산단계 = 4;
        Else If Abs(결합점수평활) >= 레벨2기준 Then 계산단계 = 3;
        Else If Abs(결합점수평활) >= 레벨1기준 Then 계산단계 = 2;
        Else 계산단계 = 1;
    }
    Else If 중심기울기 < 0 && C < 중심가격 Then
    {
        If Abs(결합점수평활) >= 레벨4기준 Then 계산단계 = -5;
        Else If Abs(결합점수평활) >= 레벨3기준 Then 계산단계 = -4;
        Else If Abs(결합점수평활) >= 레벨2기준 Then 계산단계 = -3;
        Else If Abs(결합점수평활) >= 레벨1기준 Then 계산단계 = -2;
        Else 계산단계 = -1;
    };
};

단계 = 계산단계;

Value1 = 1;
WSF_MarketWidthDir5V1 = Value1;
