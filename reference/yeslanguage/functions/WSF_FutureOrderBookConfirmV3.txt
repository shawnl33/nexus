Input : 미래방향입력(Numeric),
        호가방향상태입력(Numeric),
        호가유효입력(Numeric),
        확인봉수입력(Numeric),
        통합방향상태(NumericRef),
        호가확인여부(NumericRef),
        확인누적봉수(NumericRef),
        호가필터유효(NumericRef);

Var : 일분봉여부(0),
      미래방향값(0),
      확인봉수값(0),
      매수일치Cnt(0),
      매도일치Cnt(0);

/*
   미래곡선이 방향의 기준이다.
   호가는 1분봉에서 같은 방향을 확인하는 필터로만 사용한다.

   출력값
    100 : 미래곡선 상승 + 호가 상승 확인 완료
     50 : 미래곡선 상승, 호가 미확인 또는 충돌
    -50 : 미래곡선 하락, 호가 미확인 또는 충돌
   -100 : 미래곡선 하락 + 호가 하락 확인 완료
      0 : 미래곡선 중립
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

미래방향값 =
    IFF(미래방향입력 > 0, 1,
    IFF(미래방향입력 < 0, -1, 0));

확인봉수값 = Round(Abs(확인봉수입력), 0);
If 확인봉수값 < 1 Then
    확인봉수값 = 1;

/* 새 거래일에는 호가 연속 확인 횟수만 초기화한다. */
If BDate <> BDate[1] Then
{
    매수일치Cnt = 0;
    매도일치Cnt = 0;
};

통합방향상태 = 0;
호가확인여부 = 0;
확인누적봉수 = 0;
호가필터유효 = 0;

/* 일봉과 기타 주기에서는 호가를 배제하고 미래곡선만 사용한다. */
If 일분봉여부 == 0 Then
{
    매수일치Cnt = 0;
    매도일치Cnt = 0;

    If 미래방향값 > 0 Then
        통합방향상태 = 50;

    If 미래방향값 < 0 Then
        통합방향상태 = -50;
}
Else
{
    호가필터유효 = 호가유효입력;

    If 미래방향값 > 0 Then
    {
        매도일치Cnt = 0;

        If 호가유효입력 == 1 && 호가방향상태입력 > 0 Then
            매수일치Cnt = 매수일치Cnt[1] + 1;
        Else
            매수일치Cnt = 0;

        확인누적봉수 = 매수일치Cnt;

        If 매수일치Cnt >= 확인봉수값 Then
        {
            통합방향상태 = 100;
            호가확인여부 = 1;
        }
        Else
        {
            통합방향상태 = 50;
        };
    };

    If 미래방향값 < 0 Then
    {
        매수일치Cnt = 0;

        If 호가유효입력 == 1 && 호가방향상태입력 < 0 Then
            매도일치Cnt = 매도일치Cnt[1] + 1;
        Else
            매도일치Cnt = 0;

        확인누적봉수 = 매도일치Cnt;

        If 매도일치Cnt >= 확인봉수값 Then
        {
            통합방향상태 = -100;
            호가확인여부 = 1;
        }
        Else
        {
            통합방향상태 = -50;
        };
    };

    If 미래방향값 == 0 Then
    {
        매수일치Cnt = 0;
        매도일치Cnt = 0;
        통합방향상태 = 0;
    };
};

Value1 = 1;
WSF_FutureOrderBookConfirmV3 = Value1;
