Input : 핵심세션미래방향(Numeric),
        최소전환유지봉수(Numeric),
        시간비율(NumericRef),
        진행봉수(NumericRef);

//======================================================================
// WSF_SwingTimeRatioV1
// WSF_SwingFibMemoryV4에서 시간비율만 떼어낸 경량 버전.
// 가격(고가/저가) 및 피보나치 레벨 추적 없이, 방향전환/봉수(Count) 상태만
// 관리해서 시간비율만 계산한다. 상태 판정 로직(대기방향/전환확정 판정,
// 최소전환유지봉수 게이트)은 WSF_SwingFibMemoryV4와 완전히 동일하다.
//
// 출력 1) 시간비율 : WSF_SwingFibMemoryV4 원본과 동일한 소수값, 부호 없음
//                   (반올림 없음, 항상 0 이상)
// 출력 2) 진행봉수 : 지금 진행 중인 방향 구간이 몇 봉째 진행 중인지.
//                   상승 진행 중이면 +, 하락 진행 중이면 -.
//                   방향이 바뀌어 새 구간이 확정되는 순간(최소전환유지봉수
//                   충족 시점) 그 시점 봉수부터 다시 시작한다 ? 이전
//                   구간 값과 무관하게 초기화됨.
//======================================================================

Var : 상태_핵심상승Count(0), 상태_핵심하락Count(0),
      상태_핵심지난상승Count(0), 상태_핵심지난하락Count(0),
      상태_핵심방향구간방향(0), 상태_핵심양수전환(0), 상태_핵심음수전환(0),
      상태_시간비율(0), 상태_직전미래방향(0),
      상태_대기방향(0), 상태_대기Count(0);

If CurrentBar == 1 Then
{
    상태_핵심상승Count = 0;
    상태_핵심하락Count = 0;
    상태_핵심지난상승Count = 0;
    상태_핵심지난하락Count = 0;
    상태_핵심방향구간방향 = 0;
    상태_시간비율 = 0;
    상태_직전미래방향 = 0;
    상태_대기방향 = 0; 상태_대기Count = 0;
}
Else
{
    상태_핵심양수전환 = IFF(상태_직전미래방향 <= 0 && 핵심세션미래방향 > 0,1,0);
    상태_핵심음수전환 = IFF(상태_직전미래방향 >= 0 && 핵심세션미래방향 < 0,1,0);

    If 상태_대기방향 != 0 Then
    {
        If (상태_대기방향 == 1 && 핵심세션미래방향 > 0) || (상태_대기방향 == -1 && 핵심세션미래방향 < 0) Then
        {
            상태_대기Count = 상태_대기Count + 1;

            If 상태_대기Count >= 최소전환유지봉수 Then
            {
                If 상태_대기방향 == 1 Then
                {
                    If 상태_핵심방향구간방향 == -1 Then
                        상태_핵심지난하락Count = 상태_핵심하락Count;

                    상태_핵심상승Count = 상태_대기Count;
                    상태_핵심방향구간방향 = 1;
                }
                Else
                {
                    If 상태_핵심방향구간방향 == 1 Then
                        상태_핵심지난상승Count = 상태_핵심상승Count;

                    상태_핵심하락Count = 상태_대기Count;
                    상태_핵심방향구간방향 = -1;
                };

                상태_대기방향 = 0; 상태_대기Count = 0;
            };
        }
        Else If (상태_대기방향 == 1 && 핵심세션미래방향 < 0) || (상태_대기방향 == -1 && 핵심세션미래방향 > 0) Then
        {
            If 상태_핵심방향구간방향 == 1 Then
                상태_핵심상승Count = 상태_핵심상승Count + 상태_대기Count + 1;
            Else If 상태_핵심방향구간방향 == -1 Then
                상태_핵심하락Count = 상태_핵심하락Count + 상태_대기Count + 1;

            상태_대기방향 = 0; 상태_대기Count = 0;
        };
    }
    Else
    {
        If 상태_핵심양수전환 == 1 && 상태_핵심방향구간방향 != 1 Then
        {
            상태_대기방향 = 1;
            상태_대기Count = 1;
        }
        Else If 상태_핵심음수전환 == 1 && 상태_핵심방향구간방향 != -1 Then
        {
            상태_대기방향 = -1;
            상태_대기Count = 1;
        }
        Else If 상태_핵심방향구간방향 == 1 && 핵심세션미래방향 > 0 Then
        {
            상태_핵심상승Count = 상태_핵심상승Count + 1;
        }
        Else If 상태_핵심방향구간방향 == -1 && 핵심세션미래방향 < 0 Then
        {
            상태_핵심하락Count = 상태_핵심하락Count + 1;
        };
    };

    // 시간비율 ? 지금 되돌리고 있는 반대 구간 기준으로만 계산 (원본 로직과 동일, 부호 없음)
    상태_시간비율 = 0;
    If 상태_핵심방향구간방향 == -1 && 상태_핵심지난상승Count > 0 Then
        상태_시간비율 = 상태_핵심하락Count / 상태_핵심지난상승Count;
    If 상태_핵심방향구간방향 == 1 && 상태_핵심지난하락Count > 0 Then
        상태_시간비율 = 상태_핵심상승Count / 상태_핵심지난하락Count;

    상태_직전미래방향 = 핵심세션미래방향;
};

시간비율 = 상태_시간비율;
진행봉수 = IFF(상태_핵심방향구간방향 == 1, 상태_핵심상승Count,
           IFF(상태_핵심방향구간방향 == -1, -상태_핵심하락Count, 0));

Value1 = 1;
WSF_SwingTimeRatioV1_day = Value1;
