
Input : 관심기준입력(Numeric),
        강세기준입력(Numeric),
        부호반전입력(Numeric),
        호가방향점수(NumericRef),
        호가기울기3(NumericRef),
        호가방향상태(NumericRef),
        호가유효(NumericRef);

Var : 호가시작(False),
      선물여부(0),
      호가번호(0),
      매수잔량(0),
      매도잔량(0),
      전체잔량(0),
      잔량차이(0),
      누적잔량(0),
      평균잔량(0),
      잔량차이점수(0),
      잔량비율점수(0),
      핵심점수(0),
      방향평균3(0),
      방향평균5(0),
      계산호가방향(0),
      계산기울기3(0),
      계산방향상태(0),
      관심기준값(0),
      강세기준값(0);

관심기준값 = Abs(관심기준입력);
If 관심기준값 < 1 Then
    관심기준값 = 1;

강세기준값 = Abs(강세기준입력);
If 강세기준값 < 관심기준값 Then
    강세기준값 = 관심기준값;

매수잔량 = Bids;
매도잔량 = Asks;
전체잔량 = 매수잔량 + 매도잔량;

// 선물은 매수잔량 우세를 양수, 주식은 매도잔량 우세를 양수로 사용한다.
선물여부 = IFF(CodeCategory == 4,1,0);
If 선물여부 == 1 Then
    잔량차이 = 매수잔량 - 매도잔량;
Else
    잔량차이 = 매도잔량 - 매수잔량;

If BDate <> BDate[1] Then
{
    호가시작 = False;
    호가번호 = 0;
    누적잔량 = 0;
    평균잔량 = 0;
    잔량차이점수 = 0;
    잔량비율점수 = 0;
    핵심점수 = 0;
    방향평균3 = 0;
    방향평균5 = 0;
    계산호가방향 = 0;
    계산기울기3 = 0;
    계산방향상태 = 0;
};

If 전체잔량 > 0 Then
{
    If 호가시작 == False Then
    {
        호가시작 = True;
        호가번호 = 0;
        누적잔량 = 전체잔량;
        평균잔량 = 전체잔량;

        잔량차이점수 = 잔량차이 / 평균잔량 * 100;
        잔량비율점수 = 잔량차이 / 전체잔량 * 100;

        If 부호반전입력 == 1 Then
        {
            잔량차이점수 = -잔량차이점수;
            잔량비율점수 = -잔량비율점수;
        };

        If 잔량차이점수 > 100 Then
            잔량차이점수 = 100;
        If 잔량차이점수 < -100 Then
            잔량차이점수 = -100;

        If 잔량비율점수 > 100 Then
            잔량비율점수 = 100;
        If 잔량비율점수 < -100 Then
            잔량비율점수 = -100;

        핵심점수 = (잔량차이점수 * 60 + 잔량비율점수 * 40) / 100;
        방향평균3 = 핵심점수;
        방향평균5 = 핵심점수;
        계산호가방향 = 핵심점수;
        계산기울기3 = 0;
    }
    Else
    {
        호가번호 = 호가번호[1] + 1;
        누적잔량 = 누적잔량[1] + 전체잔량;
        평균잔량 = 누적잔량 / (호가번호 + 1);

        잔량차이점수 = 잔량차이 / 평균잔량 * 100;
        잔량비율점수 = 잔량차이 / 전체잔량 * 100;

        If 부호반전입력 == 1 Then
        {
            잔량차이점수 = -잔량차이점수;
            잔량비율점수 = -잔량비율점수;
        };

        If 잔량차이점수 > 100 Then
            잔량차이점수 = 100;
        If 잔량차이점수 < -100 Then
            잔량차이점수 = -100;

        If 잔량비율점수 > 100 Then
            잔량비율점수 = 100;
        If 잔량비율점수 < -100 Then
            잔량비율점수 = -100;

        핵심점수 = (잔량차이점수 * 60 + 잔량비율점수 * 40) / 100;

        If 호가번호 >= 2 Then
        {
            방향평균3 = (핵심점수 + 핵심점수[1] + 핵심점수[2]) / 3;
        }
        Else
        {
            방향평균3 = 핵심점수;
        }

        If 호가번호 >= 4 Then
        {
            방향평균5 = (핵심점수 + 핵심점수[1] + 핵심점수[2] + 핵심점수[3] + 핵심점수[4]) / 5;
        }
        Else
        {
            방향평균5 = 방향평균3;
        }

        계산호가방향 = (핵심점수 * 60 + 방향평균3 * 25 + 방향평균5 * 15) / 100;

        If 호가번호 >= 3 Then
        {
            계산기울기3 = 계산호가방향 - 계산호가방향[3];
        }
        Else
        {
            계산기울기3 = 0;
        }

    If 계산기울기3 > 100 Then
        계산기울기3 = 100;
    If 계산기울기3 < -100 Then
        계산기울기3 = -100;
    };

    계산방향상태 =
        IFF(계산호가방향 >= 강세기준값, 2,
        IFF(계산호가방향 >= 관심기준값, 1,
        IFF(계산호가방향 <= -강세기준값, -2,
        IFF(계산호가방향 <= -관심기준값, -1, 0))));

    호가방향점수 = 계산호가방향;
    호가기울기3 = 계산기울기3;
    호가방향상태 = 계산방향상태;
    호가유효 = 1;
}
Else
{
    호가방향점수 = 0;
    호가기울기3 = 0;
    호가방향상태 = 0;
    호가유효 = 0;
};

Value1 = 1;
WSF_OrderBookDirectionV2 = Value1;
