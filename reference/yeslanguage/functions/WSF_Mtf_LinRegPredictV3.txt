Input : 현재회귀선입력(Numeric),
        회귀기울기입력(Numeric),
        회귀유효입력(Numeric),
        회귀신뢰도입력(Numeric),
        예측봉수1입력(Numeric),
        예측봉수2입력(Numeric),
        예측봉수3입력(Numeric),
        예측가격1(NumericRef),
        예측가격2(NumericRef),
        예측가격3(NumericRef),
        예측방향(NumericRef),
        보정기울기(NumericRef),
        예측변동성(NumericRef);

//======================================================================
// WSF_Mtf_LinRegPredictV3
// - 모든 차트 주기에서 동일하게 사용하는 선형회귀 미래예측
// - 예측 단위는 시간이 아니라 해당 차트의 봉 수
// - 예: 1분봉 5봉=5분, 5분봉 5봉=25분, 일봉 5봉=5일
// - R²로 기울기를 축소하고 ATR(14)로 과도한 외삽을 제한
//======================================================================

Var : 예측봉수1(0), 예측봉수2(0), 예측봉수3(0),
      신뢰계수(0), ATR값(0), 최대기울기(0);

예측봉수1 = Max(0, 예측봉수1입력);
예측봉수2 = Max(0, 예측봉수2입력);
예측봉수3 = Max(0, 예측봉수3입력);

// 회귀 준비기간에는 현재 회귀값만 반환
예측가격1 = 현재회귀선입력;
예측가격2 = 현재회귀선입력;
예측가격3 = 현재회귀선입력;
예측방향 = 0;
보정기울기 = 0;
예측변동성 = ATR(14);

If 회귀유효입력 == 1 Then
{
    // R² 0.20 이하는 방향성 없음, 0.70 이상은 원기울기 100% 사용
    신뢰계수 = (회귀신뢰도입력 - 0.20) / 0.50;
    신뢰계수 = Max(0, Min(1, 신뢰계수));

    ATR값 = ATR(14);
    최대기울기 = ATR값 * 0.50;

    보정기울기 = 회귀기울기입력 * 신뢰계수;
    보정기울기 = Max(-최대기울기,
                    Min(최대기울기, 보정기울기));
    예측변동성 = ATR값;

    예측가격1 = 현재회귀선입력
              + 보정기울기 * 예측봉수1;

    예측가격2 = 현재회귀선입력
              + 보정기울기 * 예측봉수2;

    예측가격3 = 현재회귀선입력
              + 보정기울기 * 예측봉수3;

    예측방향 = IFF(보정기울기 > 0, 1,
                  IFF(보정기울기 < 0, -1, 0));
}

Value1 = 1;
WSF_Mtf_LinRegPredictV3 = Value1;
