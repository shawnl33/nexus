// WSF_FXFlatCountV2_CO  (_CO = Claude Opus 작성)
// V1 + 최소구간봉수: 끝난 구간이 최소구간봉수 이상일 때만 직전 구간 길이(직전상승/하락 위치·기울기)로 기록.
//      짧은 잡음 구간은 직전 값을 덮어쓰지 않음. 최소구간봉수 = 1 이면 V1과 같음. V1은 보존.
// (아래는 V1 설명)
// WSF_FXFlatCountV1_CO
// 곡선회귀선_평탄 카운트 (백업 매매신호샘플.txt의 평탄_Cnt 로직 정리판)
//   위치 카운트: 최근 2봉 종가가 모두 평탄선 위 +1 / 모두 아래 -1 상태가 이어진 봉 수
//               (위 +N, 아래 -N. 섞인 봉은 카운트 유지, 구간 고저는 계속 갱신)
//   기울기 카운트: 평탄선이 마지막으로 오른(+1)/내린(-1) 방향이 유지된 봉 수 (위 +N, 아래 -N)
//               평탄선은 틱 단위 계단이라, 값이 그대로인 봉도 직전 방향으로 이어 센다.
// 수정 사항 (원본 대비): 전환 판정을 상태 변화만으로 (Abs(Cnt)>1 조건 제거, 0을 지나 반대로 세지 않음),
//   첫 구간 고저를 H/L로 시작, 같은 봉 재평가는 직전 봉 값에서 시작, 세션(07:00/15:30) 초기화 전체 적용.
// 출력: 위치Cnt, 직전상승위치Cnt, 직전하락위치Cnt(양수), 위치구간진폭(틱),
//       기울기Cnt, 직전상승기울기Cnt, 직전하락기울기Cnt(양수)
// 반환: 평탄값이 유효하면 1
Input : 세션초기화(Numeric), 평탄값(Numeric), 최소구간봉수(Numeric),
        위치Cnt출력(NumericRef), 직전상승위치Cnt출력(NumericRef), 직전하락위치Cnt출력(NumericRef),
        위치구간진폭출력(NumericRef),
        기울기Cnt출력(NumericRef), 직전상승기울기Cnt출력(NumericRef), 직전하락기울기Cnt출력(NumericRef);
Var : 위치Cnt(0), 직전상승위치Cnt(0), 직전하락위치Cnt(0), 구간고(0), 구간저(0), 위치상태(0),
      기울기Cnt(0), 기울기방향(0), 새기울기(0), 직전상승기울기Cnt(0), 직전하락기울기Cnt(0),
      유효(0), 평탄(0);

// 같은 봉 재평가는 직전 봉 상태에서 시작
If CurrentBar > 1 Then
{
    위치Cnt = 위치Cnt[1]; 직전상승위치Cnt = 직전상승위치Cnt[1]; 직전하락위치Cnt = 직전하락위치Cnt[1];
    구간고 = 구간고[1]; 구간저 = 구간저[1];
    기울기Cnt = 기울기Cnt[1]; 기울기방향 = 기울기방향[1];
    직전상승기울기Cnt = 직전상승기울기Cnt[1]; 직전하락기울기Cnt = 직전하락기울기Cnt[1];
};
If 세션초기화 == 1 Then
{
    위치Cnt = 0; 직전상승위치Cnt = 0; 직전하락위치Cnt = 0; 구간고 = 0; 구간저 = 0;
    기울기Cnt = 0; 기울기방향 = 0; 직전상승기울기Cnt = 0; 직전하락기울기Cnt = 0;
};

// 입력값을 변수에 담아 직전 봉 값([1])을 안전하게 참조
평탄 = 평탄값;
유효 = 0;
If 평탄 > 0 Then 유효 = 1;

If 유효 == 1 && 세션초기화 == 0 && CurrentBar > 1 && 평탄[1] > 0 Then
{
    // 위치 카운트
    위치상태 = 0;
    If C > 평탄 && C[1] > 평탄[1] Then 위치상태 = 1;
    Else If C < 평탄 && C[1] < 평탄[1] Then 위치상태 = -1;

    If 위치상태 == 1 Then
    {
        If 위치Cnt <= 0 Then
        {
            If 위치Cnt < 0 && -위치Cnt >= 최소구간봉수 Then 직전하락위치Cnt = -위치Cnt;
            위치Cnt = 1; 구간고 = H; 구간저 = L;
        }
        Else
        {
            위치Cnt = 위치Cnt+1; 구간고 = Max(구간고,H); 구간저 = Min(구간저,L);
        };
    }
    Else If 위치상태 == -1 Then
    {
        If 위치Cnt >= 0 Then
        {
            If 위치Cnt > 0 && 위치Cnt >= 최소구간봉수 Then 직전상승위치Cnt = 위치Cnt;
            위치Cnt = -1; 구간고 = H; 구간저 = L;
        }
        Else
        {
            위치Cnt = 위치Cnt-1; 구간고 = Max(구간고,H); 구간저 = Min(구간저,L);
        };
    }
    Else If 위치Cnt != 0 Then
    {
        구간고 = Max(구간고,H); 구간저 = Min(구간저,L);
    };

    // 기울기 카운트
    새기울기 = 0;
    If 평탄 > 평탄[1] Then 새기울기 = 1;
    Else If 평탄 < 평탄[1] Then 새기울기 = -1;
    If 새기울기 != 0 && 새기울기 != 기울기방향 Then
    {
        If 기울기방향 == 1 && 기울기Cnt >= 최소구간봉수 Then 직전상승기울기Cnt = 기울기Cnt;
        If 기울기방향 == -1 && -기울기Cnt >= 최소구간봉수 Then 직전하락기울기Cnt = -기울기Cnt;
        기울기방향 = 새기울기;
        기울기Cnt = 새기울기;
    }
    Else If 기울기방향 != 0 Then
        기울기Cnt = 기울기Cnt + 기울기방향;
};

위치Cnt출력 = 위치Cnt;
직전상승위치Cnt출력 = 직전상승위치Cnt;
직전하락위치Cnt출력 = 직전하락위치Cnt;
위치구간진폭출력 = 0;
If 위치Cnt != 0 && PriceScale > 0 Then 위치구간진폭출력 = (구간고-구간저)/PriceScale;
기울기Cnt출력 = 기울기Cnt;
직전상승기울기Cnt출력 = 직전상승기울기Cnt;
직전하락기울기Cnt출력 = 직전하락기울기Cnt;
Value1 = 유효;
WSF_FXFlatCountV2_CO = Value1;
