Input : 미래방향입력(Numeric),
        매물대위치입력(Numeric),
        ADX입력(Numeric),
        상승DI입력(Numeric),
        하락DI입력(Numeric),
        호가상태입력(Numeric),
        호가유효입력(Numeric),
        ADX추세기준입력(Numeric),
        ADX강세기준입력(Numeric),
        통합상태(NumericRef),
        매물대상태(NumericRef),
        ADX상태(NumericRef),
        DI방향(NumericRef),
        호가필터상태(NumericRef);

Var : 미래방향(0),
      매물대위치(0),
      ADX값(0),
      ADX추세기준값(0),
      ADX강세기준값(0),
      호가상태(0),
      호가유효(0);

/*
   1분봉 통합판정 원칙

   1. 미래곡선이 매매 방향을 결정한다.
   2. 세션 매물대와 세션 ADX/DI는 방향 확인에 사용한다.
   3. 호가 방향은 마지막 실시간 진입 필터로만 사용한다.
   4. 과거 봉에 호가 자료가 없으면 신호를 지우지 않고 관찰 상태를 유지한다.

   출력 상태
      100 : 매수 확인
       50 : 매수 관찰
        0 : 중립
      -50 : 매도 관찰
     -100 : 매도 확인
*/

미래방향 = IFF(미래방향입력 > 0, 1,
             IFF(미래방향입력 < 0, -1, 0));

매물대위치 = IFF(매물대위치입력 > 0, 1,
               IFF(매물대위치입력 < 0, -1, 0));

호가상태 = IFF(호가상태입력 >= 2, 2,
             IFF(호가상태입력 > 0, 1,
             IFF(호가상태입력 <= -2, -2,
             IFF(호가상태입력 < 0, -1, 0))));

호가유효 = IFF(호가유효입력 == 1, 1, 0);

ADX값 = Abs(ADX입력);
ADX추세기준값 = Abs(ADX추세기준입력);
ADX강세기준값 = Abs(ADX강세기준입력);

If ADX추세기준값 < 1 Then
    ADX추세기준값 = 1;

If ADX강세기준값 < ADX추세기준값 Then
    ADX강세기준값 = ADX추세기준값;

매물대상태 = 매물대위치;

DI방향 = IFF(상승DI입력 > 하락DI입력, 1,
          IFF(상승DI입력 < 하락DI입력, -1, 0));

ADX상태 = 0;

If ADX값 >= ADX강세기준값 Then
    ADX상태 = 2;
Else
{
    If ADX값 >= ADX추세기준값 Then
        ADX상태 = 1;
};

호가필터상태 = 0;

If 호가유효 == 1 Then
{
    If 호가상태 > 0 Then
        호가필터상태 = 1;

    If 호가상태 < 0 Then
        호가필터상태 = -1;
};

통합상태 = 0;

/* 미래곡선 상승: 호가가 없거나 불일치하면 +50 관찰을 유지한다. */
If 미래방향 == 1 Then
{
    통합상태 = 50;

    If 매물대상태 >= 0 &&
       DI방향 == 1 &&
       ADX상태 >= 1 &&
       호가유효 == 1 &&
       호가필터상태 == 1 Then
    {
        통합상태 = 100;
    };
};

/* 미래곡선 하락: 호가가 없거나 불일치하면 -50 관찰을 유지한다. */
If 미래방향 == -1 Then
{
    통합상태 = -50;

    If 매물대상태 <= 0 &&
       DI방향 == -1 &&
       ADX상태 >= 1 &&
       호가유효 == 1 &&
       호가필터상태 == -1 Then
    {
        통합상태 = -100;
    };
};

Value1 = 통합상태;
WSF_1m_FutureDecisionV1 = Value1;
