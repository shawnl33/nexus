Input : 후보방향입력(Numeric),
        진입가격입력(Numeric),
        청산입력(Numeric),
        진입활성출력(NumericRef),
        저장방향출력(NumericRef),
        저장가격출력(NumericRef),
        저장일자출력(NumericRef),
        저장시간출력(NumericRef),
        저장봉번호출력(NumericRef);

Var : 일분봉여부(0),
      내부진입활성(0),
      내부저장방향(0),
      내부저장가격(0),
      내부저장일자(0),
      내부저장시간(0),
      내부저장봉번호(0),
      적용진입가격(0);

/*
   WSF_1m_EntrySnapshotV1

   목적
   1. 1분봉 매수/매도 후보가 처음 발생한 순간을 저장한다.
   2. 진입 방향, 가격, 일자, 시간, 봉 번호를 다음 봉에도 유지한다.
   3. 새 거래일에는 이전 진입 상태를 초기화한다.
   4. 청산입력이 들어오면 진입활성만 해제한다.

   입력 규칙
   후보방향입력 : 매수 후보 1, 매도 후보 -1, 후보 없음 0
   진입가격입력 : 보통 C를 전달한다. 0 이하면 현재 종가 C를 사용한다.
   청산입력     : 청산 조건 발생 봉에서 1, 평상시 0
*/

일분봉여부 = IFF(DataCompress == 2 && BarInterval == 1, 1, 0);

/* 직전 봉에 저장된 상태를 이어받는다. */
If CurrentBar == 1 Then
{
    내부진입활성 = 0;
    내부저장방향 = 0;
    내부저장가격 = 0;
    내부저장일자 = 0;
    내부저장시간 = 0;
    내부저장봉번호 = 0;
}
Else
{
    내부진입활성 = 내부진입활성[1];
    내부저장방향 = 내부저장방향[1];
    내부저장가격 = 내부저장가격[1];
    내부저장일자 = 내부저장일자[1];
    내부저장시간 = 내부저장시간[1];
    내부저장봉번호 = 내부저장봉번호[1];
};

/* 1분봉이 아니면 이 함수의 거래 상태를 사용하지 않는다. */
If 일분봉여부 == 0 Then
{
    내부진입활성 = 0;
    내부저장방향 = 0;
    내부저장가격 = 0;
    내부저장일자 = 0;
    내부저장시간 = 0;
    내부저장봉번호 = 0;
}
Else
{
    /* 새 거래일에는 당일 단타 상태를 초기화한다. */
    If CurrentBar > 1 && BDate <> BDate[1] Then
    {
        내부진입활성 = 0;
        내부저장방향 = 0;
        내부저장가격 = 0;
        내부저장일자 = 0;
        내부저장시간 = 0;
        내부저장봉번호 = 0;
    };

    /* 청산 후에는 같은 날에도 다음 신규 후보를 받을 수 있다. */
    If 청산입력 <> 0 Then
        내부진입활성 = 0;

    /* 미보유 상태에서 발생한 첫 후보만 진입 정보로 저장한다. */
    If 내부진입활성 == 0 && 후보방향입력 <> 0 Then
    {
        적용진입가격 = 진입가격입력;

        If 적용진입가격 <= 0 Then
            적용진입가격 = C;

        내부진입활성 = 1;
        내부저장방향 = IFF(후보방향입력 > 0, 1, -1);
        내부저장가격 = 적용진입가격;
        내부저장일자 = BDate;
        내부저장시간 = STime;
        내부저장봉번호 = CurrentBar;
    };
};

진입활성출력 = 내부진입활성;
저장방향출력 = 내부저장방향;
저장가격출력 = 내부저장가격;
저장일자출력 = 내부저장일자;
저장시간출력 = 내부저장시간;
저장봉번호출력 = 내부저장봉번호;

Value1 = 저장방향출력;
WSF_1m_EntrySnapshotV1 = Value1;
