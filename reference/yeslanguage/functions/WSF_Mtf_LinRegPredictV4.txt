
Input : 현재회귀선입력(Numeric),
        회귀기울기입력(Numeric),
        회귀유효입력(Numeric),
        회귀신뢰도입력(Numeric),
        예측봉수1입력(Numeric),
        예측봉수2입력(Numeric),
        예측봉수3입력(Numeric),
        예측가격1(NumericRef),
        예측가격2(NumericRef),
        예측가격3(NumericRef),
        예측방향1(NumericRef),
        예측방향2(NumericRef),
        예측방향3(NumericRef),
        보정기울기(NumericRef),
        기울기가속도(NumericRef),
        예측변동성(NumericRef);

//======================================================================
// WSF_Mtf_LinRegPredictV4
// - V3의 R²/ATR 제한을 유지
// - 최근 3봉의 기울기 둔화를 반영하여 전환부 과대 외삽을 완화
// - 같은 방향의 가속은 사용하지 않고, 둔화/반전 성분만 사용
//======================================================================

Var : 예측봉수1(0), 예측봉수2(0), 예측봉수3(0),
      신뢰계수(0), ATR값(0), 최대기울기(0), 최대가속도(0),
      원가속도(0), 이동값1(0), 이동값2(0), 이동값3(0),
      최대이동1(0), 최대이동2(0), 최대이동3(0);

예측봉수1 = Max(0,예측봉수1입력);
예측봉수2 = Max(0,예측봉수2입력);
예측봉수3 = Max(0,예측봉수3입력);

예측가격1 = 현재회귀선입력;
예측가격2 = 현재회귀선입력;
예측가격3 = 현재회귀선입력;
예측방향1 = 0;
예측방향2 = 0;
예측방향3 = 0;
보정기울기 = 0;
기울기가속도 = 0;
예측변동성 = ATR(14);

If 회귀유효입력 == 1 Then
{
    신뢰계수 = (회귀신뢰도입력-0.20)/0.50;
    신뢰계수 = Max(0,Min(1,신뢰계수));

    ATR값 = ATR(14);
    최대기울기 = ATR값*0.50;
    최대가속도 = ATR값*0.08;

    보정기울기 = 회귀기울기입력*신뢰계수;
    보정기울기 = Max(-최대기울기,Min(최대기울기,보정기울기));

    원가속도 = (회귀기울기입력-회귀기울기입력[3])/3;
    기울기가속도 = 원가속도*신뢰계수;
    기울기가속도 = Max(-최대가속도,Min(최대가속도,기울기가속도));

    // 추세를 더 과장하지 않고 둔화되는 성분만 반영
    If 보정기울기*기울기가속도 >= 0 Then 기울기가속도 = 0;

    // 30분 이하에서는 전일 기울기를 가속도 계산에 섞지 않음
    If DataCompress == 2 && BarInterval <= 30 && sDate != sDate[3] Then
        기울기가속도 = 0;

    이동값1 = 보정기울기*예측봉수1
              + 0.5*기울기가속도*예측봉수1*예측봉수1;
    이동값2 = 보정기울기*예측봉수2
              + 0.5*기울기가속도*예측봉수2*예측봉수2;
    이동값3 = 보정기울기*예측봉수3
              + 0.5*기울기가속도*예측봉수3*예측봉수3;

    최대이동1 = ATR값*1.50*SquareRoot(Max(1,예측봉수1));
    최대이동2 = ATR값*1.50*SquareRoot(Max(1,예측봉수2));
    최대이동3 = ATR값*1.50*SquareRoot(Max(1,예측봉수3));

    이동값1 = Max(-최대이동1,Min(최대이동1,이동값1));
    이동값2 = Max(-최대이동2,Min(최대이동2,이동값2));
    이동값3 = Max(-최대이동3,Min(최대이동3,이동값3));

    예측가격1 = 현재회귀선입력+이동값1;
    예측가격2 = 현재회귀선입력+이동값2;
    예측가격3 = 현재회귀선입력+이동값3;

    예측방향1 = IFF(이동값1 > 0,1,IFF(이동값1 < 0,-1,0));
    예측방향2 = IFF(이동값2 > 0,1,IFF(이동값2 < 0,-1,0));
    예측방향3 = IFF(이동값3 > 0,1,IFF(이동값3 < 0,-1,0));
    예측변동성 = ATR값;
}

Value1 = 1;
WSF_Mtf_LinRegPredictV4 = Value1;
