// WSF_FXSyntheticLinesV1: 완료된 1분봉으로 합성한 5/15/30분 기준선.
// 봉시각기준: 0=1분봉 시작시각, 1=1분봉 종료시각.
// 새 1분봉 첫 평가에서 직전 완료 1분봉만 처리. 현재 OHLCV 사용 없음.
// 구간 시작 07:00/15:30. 누락 분이 있는 합성봉은 제외. 이전 출력은 유지.
Input : 합성분(Numeric), 회귀기간(Numeric), 마켓기간(Numeric), 봉시각기준(Numeric),
        평탄회귀출력(NumericRef), 마켓중심출력(NumericRef),
        회귀유효출력(NumericRef), 마켓유효출력(NumericRef);
Var : 최근봉(-1), 저장세션키(0), 현재세션키(0), 이전세션키(0),
      이전날짜(0), 이전시각(0), 이전분(0), 경과분(0), 버킷(0), 저장버킷(-1),
      집계개수(0), 직전경과분(-2), 집계고가(0), 집계저가(0), 집계종가(0), 집계거래량(0),
      완료개수(0), j(0), 회귀계산개수(0), 마켓계산개수(0), sumX(0), sumY(0), sumXY(0), sumX2(0),
      분모(0), 기울기(0), 절편(0), 가격(0), 거래량합(0), 가격거래량합(0),
      저장회귀(0), 저장마켓(0), 저장회귀유효(0), 저장마켓유효(0),
      연도(0), 월(0), 일(0), 실행가능(0);
Array : 완료중간[100](0), 완료대표[100](0), 완료거래량[100](0);

평탄회귀출력 = 0; 마켓중심출력 = 0;
회귀유효출력 = 0; 마켓유효출력 = 0;
실행가능 = 0;
If DataCompress == 2 && BarInterval == 1 && PriceScale > 0 &&
   (합성분 == 5 || 합성분 == 15 || 합성분 == 30) &&
   (봉시각기준 == 0 || 봉시각기준 == 1) Then 실행가능 = 1;

If 실행가능 == 1 Then
{
    현재세션키 = WSF_FXSessionKeyV1(sDate,sTime);
    If Index != 최근봉 Then
    {
        최근봉 = Index;
        If 현재세션키 != 저장세션키 Then
        {
            저장세션키 = 현재세션키;
            저장버킷 = -1; 집계개수 = 0; 직전경과분 = -2;
            완료개수 = 0;
            저장회귀 = 0; 저장마켓 = 0;
            저장회귀유효 = 0; 저장마켓유효 = 0;
            For j = 0 To 99
            { 완료중간[j] = 0; 완료대표[j] = 0; 완료거래량[j] = 0; };
        };

        If CurrentBar > 1 Then
        {
            이전날짜 = sDate[1];
            이전시각 = sTime[1];
            이전분 = (이전시각-이전시각%10000)/10000*60
                     + (이전시각%10000-이전시각%100)/100;
            If 봉시각기준 == 1 Then
            {
                이전분 = 이전분-1;
                If 이전분 < 0 Then
                {
                    이전분 = 1439;
                    일 = 이전날짜%100;
                    월 = ((이전날짜-일)/100)%100;
                    연도 = (이전날짜-월*100-일)/10000;
                    일 = 일-1;
                    If 일 == 0 Then
                    {
                        월 = 월-1;
                        If 월 == 0 Then { 월 = 12; 연도 = 연도-1; };
                        일 = 31;
                        If 월 == 4 || 월 == 6 || 월 == 9 || 월 == 11 Then 일 = 30;
                        If 월 == 2 Then
                        {
                            일 = 28;
                            If 연도%4 == 0 && (연도%100 != 0 || 연도%400 == 0) Then 일 = 29;
                        };
                    };
                    이전날짜 = 연도*10000+월*100+일;
                };
            };
            이전시각 = (이전분-이전분%60)/60*10000+이전분%60*100;
            이전세션키 = WSF_FXSessionKeyV1(이전날짜,이전시각);
            If 이전세션키 == 현재세션키 Then
            {
                If 이전분 >= 420 && 이전분 < 930 Then 경과분 = 이전분-420;
                Else If 이전분 >= 930 Then 경과분 = 이전분-930;
                Else 경과분 = 이전분+510;
                버킷 = (경과분-경과분%합성분)/합성분;

                If 버킷 != 저장버킷 Then
                {
                    저장버킷 = 버킷; 집계개수 = 0; 직전경과분 = -2;
                };
                // 중간에서 읽기 시작한 버킷 또는 누락이 있는 버킷은 채우지 않는다.
                If 경과분%합성분 == 0 Then
                {
                    집계개수 = 1;
                    집계고가 = H[1]; 집계저가 = L[1]; 집계종가 = C[1];
                    집계거래량 = Max(0,V[1]);
                }
                Else If 집계개수 > 0 && 경과분 == 직전경과분+1 Then
                {
                    집계개수 = 집계개수+1;
                    집계고가 = Max(집계고가,H[1]);
                    집계저가 = Min(집계저가,L[1]);
                    집계종가 = C[1];
                    집계거래량 = 집계거래량+Max(0,V[1]);
                }
                Else 집계개수 = 0;
                직전경과분 = 경과분;

                If 집계개수 == 합성분 && 경과분%합성분 == 합성분-1 Then
                {
                    For j = 98 DownTo 0
                    {
                        완료중간[j+1] = 완료중간[j];
                        완료대표[j+1] = 완료대표[j];
                        완료거래량[j+1] = 완료거래량[j];
                    };
                    완료중간[0] = (집계고가+집계저가)/2;
                    완료대표[0] = (집계고가+집계저가+집계종가)/3;
                    완료거래량[0] = 집계거래량;
                    완료개수 = Min(100,완료개수+1);
                    집계개수 = 0;

                    저장회귀유효 = 0; 저장마켓유효 = 0;
                    저장회귀 = 0; 저장마켓 = 0;
                    회귀계산개수 = Min(완료개수,Min(100,Max(5,Round(회귀기간,0))));
                    If 회귀계산개수 >= 5 Then
                    {
                        sumX = 0; sumY = 0; sumXY = 0; sumX2 = 0;
                        For j = 0 To 회귀계산개수-1
                        {
                            가격 = 완료중간[회귀계산개수-1-j];
                            sumX = sumX+j; sumY = sumY+가격;
                            sumXY = sumXY+j*가격; sumX2 = sumX2+j*j;
                        };
                        분모 = 회귀계산개수*sumX2-sumX*sumX;
                        If 분모 > 0 Then
                        {
                            기울기 = (회귀계산개수*sumXY-sumX*sumY)/분모;
                            절편 = (sumY-기울기*sumX)/회귀계산개수;
                            저장회귀 = Round((기울기*(회귀계산개수-1)+절편)/PriceScale,0)*PriceScale;
                            저장회귀유효 = 1;
                        };
                    };

                    마켓계산개수 = Min(완료개수,Min(100,Max(2,Round(마켓기간,0))));
                    거래량합 = 0; 가격거래량합 = 0;
                    For j = 0 To 마켓계산개수-1
                    {
                        거래량합 = 거래량합+완료거래량[j];
                        가격거래량합 = 가격거래량합+완료대표[j]*완료거래량[j];
                    };
                    If 마켓계산개수 >= 2 && 거래량합 > 0 Then
                    {
                        저장마켓 = 가격거래량합/거래량합;
                        저장마켓유효 = 1;
                    };
                };
            };
        };
    };
    평탄회귀출력 = 저장회귀;
    마켓중심출력 = 저장마켓;
    회귀유효출력 = 저장회귀유효;
    마켓유효출력 = 저장마켓유효;
};
WSF_FXSyntheticLinesV1 = 실행가능;
