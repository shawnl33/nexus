Input : 분봉예측방향1입력(Numeric),
        분봉예측방향2입력(Numeric),
        분봉예측방향3입력(Numeric),
        분봉회귀유효입력(Numeric),
        분봉회귀신뢰도입력(Numeric),
        일봉추세방향입력(Numeric),
        일봉추세상태입력(Numeric),
        일봉추세강도입력(Numeric),
        일봉추세유효입력(Numeric),
        갭등급입력(Numeric),
        일봉적용비중입력(Numeric),
        장시작경과분입력(Numeric),
        큰갭재평가분입력(Numeric),
        분봉최소신뢰도입력(Numeric),
        최종최소강도입력(Numeric),
        분봉합의방향(NumericRef),
        분봉합의상태(NumericRef),
        적용일봉비중(NumericRef),
        일치방향(NumericRef),
        일치상태(NumericRef),
        일치강도(NumericRef),
        일치유효(NumericRef);

//======================================================================
// OSF_1m_DailyAlignV2
// - 매매 방향의 중심은 1분봉 선형회귀 예측
// - 일반장      : 기존처럼 일봉과 같은 방향만 허용
// - 중간 갭     : 일봉은 50% 보정값, 반대여도 1분 신호는 유지
// - 큰 갭 초기  : 일봉 방향을 무시하고 1분 신호만 사용
// - 큰 갭 재평가: 지정 시간이 지나면 일봉을 50% 보정값으로 복원
//======================================================================

Var : 상승개수(0), 하락개수(0), 분봉기본강도(0),
      보정강도(0), 같은방향(0), 반대방향(0);

분봉합의방향 = 0;
분봉합의상태 = 0;
적용일봉비중 = Min(1,Max(0,일봉적용비중입력));
일치방향 = 0;
일치상태 = 0;
일치강도 = 0;
일치유효 = 0;

상승개수 = 0;
하락개수 = 0;

If 분봉예측방향1입력 > 0 Then 상승개수 = 상승개수+1;
If 분봉예측방향2입력 > 0 Then 상승개수 = 상승개수+1;
If 분봉예측방향3입력 > 0 Then 상승개수 = 상승개수+1;

If 분봉예측방향1입력 < 0 Then 하락개수 = 하락개수+1;
If 분봉예측방향2입력 < 0 Then 하락개수 = 하락개수+1;
If 분봉예측방향3입력 < 0 Then 하락개수 = 하락개수+1;

If 상승개수 >= 2 Then
{
    분봉합의방향 = 1;
    분봉합의상태 = 1;
    If 상승개수 == 3 Then 분봉합의상태 = 2;
}

If 하락개수 >= 2 Then
{
    분봉합의방향 = -1;
    분봉합의상태 = -1;
    If 하락개수 == 3 Then 분봉합의상태 = -2;
}

// 큰 갭은 개장 초기 일봉 무시, 재평가 시간 이후에도 절반만 복원
If 갭등급입력 >= 2 Then
{
    적용일봉비중 = 0;

    If 장시작경과분입력 >= 큰갭재평가분입력 Then
        적용일봉비중 = 0.5;
}

If 분봉회귀유효입력 == 1 &&
   분봉회귀신뢰도입력 >= 분봉최소신뢰도입력 &&
   분봉합의방향 != 0 Then
{
    분봉기본강도 = 분봉회귀신뢰도입력*100;
    If Abs(분봉합의상태) == 1 Then 분봉기본강도 = 분봉기본강도*0.75;

    같은방향 = 0;
    반대방향 = 0;

    If 일봉추세유효입력 == 1 Then
    {
        If 분봉합의방향 == 일봉추세방향입력 Then 같은방향 = 1;
        If 분봉합의방향 != 일봉추세방향입력 Then 반대방향 = 1;
    }

    // 일반장에서는 일봉이 유효하고 같은 방향이어야 함
    If 적용일봉비중 >= 1 Then
    {
        If 같은방향 == 1 Then
        {
            보정강도 = 분봉기본강도+
                       Min(20,일봉추세강도입력*0.20);
            일치유효 = 1;
        }
    }
    Else
    {
        // 갭 장에서는 1분봉 신호를 유지하며 일봉은 가감점만 부여
        보정강도 = 분봉기본강도;

        If 같은방향 == 1 Then
            보정강도 = 보정강도+
                       Min(20,일봉추세강도입력*0.20)*적용일봉비중;

        If 반대방향 == 1 Then
            보정강도 = 보정강도-
                       Min(35,일봉추세강도입력*0.35)*적용일봉비중;

        일치유효 = 1;
    }

    보정강도 = Min(100,Max(0,보정강도));

    If 일치유효 == 1 && 보정강도 >= 최종최소강도입력 Then
    {
        일치방향 = 분봉합의방향;
        일치강도 = 보정강도;
        일치상태 = 일치방향;

        If Abs(분봉합의상태) == 2 && 일치강도 >= 60 Then
            일치상태 = 일치방향*2;
    }
    Else
    {
        일치방향 = 0;
        일치상태 = 0;
        일치강도 = 0;
        일치유효 = 0;
    }
}

Value1 = 1;
OSF_1m_DailyAlignV2 = Value1;
