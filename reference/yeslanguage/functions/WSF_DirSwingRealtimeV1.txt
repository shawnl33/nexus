Input : 핵심세션미래방향(Numeric),
        핵심실시간방향(NumericRef),
        핵심실시간상승최고가(NumericRef),
        핵심실시간상승최저가(NumericRef),
        핵심실시간상승Count(NumericRef),
        핵심실시간하락최고가(NumericRef),
        핵심실시간하락최저가(NumericRef),
        핵심실시간하락Count(NumericRef);

//======================================================================
// WSF_DirSwingRealtimeV1
// - 입력으로 받은 세션미래방향(핵심세션미래방향)이 방향을 "전환"하는
//   바로 그 순간(0선을 넘어가는 순간)에만 초기화하고, 그 뒤로는
//   같은 방향이 이어지는 동안 실시간으로 고가/저가/카운트를 계속
//   갱신해서 돌려주는 순수 실시간 구간 추적 함수.
// - WSF_SwingFibMemoryV4류와 달리 "확정된 지난 구간"은 기억하지
//   않는다. 지금 진행 중인 구간의 실시간 값만 내놓는다.
// - 상승/하락을 완전히 분리된 변수(핵심실시간상승.../핵심실시간하락...)
//   로 관리한다 ? 방향이 바뀌면 반대쪽 값은 그 시점 값 그대로 멈춰있고,
//   새로 시작하는 쪽만 초기화되어 갱신된다.
// - 이 함수는 Plot을 하지 않는다.
//
// 설계 메모 (WSF_SwingFibMemoryV4와 동일한 원칙)
// - 지속 상태는 전부 내부 Var(상태_...)에만 저장하고, NumericRef
//   출력 파라미터는 매 호출 끝에서 값만 "쓰기"로 채운다.
// - 핵심세션미래방향의 과거값은 [1] 인덱싱 대신 상태_직전미래방향에
//   저장해뒀다가 다음 호출에서 비교한다.
// - 호출부는 반드시 매 봉 이 함수를 호출해야 한다.
// - 리셋 기준은 DayIndex==0(1분봉 등 세션이 매일 새로 시작하는
//   차트 기준)이다. 일봉처럼 세션 리셋 개념이 없는 차트에서 쓰려면
//   CurrentBar==1 기준의 _day 버전을 별도로 요청하면 된다.
//======================================================================

Var : 상태_핵심실시간방향(0), 상태_직전미래방향(0),
      상태_핵심실시간상승최고가(0), 상태_핵심실시간상승최저가(0), 상태_핵심실시간상승Count(0),
      상태_핵심실시간하락최고가(0), 상태_핵심실시간하락최저가(0), 상태_핵심실시간하락Count(0);

If DayIndex == 0 Then
{
    상태_핵심실시간방향 = 0;
    상태_직전미래방향 = 0;

    상태_핵심실시간상승최고가 = 0; 상태_핵심실시간상승최저가 = 0; 상태_핵심실시간상승Count = 0;
    상태_핵심실시간하락최고가 = 0; 상태_핵심실시간하락최저가 = 0; 상태_핵심실시간하락Count = 0;
}
Else
{
    // 방향 전환 감지 ? 0선을 넘어가는 순간만 잡는다
    // (직전값 <=0 → 신규값 >0, 그 반대도 동일)
    If 상태_직전미래방향 <= 0 && 핵심세션미래방향 > 0 Then
    {
        상태_핵심실시간상승최고가 = H;
        상태_핵심실시간상승최저가 = L;
        상태_핵심실시간상승Count = 1;
        상태_핵심실시간방향 = 1;
    }
    Else If 상태_직전미래방향 >= 0 && 핵심세션미래방향 < 0 Then
    {
        상태_핵심실시간하락최고가 = H;
        상태_핵심실시간하락최저가 = L;
        상태_핵심실시간하락Count = 1;
        상태_핵심실시간방향 = -1;
    }
    Else If 상태_핵심실시간방향 == 1 && 핵심세션미래방향 > 0 Then
    {
        상태_핵심실시간상승최고가 = Max(상태_핵심실시간상승최고가,H);
        상태_핵심실시간상승최저가 = Min(상태_핵심실시간상승최저가,L);
        상태_핵심실시간상승Count = 상태_핵심실시간상승Count + 1;
    }
    Else If 상태_핵심실시간방향 == -1 && 핵심세션미래방향 < 0 Then
    {
        상태_핵심실시간하락최고가 = Max(상태_핵심실시간하락최고가,H);
        상태_핵심실시간하락최저가 = Min(상태_핵심실시간하락최저가,L);
        상태_핵심실시간하락Count = 상태_핵심실시간하락Count + 1;
    };

    상태_직전미래방향 = 핵심세션미래방향;
};

// 출력 ? 내부 상태값을 그대로 복사해서 넘긴다
핵심실시간방향 = 상태_핵심실시간방향;

핵심실시간상승최고가 = 상태_핵심실시간상승최고가;
핵심실시간상승최저가 = 상태_핵심실시간상승최저가;
핵심실시간상승Count = 상태_핵심실시간상승Count;

핵심실시간하락최고가 = 상태_핵심실시간하락최고가;
핵심실시간하락최저가 = 상태_핵심실시간하락최저가;
핵심실시간하락Count = 상태_핵심실시간하락Count;

Value1 = 1;
WSF_DirSwingRealtimeV1 = Value1;
